    31→/** Re-scale one side's enabled rows so they always sum to 100%.
    32→ *  PCT mode only — FIXED (₹) allocations are absolute, never normalized. */
    33→function normalizeSide(rows: StrikeRowCfg[], type: 'CE' | 'PE'): StrikeRowCfg[] {
    34→  const side = rows.filter(r => r.type === type && r.enabled)
    35→  if (side.length === 0) return rows
    36→  const list = side.map(r => ({ ...r }))
    37→  const n = list.length
    38→  // zero/unset allocs enter with an equal-share weight, then everything
    39→  // is proportionally scaled so the side sums to exactly 100
    40→  for (const r of list) if (!(r.allocPct > 0)) r.allocPct = 100 / n
    41→  const total = list.reduce((s, r) => s + r.allocPct, 0)
    42→  for (const r of list) r.allocPct = Math.round((r.allocPct / total) * 1000) / 10
    43→  // dump rounding drift on the largest allocation
    44→  const drift = Math.round((100 - list.reduce((s, r) => s + r.allocPct, 0)) * 10) / 10
    45→  if (drift !== 0) {
    46→    const big = list.reduce((a, b) => (b.allocPct > a.allocPct ? b : a))
    47→    big.allocPct = Math.round(Math.max(0, big.allocPct + drift) * 10) / 10
    48→  }
    49→  const fixed = new Map(list.map(r => [r.id, r] as const))
    50→  return rows.map(r => fixed.get(r.id) ?? r)
    51→}

   273→  /** Deploy every enabled strike of one side — the shared core used by
   274→   *  Deploy buttons AND the swap buttons (after the other side is closed). */
   275→  const deployRows = async (type: 'CE' | 'PE', rows: StrikeRowCfg[]) => {
   276→    let strikesOk = 0, strikesFail = 0, fills = 0, skipped = 0
   277→    // fresh quotes so a swap doesn't deploy off stale LTP cells
   278→    const allStrikes = [...new Set(rows.map(r => r.strike))]
   279→    const cd = allStrikes.length ? await api.optionChain(underlying!, st.expiry, allStrikes) : null
   280→    const fresh = new Map<number, ChainStrikeRow>((cd?.rows ?? []).map(r => [r.strike, r]))
   281→    for (const row of rows) {
   282→      const c = fresh.get(row.strike)
   283→      const cell = c ? (row.type === 'CE' ? c.ce : c.pe) : cellFor(row)
   284→      const alloc = allocMode === 'FIXED' ? (row.allocAmt ?? 0) : row.allocPct

   286→      try {
   287→        const res = await api.placeOrder({
   288→          symbol: cell.symbol, exchange: 'NFO', side: 'BUY', qty: 0,
   289→          orderType: 'MARKET', productType: 'MIS', source: 'MANUAL',
   290→          ...sizingFor(row),
   291→        })
   292→        const f = res.legs.filter(l => l.status === 'FILLED').length
   293→        fills += f
   294→        skipped += res.legs.length - f
   295→        f > 0 ? strikesOk++ : strikesFail++
   296→      } catch { strikesFail++ }
   297→    }
   298→    return { strikesOk, strikesFail, fills, skipped, total: rows.length }

   326→  const swapSide = async (from: 'CE' | 'PE') => {
   327→    const to = from === 'CE' ? 'PE' : 'CE'

   331→      let closedLegs = 0
   332→      if (fromSyms.length > 0) {
   333→        const so = await api.squareOffAll(false, fromSyms)

   335→      }

   343→      }
   344→      const r = await deployRows(to, toRows)
   345→      r.strikesOk > 0
   346→        ? toast.success(`Swap ${from}→${to}: ${from} flat (${closedLegs} legs) · ${r.strikesOk}/${toRows.length} ${to} strikes · ${r.fills} fills${r.skipped ? ` · ${r.skipped} skipped` : ''}`)
   347→        : toast.error(`Swap ${from}→${to} — ${from} flat but no ${to} fills`)
   348→    } catch (e) {
   349→      toast.error(e instanceof Error ? e.message : `Swap ${from}→${to} failed`)

   365→  /** B = plain market BUY of THIS strike; S = plain market SELL of THIS strike.

   367→  const rowOrder = async (row: StrikeRowCfg, side: 'BUY' | 'SELL') => {

   374→        ...sizingFor(row),

   395→  /** LIMIT entry at a specific price — auto-sized by the row's alloc % per account. */
   396→  const placeEntry = async (row: StrikeRowCfg, side: 'BUY' | 'SELL', price: number) => {
   397→    const cell = cellFor(row)
   398→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   399→    if (!Number.isFinite(price) || price <= 0) { toast.error('Enter a valid price'); return }
   400→    try {
   401→      const res = await api.placeOrder({
   402→        symbol: cell.symbol, exchange: 'NFO', side, qty: 0,
   403→        orderType: 'LIMIT', productType: 'MIS', source: 'MANUAL',
   404→        capitalPct: row.allocPct, limitPrice: price,
   405→      })
   406→      const f = res.legs.filter(l => l.status === 'FILLED').length
   407→      const r = res.legs.filter(l => l.status === 'OPEN').length
   408→      if (f + r === 0) toast.error(`Entry failed — ${res.message}`)

   783→// ------------------------------------------------------------
   784→// Per-strike allocation cell — % (with inline slider) or fixed ₹.
   785→// Local edit state, commits on blur/Enter/release (re-normalizing
   786→// on every keystroke would fight the typist).
   787→// ------------------------------------------------------------
   788→function AllocCell({ value, mode, enabled, onCommit }: {
   789→  value: number
   790→  mode: AllocMode
   791→  enabled: boolean
   792→  onCommit: (v: number) => void
   793→}) {
   794→  const [txt, setTxt] = useState(mode === 'FIXED' ? String(Math.round(value)) : String(value))
   795→  const [focus, setFocus] = useState(false)
   796→  useEffect(() => { if (!focus) setTxt(mode === 'FIXED' ? String(Math.round(value)) : String(Math.round(value * 10) / 10)) }, [value, focus, mode])
   797→
   798→  if (mode === 'FIXED') {
   799→    return (
   800→      <input
   801→        value={txt}
   802→        disabled={!enabled}
   803→        onChange={e => setTxt(e.target.value.replace(/[^0-9]/g, ''))}
   804→        onFocus={() => setFocus(true)}
   805→        onBlur={() => { setFocus(false); onCommit(Number(txt)) }}
   806→        onKeyDown={e => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   807→        title="Fixed ₹ allocation per account for this strike"
   808→        className="h-4.5 w-11 shrink-0 rounded bg-secondary/60 text-center text-[9px] tabular outline-none focus:ring-1 ring-primary/60 disabled:opacity-40"
   809→      />
   810→    )
   811→  }
   812→  return (
   813→    <span className="flex items-center gap-0.5 shrink-0" title="Alloc % of usable capital — drag the slider or type; side auto-normalizes to 100%">
   814→      <input
   815→        type="range"
   816→        min={0}
   817→        max={100}
   818→        step={1}
   819→        value={Math.max(0, Math.min(100, Number(txt) || 0))}
   820→        disabled={!enabled}
   821→        onChange={e => setTxt(e.target.value)}
   822→        onMouseUp={e => onCommit(Number((e.target as HTMLInputElement).value))}
   823→        onTouchEnd={e => onCommit(Number((e.target as HTMLInputElement).value))}
   824→        className="h-1 w-8 accent-[#3b82f6] cursor-pointer disabled:opacity-40"
   825→      />
   826→      <input
   827→        value={txt}
   828→        disabled={!enabled}
   829→        onChange={e => setTxt(e.target.value.replace(/[^0-9.]/g, ''))}
   830→        onFocus={() => setFocus(true)}
   831→        onBlur={() => { setFocus(false); onCommit(Number(txt)) }}
   832→        onKeyDown={e => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   833→        className="h-4.5 w-9 shrink-0 rounded bg-secondary/60 text-center text-[9px] tabular outline-none focus:ring-1 ring-primary/60 disabled:opacity-40"
   834→      />
   835→    </span>
   836→  )
   837→}

   840→// Inline "edit open position" editor + price-order desk:
   841→// Entry (limit at price) · Target (reduce-only limit) ·
   842→// StopLoss (reduce-only SL-M trigger)
   843→// ------------------------------------------------------------
   844→function StrikeRowEditor({ row, cell, lots, pos, lotSize, onApply, onExit, onEntry, onTarget, onStop, onRemove }: {
   845→  row: StrikeRowCfg
   846→  cell: OptionQuoteCell | null
   847→  lots: number
   848→  pos: { netQty: number; avgPrice: number; unrealizedPnl: number } | null
   849→  lotSize: number
   850→  onApply: (deltaLots: number) => Promise<void>
   851→  onExit: () => Promise<void>
   852→  onEntry: (side: 'BUY' | 'SELL', price: number) => Promise<void>
   853→  onTarget: (price: number) => Promise<void>
   854→  onStop: (price: number) => Promise<void>
   855→  onRemove: () => void
   856→}) {
   857→  const [delta, setDelta] = useState('')
   858→  const [entry, setEntry] = useState('')
   859→  const [target, setTarget] = useState('')
   860→  const [stop, setStop] = useState('')
   861→  const [busy, setBusy] = useState(false)
   862→  useEffect(() => setDelta(''), [lots])
   863→
   864→  const flat = !pos || pos.netQty === 0
   865→  const num = (s: string) => (s === '' ? NaN : Number(s))
   866→  const ok = (s: string) => Number.isFinite(num(s)) && num(s) > 0
   867→
   868→  const run = async (fn: () => Promise<void>) => {
   869→    setBusy(true)
   870→    try { await fn() } finally { setBusy(false) }
   871→  }
   872→
   873→  return (
   874→    <div className="px-2 pb-2 pt-1 space-y-1.5 border-t border-border/40 bg-secondary/20">
   875→      <div className="flex items-center justify-between text-[10px]">
   876→        <span className="text-muted-foreground">Position — {cell?.symbol ?? row.type}</span>
   877→        <span className="tabular">
   878→          {pos ? (
   879→            <>
   880→              {pos.netQty > 0 ? 'LONG' : 'SHORT'} {lots} lot(s) @ {fmtMoney(pos.avgPrice)} ·{' '}
   881→              <b className={pos.unrealizedPnl >= 0 ? 'term-up' : 'term-down'}>{fmtMoney(pos.unrealizedPnl)}</b>
   882→            </>
   883→          ) : 'flat'}
   884→        </span>
   885→      </div>
   886→
   887→      {/* price orders: entry / target / stoploss */}
   888→      <div className="space-y-1 rounded bg-card/60 p-1.5">
   889→        <div className="text-[8px] font-semibold uppercase tracking-wide text-muted-foreground">orders at price</div>
   890→        <div className="flex items-center gap-1 text-[10px]">
   891→          <span className="w-10 shrink-0 text-muted-foreground">Entry</span>
   892→          <Input
   893→            value={entry}
   894→            onChange={e => setEntry(e.target.value.replace(/[^0-9.]/g, ''))}
   895→            placeholder="limit ₹"
   896→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   897→          />
   898→          <button
   899→            disabled={busy || !ok(entry)}
   900→            onClick={() => run(async () => { await onEntry('BUY', num(entry)); setEntry('') })}
   901→            title={`BUY LIMIT ${row.strike} ${row.type} at this price (${row.allocPct}% per account)`}
   902→            className="px-1.5 h-6 rounded bg-up text-[#00c896] hover:brightness-125 text-[9px] font-bold disabled:opacity-40"
   903→          >B@</button>
   904→          <button
   905→            disabled={busy || !ok(entry)}
   906→            onClick={() => run(async () => { await onEntry('SELL', num(entry)); setEntry('') })}
   907→            title={`SELL LIMIT ${row.strike} ${row.type} at this price (${row.allocPct}% per account)`}
   908→            className="px-1.5 h-6 rounded bg-down text-[#ff4d5e] hover:brightness-125 text-[9px] font-bold disabled:opacity-40"
   909→          >S@</button>
   910→        </div>
   911→        <div className="flex items-center gap-1 text-[10px]">
   912→          <span className="w-10 shrink-0 text-muted-foreground">Target</span>
   913→          <Input
   914→            value={target}
   915→            onChange={e => setTarget(e.target.value.replace(/[^0-9.]/g, ''))}
   916→            placeholder={pos && pos.netQty > 0 ? 'book long ₹' : 'book short ₹'}
   917→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   918→          />
   919→          <button
   920→            disabled={busy || flat || !ok(target)}
   921→            onClick={() => run(async () => { await onTarget(num(target)); setTarget('') })}
   922→            title={`Reduce-only LIMIT at this price — closes the open position when LTP reaches it`}
   923→            className="px-1.5 h-6 rounded bg-secondary text-foreground hover:bg-accent text-[9px] font-bold disabled:opacity-40"
   924→          >Place</button>
   925→        </div>
   926→        <div className="flex items-center gap-1 text-[10px]">
   927→          <span className="w-10 shrink-0 text-muted-foreground">StopLoss</span>
   928→          <Input
   929→            value={stop}
   930→            onChange={e => setStop(e.target.value.replace(/[^0-9.]/g, ''))}
   931→            placeholder="trigger ₹"
   932→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   933→          />
   934→          <button
   935→            disabled={busy || flat || !ok(stop)}
   936→            onClick={() => run(async () => { await onStop(num(stop)); setStop('') })}
   937→            title={`Reduce-only SL-M — market-exits the open position when LTP triggers at this price`}
   938→            className="px-1.5 h-6 rounded bg-secondary text-foreground hover:bg-accent text-[9px] font-bold disabled:opacity-40"
   939→          >Arm</button>
   940→        </div>
   941→        <div className="text-[8px] text-muted-foreground leading-snug">
   942→          Entry rests as LIMIT until hit · Target/SL attach to the current position ({lots} lot(s) · lot {lotSize}) and auto-cancel if it closes
   943→        </div>
   944→      </div>
   945→
   946→      <div className="flex items-center gap-1 text-[10px]">
   947→        <span className="text-muted-foreground">Δ Lots /leg</span>
   948→        <Input
   949→          value={delta}
   950→          onChange={e => setDelta(e.target.value.replace(/[^0-9-]/g, ''))}
   951→          placeholder="+2 / -1"
   952→          className="h-6 w-16 text-[10px] bg-secondary/60"
   953→        />
   954→        <Button
   955→          size="sm" variant="secondary" className="h-6 px-2 text-[10px]" disabled={busy || !delta || delta === '-' || delta === '+0'}
   956→          onClick={async () => { setBusy(true); await onApply(Number(delta)); setBusy(false) }}
   957→        >Apply</Button>
   958→        <span className="text-[9px] text-muted-foreground">+ buys, − exits (per enabled account) · now {lots} lot(s) agg</span>
   959→        <div className="flex-1" />
   960→        <Button
   961→          size="sm" variant="secondary" className="h-6 px-2 text-[10px] text-[#ff4d5e]" disabled={busy || !pos}
   962→          onClick={async () => { setBusy(true); await onExit(); setBusy(false) }}
   963→        >Exit full</Button>
   964→        <Button size="sm" variant="ghost" className="h-6 px-1.5 text-[10px] text-muted-foreground hover:text-[#ff4d5e]" onClick={onRemove}>
   965→          <Trash2 className="h-3 w-3" />
   966→        </Button>
   967→      </div>
   968→      <div className="flex items-center gap-1 text-[10px]">
   969→        <span className="text-[9px] text-muted-foreground">alloc {row.allocPct}% · lot {lotSize} · orders fan out to all enabled accounts</span>
   970→      </div>
   971→    </div>
   972→  )
