    30→    30→/** Re-scale one side's enabled rows so they always sum to 100%.
    31→    31→ *  PCT mode only — FIXED (₹) allocations are absolute, never normalized. */
    32→    32→function normalizeSide(rows: StrikeRowCfg[], type: 'CE' | 'PE'): StrikeRowCfg[] {
    33→    33→  const side = rows.filter(r => r.type === type && r.enabled)
    34→    34→  if (side.length === 0) return rows
    35→    35→  const list = side.map(r => ({ ...r }))
    36→    36→  const n = list.length
    37→    37→  // zero/unset allocs enter with an equal-share weight, then everything
    38→    38→  // is proportionally scaled so the side sums to exactly 100
    39→    39→  for (const r of list) if (!(r.allocPct > 0)) r.allocPct = 100 / n
    40→    40→  const total = list.reduce((s, r) => s + r.allocPct, 0)
    41→    41→  for (const r of list) r.allocPct = Math.round((r.allocPct / total) * 1000) / 10
    42→    42→  // dump rounding drift on the largest allocation
    43→    43→  const drift = Math.round((100 - list.reduce((s, r) => s + r.allocPct, 0)) * 10) / 10
    44→    44→  if (drift !== 0) {
    45→    45→    const big = list.reduce((a, b) => (b.allocPct > a.allocPct ? b : a))
    46→    46→    big.allocPct = Math.round(Math.max(0, big.allocPct + drift) * 10) / 10
    47→    47→  }
    48→    48→  const fixed = new Map(list.map(r => [r.id, r] as const))
    49→    49→  return rows.map(r => fixed.get(r.id) ?? r)
    50→    50→}

   202→   202→  /** Deploy every enabled strike of one side — the shared core used by
   203→   203→   *  Deploy buttons AND the swap buttons (after the other side is closed). */
   204→   204→  const deployRows = async (type: 'CE' | 'PE', rows: StrikeRowCfg[]) => {
   205→   205→    let strikesOk = 0, strikesFail = 0, fills = 0, skipped = 0
   206→   206→    // fresh quotes so a swap doesn't deploy off stale LTP cells
   207→   207→    const allStrikes = [...new Set(rows.map(r => r.strike))]
   208→   208→    const cd = allStrikes.length ? await api.optionChain(underlying!, st.expiry, allStrikes) : null
   209→   209→    const fresh = new Map<number, ChainStrikeRow>((cd?.rows ?? []).map(r => [r.strike, r]))
   210→   210→    for (const row of rows) {
   211→   211→      const c = fresh.get(row.strike)
   212→   212→      const cell = c ? (row.type === 'CE' ? c.ce : c.pe) : cellFor(row)
   213→   213→      const alloc = allocMode === 'FIXED' ? (row.allocAmt ?? 0) : row.allocPct
   214→   214→      if (!cell?.exists || !(alloc > 0)) { strikesFail++; continue }
   215→   215→      try {
   216→   216→        const res = await api.placeOrder({
   217→   217→          symbol: cell.symbol, exchange: 'NFO', side: 'BUY', qty: 0,
   218→   218→          orderType: 'MARKET', productType: 'MIS', source: 'MANUAL',
   219→   219→          ...sizingFor(row),
   220→   220→        })
   221→   221→        const f = res.legs.filter(l => l.status === 'FILLED').length
   222→   222→        fills += f
   223→   223→        skipped += res.legs.length - f
   224→   224→        f > 0 ? strikesOk++ : strikesFail++
   225→   225→      } catch { strikesFail++ }
   226→   226→    }
   227→   227→    return { strikesOk, strikesFail, fills, skipped, total: rows.length }
   228→   228→  }
   229→   229→
   230→   230→  const deploy = async (type: 'CE' | 'PE') => {
   231→   231→    const rows = rowsFor(type) // ONLY enabled strikes participate
   232→   232→    if (rows.length === 0) { toast.error(`No enabled ${type} strikes`); return }
   233→   233→    setBusy(true)
   234→   234→    const r = await deployRows(type, rows)
   235→   235→    setBusy(false)
   236→   236→    r.strikesOk > 0
   237→   237→      ? toast.success(`Deploy ${type}: ${r.strikesOk}/${rows.length} enabled strikes · ${r.fills} account fills${r.skipped ? ` · ${r.skipped} legs skipped` : ''}`)
   238→   238→      : toast.error(`Deploy ${type} failed — no fills (${r.strikesFail} strikes skipped)`)
   239→   239→  }

   242→   242→    const syms = rowsFor(type).map(r => cellFor(r)?.symbol).filter((s): s is string => !!s)
   243→   243→    if (syms.length === 0) { toast.error(`No enabled ${type} strikes`); return }
   244→   244→    try {
   245→   245→      const r = await api.cancelAll(syms)
   246→   246→      r.cancelled > 0 ? toast.success(`Cancelled ${r.cancelled} pending ${type} orders`) : toast.info(`No pending ${type} orders`)
   247→   247→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Cancel failed') }
   248→   248→  }
   249→   249→
   250→   250→  const peSwap = async () => swapSide('PE')
   251→   251→
   252→   252→  /** SWAP: close every enabled position of side `from`, then deploy the
   253→   253→   *  opposite side (CE↔PE for FNO; for equities the order pad B/S covers
   254→   254→   *  buy↔sell). Auto-adds an ATM strike of the target side if none listed. */
   255→   255→  const swapSide = async (from: 'CE' | 'PE') => {
   256→   256→    const to = from === 'CE' ? 'PE' : 'CE'
   257→   257→    const fromSyms = rowsFor(from).map(r => cellFor(r)?.symbol).filter((s): s is string => !!s)
   258→   258→    setBusy(true)
   259→   259→    try {
   260→   260→      let closedLegs = 0
   261→   261→      if (fromSyms.length > 0) {
   262→   262→        const so = await api.squareOffAll(false, fromSyms)
   263→   263→        closedLegs = so.filledLegs
   264→   264→      }
   265→   265→      // deploy target side on enabled rows (auto-add side ATM if none listed)
   266→   266→      let toRows = rowsFor(to)
   267→   267→      if (toRows.length === 0 && meta) {
   268→   268→        await addStrike(to === 'CE' ? meta.ceAtm : meta.peAtm, to)
   269→   269→        await new Promise(r => setTimeout(r, 500))
   270→   270→        const ws2 = useTerminal.getState().workspaces.find(w => w.id === ws.id)
   271→   271→        toRows = (ws2?.strikes?.rows ?? []).filter(r => r.type === to && r.enabled)
   272→   272→      }
   273→   273→      const r = await deployRows(to, toRows)
   274→   274→      r.strikesOk > 0
   275→   275→        ? toast.success(`Swap ${from}→${to}: ${from} flat (${closedLegs} legs) · ${r.strikesOk}/${toRows.length} ${to} strikes · ${r.fills} fills${r.skipped ? ` · ${r.skipped} skipped` : ''}`)
   276→   276→        : toast.error(`Swap ${from}→${to} — ${from} flat but no ${to} fills`)
   277→   277→    } catch (e) {
   278→   278→      toast.error(e instanceof Error ? e.message : `Swap ${from}→${to} failed`)
   279→   279→    } finally { setBusy(false) }
   280→   280→  }
   281→   281→
   282→   282→  /** EXIT ALL — square off every open position across all symbols/accounts. */
   283→   283→  const exitAll = async () => {
   284→   284→    setBusy(true)
   285→   285→    try {
   286→   286→      const so = await api.squareOffAll(false)
   287→   287→      so.orders > 0
   288→   288→        ? toast.success(`Exit All — ${so.orders} orders · ${so.filledLegs} legs closed`)
   289→   289→        : toast.info('No open positions')
   290→   290→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Exit All failed') }
   291→   291→    finally { setBusy(false) }
   292→   292→  }
   293→   293→
   294→   294→  /** B = plain market BUY of THIS strike; S = plain market SELL of THIS strike.
   295→   295→   *  Sizing = the row's alloc (PCT % of usable capital, or fixed ₹) per account. */
   296→   296→  const rowOrder = async (row: StrikeRowCfg, side: 'BUY' | 'SELL') => {
   297→   297→    const cell = cellFor(row)
   298→   298→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   299→   299→    try {
   300→   300→      const res = await api.placeOrder({
   301→   301→        symbol: cell.symbol, exchange: 'NFO', side, qty: 0,
   302→   302→        orderType: 'MARKET', productType: 'MIS', source: 'MANUAL',
   303→   303→        ...sizingFor(row),
   304→   304→      })
   305→   305→      res.status === 'FAILED'
   306→   306→        ? toast.error(`${side} failed — ${res.message}`)
   307→   307→        : toast.success(`${side} ${row.strike} ${row.type} @ MKT — ${res.legs.filter(l => l.status === 'FILLED').length}/${res.legs.length} accounts`)
   308→   308→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Order failed') }
   309→   309→  }
   310→   310→
   311→   311→  const rowExit = async (row: StrikeRowCfg) => {
   312→   312→    const cell = cellFor(row)
   313→   313→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   314→   314→    const pos = posBySymbol.get(cell.symbol)
   315→   315→    if (!pos || pos.netQty === 0) { toast.info('No open position'); return }
   316→   316→    try {
   317→   317→      const so = await api.squareOffAll(false, [cell.symbol])
   318→   318→      so.orders > 0 ? toast.success(`Exit ${row.strike} ${row.type} — ${so.filledLegs} legs`) : toast.info('Nothing to exit')
   319→   319→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Exit failed') }
   320→   320→  }
   321→   321→
   322→   322→  // --- price-order helpers (entry / target / stoploss at a specific price) ---
   323→   323→
   324→   324→  /** LIMIT entry at a specific price — auto-sized by the row's alloc % per account. */
   325→   325→  const placeEntry = async (row: StrikeRowCfg, side: 'BUY' | 'SELL', price: number) => {
   326→   326→    const cell = cellFor(row)
   327→   327→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   328→   328→    if (!Number.isFinite(price) || price <= 0) { toast.error('Enter a valid price'); return }
   329→   329→    try {
   330→   330→      const res = await api.placeOrder({
   331→   331→        symbol: cell.symbol, exchange: 'NFO', side, qty: 0,
   332→   332→        orderType: 'LIMIT', productType: 'MIS', source: 'MANUAL',
   333→   333→        capitalPct: row.allocPct, limitPrice: price,
   334→   334→      })
   335→   335→      const f = res.legs.filter(l => l.status === 'FILLED').length
   336→   336→      const r = res.legs.filter(l => l.status === 'OPEN').length
   337→   337→      if (f + r === 0) toast.error(`Entry failed — ${res.message}`)
   338→   338→      else toast.success(`${side} ${row.strike} ${row.type} LIMIT @ ${fmtMoney(price)} — ${f} filled${r ? ` · ${r} resting` : ''}`)
   339→   339→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Order failed') }
   340→   340→  }
   341→   341→
   342→   342→  /** Target — reduce-only LIMIT at the price (books the open position there). */
   343→   343→  const placeTarget = async (row: StrikeRowCfg, price: number) => {
   344→   344→    const cell = cellFor(row)
   345→   345→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   346→   346→    if (!Number.isFinite(price) || price <= 0) { toast.error('Enter a valid price'); return }
   347→   347→    const pos = posBySymbol.get(cell.symbol)
   348→   348→    if (!pos || pos.netQty === 0) { toast.info('No open position to set a target on'); return }
   349→   349→    const side = pos.netQty > 0 ? 'SELL' : 'BUY'
   350→   350→    const lots = Math.abs(pos.netQty) / (meta?.lotSize || 1)
   351→   351→    try {
   352→   352→      const res = await api.placeOrder({

   669→   669→// ------------------------------------------------------------
   670→   670→// Per-strike allocation cell — % (with inline slider) or fixed ₹.
   671→   671→// Local edit state, commits on blur/Enter/release (re-normalizing
   672→   672→// on every keystroke would fight the typist).
   673→   673→// ------------------------------------------------------------
   674→   674→function AllocCell({ value, mode, enabled, onCommit }: {
   675→   675→  value: number
   676→   676→  mode: AllocMode
   677→   677→  enabled: boolean
   678→   678→  onCommit: (v: number) => void
   679→   679→}) {
   680→   680→  const [txt, setTxt] = useState(mode === 'FIXED' ? String(Math.round(value)) : String(value))
   681→   681→  const [focus, setFocus] = useState(false)
   682→   682→  useEffect(() => { if (!focus) setTxt(mode === 'FIXED' ? String(Math.round(value)) : String(Math.round(value * 10) / 10)) }, [value, focus, mode])
   683→   683→
   684→   684→  if (mode === 'FIXED') {
   685→   685→    return (
   686→   686→      <input

   733→   733→  lots: number
