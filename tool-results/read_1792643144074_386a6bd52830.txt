    34→/** Re-scale one side's enabled rows so they always sum to 100%.
    35→ *  PCT mode only — FIXED (₹) allocations are absolute, never normalized. */
    36→function normalizeSide(rows: StrikeRowCfg[], type: 'CE' | 'PE'): StrikeRowCfg[] {
    37→  const side = rows.filter(r => r.type === type && r.enabled)
    38→  if (side.length === 0) return rows
    39→  const list = side.map(r => ({ ...r }))
    40→  const n = list.length
    41→  // zero/unset allocs enter with an equal-share weight, then everything
    42→  // is proportionally scaled so the side sums to exactly 100
    43→  for (const r of list) if (!(r.allocPct > 0)) r.allocPct = 100 / n
    44→  const total = list.reduce((s, r) => s + r.allocPct, 0)
    45→  for (const r of list) r.allocPct = Math.round((r.allocPct / total) * 1000) / 10
    46→  // dump rounding drift on the largest allocation
    47→  const drift = Math.round((100 - list.reduce((s, r) => s + r.allocPct, 0)) * 10) / 10
    48→  if (drift !== 0) {
    49→    const big = list.reduce((a, b) => (b.allocPct > a.allocPct ? b : a))
    50→    big.allocPct = Math.round(Math.max(0, big.allocPct + drift) * 10) / 10
    51→  }
    52→  const fixed = new Map(list.map(r => [r.id, r] as const))
    53→  return rows.map(r => fixed.get(r.id) ?? r)
    54→}

   282→  const deployRows = async (type: 'CE' | 'PE', rows: StrikeRowCfg[]) => {
   283→    let strikesOk = 0, strikesFail = 0, fills = 0, skipped = 0
   284→    // fresh quotes so a swap doesn't deploy off stale LTP cells

   289→      const c = fresh.get(row.strike)
   290→      const cell = c ? (row.type === 'CE' ? c.ce : c.pe) : cellFor(row)
   291→      const alloc = allocMode === 'FIXED' ? (row.allocAmt ?? 0) : row.allocPct
   292→      if (!cell?.exists || !(alloc > 0)) { strikesFail++; continue }

   301→        skipped += res.legs.length - f
   302→        f > 0 ? strikesOk++ : strikesFail++
   303→      } catch { strikesFail++ }
   304→    }
   305→    return { strikesOk, strikesFail, fills, skipped, total: rows.length }

   354→        : toast.error(`Swap ${from}→${to} — ${from} flat but no ${to} fills`)

   402→  /** LIMIT entry at a specific price — auto-sized by the row's alloc % per account. */
   403→  const placeEntry = async (row: StrikeRowCfg, side: 'BUY' | 'SELL', price: number) => {

   856→// ------------------------------------------------------------
   857→// Per-strike allocation cell — % (with inline slider) or fixed ₹.
   858→// Local edit state, commits on blur/Enter/release (re-normalizing
   859→// on every keystroke would fight the typist).
   860→// ------------------------------------------------------------
   861→function AllocCell({ value, mode, enabled, onCommit }: {
   862→  value: number
   863→  mode: AllocMode
   864→  enabled: boolean
   865→  onCommit: (v: number) => void
   866→}) {
   867→  const [txt, setTxt] = useState(mode === 'FIXED' ? String(Math.round(value)) : String(value))
   868→  const [focus, setFocus] = useState(false)
   869→  useEffect(() => { if (!focus) setTxt(mode === 'FIXED' ? String(Math.round(value)) : String(Math.round(value * 10) / 10)) }, [value, focus, mode])
   870→
   871→  if (mode === 'FIXED') {
   872→    return (
   873→      <input
   874→        value={txt}
   875→        disabled={!enabled}
   876→        onChange={e => setTxt(e.target.value.replace(/[^0-9]/g, ''))}
   877→        onFocus={() => setFocus(true)}
   878→        onBlur={() => { setFocus(false); onCommit(Number(txt)) }}
   879→        onKeyDown={e => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   880→        title="Fixed ₹ allocation per account for this strike"
   881→        className="h-4.5 w-11 shrink-0 rounded bg-secondary/60 text-center text-[9px] tabular outline-none focus:ring-1 ring-primary/60 disabled:opacity-40"
   882→      />
   883→    )
   884→  }
   885→  return (
   886→    <span className="flex items-center gap-1 flex-1 min-w-0 max-w-[150px] shrink" title="Alloc % of usable capital — drag the slider or type; side auto-normalizes to 100%">
   887→      <input
   888→        type="range"
   889→        min={0}
   890→        max={100}
   891→        step={1}
   892→        value={Math.max(0, Math.min(100, Number(txt) || 0))}
   893→        disabled={!enabled}
   894→        onChange={e => setTxt(e.target.value)}
   895→        onMouseUp={e => onCommit(Number((e.target as HTMLInputElement).value))}
   896→        onTouchEnd={e => onCommit(Number((e.target as HTMLInputElement).value))}
   897→        className="h-1 flex-1 min-w-8 accent-[#3b82f6] cursor-pointer disabled:opacity-40"
   898→      />
   899→      <input
   900→        value={txt}
   901→        disabled={!enabled}
   902→        onChange={e => setTxt(e.target.value.replace(/[^0-9.]/g, ''))}
   903→        onFocus={() => setFocus(true)}
   904→        onBlur={() => { setFocus(false); onCommit(Number(txt)) }}
   905→        onKeyDown={e => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   906→        className="h-4.5 w-9 shrink-0 rounded bg-secondary/60 text-center text-[9px] tabular outline-none focus:ring-1 ring-primary/60 disabled:opacity-40"
   907→      />
   908→    </span>
   909→  )
   910→}

   913→// Inline "edit open position" editor + price-order desk:
   914→// Entry (limit at price) · Target (reduce-only limit) ·
   915→// StopLoss (reduce-only SL-M trigger)
   916→// ------------------------------------------------------------
   917→function StrikeRowEditor({ row, cell, lots, pos, lotSize, onApply, onExit, onEntry, onTarget, onStop, onRemove }: {
   918→  row: StrikeRowCfg
   919→  cell: OptionQuoteCell | null
   920→  lots: number
   921→  pos: { netQty: number; avgPrice: number; unrealizedPnl: number } | null
   922→  lotSize: number
   923→  onApply: (deltaLots: number) => Promise<void>
   924→  onExit: () => Promise<void>
   925→  onEntry: (side: 'BUY' | 'SELL', price: number) => Promise<void>
   926→  onTarget: (price: number) => Promise<void>
   927→  onStop: (price: number) => Promise<void>
   928→  onRemove: () => void
   929→}) {
   930→  const [delta, setDelta] = useState('')
   931→  const [entry, setEntry] = useState('')
   932→  const [target, setTarget] = useState('')
   933→  const [stop, setStop] = useState('')
   934→  const [busy, setBusy] = useState(false)
   935→  useEffect(() => setDelta(''), [lots])
   936→
   937→  const flat = !pos || pos.netQty === 0
   938→  const num = (s: string) => (s === '' ? NaN : Number(s))
   939→  const ok = (s: string) => Number.isFinite(num(s)) && num(s) > 0
   940→
   941→  const run = async (fn: () => Promise<void>) => {
   942→    setBusy(true)
   943→    try { await fn() } finally { setBusy(false) }
   944→  }
   945→
   946→  return (
   947→    <div className="px-2 pb-2 pt-1 space-y-1.5 border-t border-border/40 bg-secondary/20">
   948→      <div className="flex items-center justify-between text-[10px]">
   949→        <span className="text-muted-foreground">Position — {cell?.symbol ?? row.type}</span>
   950→        <span className="tabular">
   951→          {pos ? (
   952→            <>
   953→              {pos.netQty > 0 ? 'LONG' : 'SHORT'} {lots} lot(s) @ {fmtMoney(pos.avgPrice)} ·{' '}
   954→              <b className={pos.unrealizedPnl >= 0 ? 'term-up' : 'term-down'}>{fmtMoney(pos.unrealizedPnl)}</b>
   955→            </>
   956→          ) : 'flat'}
   957→        </span>
   958→      </div>
   959→
   960→      {/* price orders: entry / target / stoploss */}
   961→      <div className="space-y-1 rounded bg-card/60 p-1.5">
   962→        <div className="text-[8px] font-semibold uppercase tracking-wide text-muted-foreground">orders at price</div>
   963→        <div className="flex items-center gap-1 text-[10px]">
   964→          <span className="w-10 shrink-0 text-muted-foreground">Entry</span>
   965→          <Input
   966→            value={entry}
   967→            onChange={e => setEntry(e.target.value.replace(/[^0-9.]/g, ''))}
   968→            placeholder="limit ₹"
   969→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   970→          />
   971→          <button
   972→            disabled={busy || !ok(entry)}
   973→            onClick={() => run(async () => { await onEntry('BUY', num(entry)); setEntry('') })}
   974→            title={`BUY LIMIT ${row.strike} ${row.type} at this price (${row.allocPct}% per account)`}
   975→            className="px-1.5 h-6 rounded bg-up text-[#00c896] hover:brightness-125 text-[9px] font-bold disabled:opacity-40"
   976→          >B@</button>
   977→          <button
   978→            disabled={busy || !ok(entry)}
   979→            onClick={() => run(async () => { await onEntry('SELL', num(entry)); setEntry('') })}
   980→            title={`SELL LIMIT ${row.strike} ${row.type} at this price (${row.allocPct}% per account)`}
   981→            className="px-1.5 h-6 rounded bg-down text-[#ff4d5e] hover:brightness-125 text-[9px] font-bold disabled:opacity-40"
   982→          >S@</button>
   983→        </div>
   984→        <div className="flex items-center gap-1 text-[10px]">
   985→          <span className="w-10 shrink-0 text-muted-foreground">Target</span>
   986→          <Input
   987→            value={target}
   988→            onChange={e => setTarget(e.target.value.replace(/[^0-9.]/g, ''))}
   989→            placeholder={pos && pos.netQty > 0 ? 'book long ₹' : 'book short ₹'}
   990→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   991→          />
   992→          <button
   993→            disabled={busy || flat || !ok(target)}
   994→            onClick={() => run(async () => { await onTarget(num(target)); setTarget('') })}
   995→            title={`Reduce-only LIMIT at this price — closes the open position when LTP reaches it`}
   996→            className="px-1.5 h-6 rounded bg-secondary text-foreground hover:bg-accent text-[9px] font-bold disabled:opacity-40"
   997→          >Place</button>
   998→        </div>
   999→        <div className="flex items-center gap-1 text-[10px]">
  1000→          <span className="w-10 shrink-0 text-muted-foreground">StopLoss</span>
  1001→          <Input
  1002→            value={stop}
  1003→            onChange={e => setStop(e.target.value.replace(/[^0-9.]/g, ''))}
  1004→            placeholder="trigger ₹"
  1005→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
  1006→          />
  1007→          <button
  1008→            disabled={busy || flat || !ok(stop)}
  1009→            onClick={() => run(async () => { await onStop(num(stop)); setStop('') })}
  1010→            title={`Reduce-only SL-M — market-exits the open position when LTP triggers at this price`}
  1011→            className="px-1.5 h-6 rounded bg-secondary text-foreground hover:bg-accent text-[9px] font-bold disabled:opacity-40"
  1012→          >Arm</button>
  1013→        </div>
  1014→        <div className="text-[8px] text-muted-foreground leading-snug">
  1015→          Entry rests as LIMIT until hit · Target/SL attach to the current position ({lots} lot(s) · lot {lotSize}) and auto-cancel if it closes
  1016→        </div>
  1017→      </div>
  1018→
  1019→      <div className="flex items-center gap-1 text-[10px]">
  1020→        <span className="text-muted-foreground">Δ Lots /leg</span>
  1021→        <Input
  1022→          value={delta}
  1023→          onChange={e => setDelta(e.target.value.replace(/[^0-9-]/g, ''))}
  1024→          placeholder="+2 / -1"
  1025→          className="h-6 w-16 text-[10px] bg-secondary/60"
  1026→        />
  1027→        <Button
  1028→          size="sm" variant="secondary" className="h-6 px-2 text-[10px]" disabled={busy || !delta || delta === '-' || delta === '+0'}
  1029→          onClick={async () => { setBusy(true); await onApply(Number(delta)); setBusy(false) }}
  1030→        >Apply</Button>
  1031→        <span className="text-[9px] text-muted-foreground">+ buys, − exits (per enabled account) · now {lots} lot(s) agg</span>
  1032→        <div className="flex-1" />
  1033→        <Button
  1034→          size="sm" variant="secondary" className="h-6 px-2 text-[10px] text-[#ff4d5e]" disabled={busy || !pos}
  1035→          onClick={async () => { setBusy(true); await onExit(); setBusy(false) }}
  1036→        >Exit full</Button>
  1037→        <Button size="sm" variant="ghost" className="h-6 px-1.5 text-[10px] text-muted-foreground hover:text-[#ff4d5e]" onClick={onRemove}>
  1038→          <Trash2 className="h-3 w-3" />
  1039→        </Button>
  1040→      </div>
  1041→      <div className="flex items-center gap-1 text-[10px]">
  1042→        <span className="text-[9px] text-muted-foreground">alloc {row.allocPct}% · lot {lotSize} · orders fan out to all enabled accounts</span>
  1043→      </div>
  1044→    </div>
  1045→  )
  1046→}
