    66→function sma(data: Candle[], period: number): (number | null)[] {
    67→  const out: (number | null)[] = []
    68→  let sum = 0
    69→  for (let i = 0; i < data.length; i++) {
    70→    sum += data[i].c
    71→    if (i >= period) sum -= data[i - period].c
    72→    out.push(i >= period - 1 ? sum / period : null)
    73→  }
    74→  return out
    75→}
    76→
    77→function ema(data: Candle[], period: number): (number | null)[] {
    78→  const out: (number | null)[] = []
    79→  const k = 2 / (period + 1)
    80→  let prev = 0
    81→  for (let i = 0; i < data.length; i++) {
    82→    if (i < period - 1) { out.push(null); prev += data[i].c; continue }
    83→    if (i === period - 1) { prev = (prev + data[i].c) / period; out.push(prev); continue }
    84→    prev = data[i].c * k + prev * (1 - k)
    85→    out.push(prev)
    86→  }
    87→  return out
    88→}

    90→/** Session VWAP — resets each calendar day (IST) like real intraday VWAP. */
    91→function vwap(data: Candle[]): (number | null)[] {
    92→  const out: (number | null)[] = []
    93→  let day = '', cumPV = 0, cumV = 0
    94→  for (const c of data) {
    95→    const d = new Date(c.t + 5.5 * 3600_000).toISOString().slice(0, 10) // IST day bucket
    96→    if (d !== day) { day = d; cumPV = 0; cumV = 0 }
    97→    const tp = (c.h + c.l + c.c) / 3
    98→    cumPV += tp * (c.v || 1)
    99→    cumV += c.v || 1
   100→    out.push(cumV > 0 ? cumPV / cumV : null)
   101→  }
   102→  return out
   103→}

   105→function bollinger(data: Candle[], period = 20, mult = 2) {
   106→  const mid = sma(data, period)
   107→  const up: (number | null)[] = []
   108→  const lo: (number | null)[] = []
   109→  for (let i = 0; i < data.length; i++) {
   110→    if (mid[i] == null) { up.push(null); lo.push(null); continue }
   111→    let s = 0
   112→    for (let j = i - period + 1; j <= i; j++) s += (data[j].c - mid[i]!) ** 2
   113→    const sd = Math.sqrt(s / period)
   114→    up.push(mid[i]! + mult * sd)
   115→    lo.push(mid[i]! - mult * sd)
   116→  }
   117→  return { mid, up, lo }
   118→}

   120→function rsi(data: Candle[], period = 14): (number | null)[] {
   121→  const out: (number | null)[] = []
   122→  let avgG = 0, avgL = 0
   123→  for (let i = 0; i < data.length; i++) {
   124→    if (i === 0) { out.push(null); continue }
   125→    const ch = data[i].c - data[i - 1].c
   126→    const g = Math.max(0, ch), l = Math.max(0, -ch)
   127→    if (i <= period) {
   128→      avgG += g / period; avgL += l / period
   129→      out.push(i === period ? toRsi(avgG, avgL) : null)
   130→      continue
   131→    }
   132→    avgG = (avgG * (period - 1) + g) / period
   133→    avgL = (avgL * (period - 1) + l) / period
   134→    out.push(toRsi(avgG, avgL))
   135→  }
   136→  return out
   137→}
   138→function toRsi(g: number, l: number): number | null {
   139→  if (l === 0) return 100
   140→  const rs = g / l
   141→  return 100 - 100 / (1 + rs)
   142→}

   144→function macd(data: Candle[], fast = 12, slow = 26, sig = 9) {
   145→  const ef = ema(data, fast)
   146→  const es = ema(data, slow)
   147→  const line = data.map((_, i) => (ef[i] != null && es[i] != null ? ef[i]! - es[i]! : null))
   148→  // signal = EMA9 of the macd line (skip nulls at head)
   149→  const signal: (number | null)[] = []
   150→  const k = 2 / (sig + 1)
   151→  let prev = 0
   152→  let seeded = false
   153→  for (let i = 0; i < data.length; i++) {
   154→    if (line[i] == null) { signal.push(null); continue }
   155→    if (!seeded) { prev = line[i]!; seeded = true; signal.push(prev); continue }
   156→    prev = line[i]! * k + prev * (1 - k)
   157→    signal.push(prev)
   158→  }
   159→  const hist = data.map((_, i) => (line[i] != null && signal[i] != null ? line[i]! - signal[i]! : null))
   160→  return { line, signal, hist }
   161→}

   175→function niceTicks(min: number, max: number, count: number): number[] {
   176→  const span = max - min
   177→  if (span <= 0) return [min]
   178→  const raw = span / count
   179→  const mag = Math.pow(10, Math.floor(Math.log10(raw)))
   180→  const norm = raw / mag
   181→  const step = (norm >= 5 ? 10 : norm >= 2 ? 5 : norm >= 1 ? 2 : 1) * mag
   182→  const start = Math.ceil(min / step) * step
   183→  const out: number[] = []
   184→  for (let v = start; v <= max; v += step) out.push(v)
   185→  return out
   186→}

   188→function fmtAxisTime(t: number, tf: string): string {
   189→  const d = new Date(t)
   190→  if (tf === '1d') return d.toLocaleDateString('en-IN', { day: '2-digit', month: 'short' })
   191→  return d.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', hour12: false })
   192→}

   194→function fmtVol(v: number): string {
   195→  if (v >= 1e7) return `${(v / 1e7).toFixed(1)}Cr`
   196→  if (v >= 1e5) return `${(v / 1e5).toFixed(1)}L`
   197→  if (v >= 1000) return `${(v / 1000).toFixed(1)}k`
   198→  return String(Math.round(v))
   199→}

   201→// ---- scale labels (percent / points vs baseline) ----
   202→function labelPct(v: number): string {
   203→  const s = v >= 0 ? '+' : '-', a = Math.abs(v)
   204→  return s + (a >= 100 ? a.toFixed(0) : a >= 10 ? a.toFixed(1) : a.toFixed(2)) + '%'
   205→}
   206→function labelPts(v: number): string {
   207→  const s = v >= 0 ? '+' : '-', a = Math.abs(v)
   208→  return s + (a >= 1000 ? a.toFixed(0) : a >= 100 ? a.toFixed(1) : a.toFixed(2))
   209→}

   216→function lowerBoundTime(arr: Candle[], t: number): number {
   217→  let lo = 0, hi = arr.length
   218→  while (lo < hi) { const mid = (lo + hi) >> 1; if (arr[mid].t < t) lo = mid + 1; else hi = mid }
   219→  return lo
   220→}

   233→export function CanvasChart({
   234→  symbol, tf, chartType, showVolume, showSMA20, showSMA50,
   235→  showEMA = false, showVWAP = false, showBOLL = false, showRSI = false, showMACD = false,
   236→  showGhost = false, ghostStyle = 'hollow', scales,
   237→  markers = [], onPriceClick, height, mini = false,
   238→}: Props) {
   239→  const wrapRef = useRef<HTMLDivElement>(null)
   240→  const canvasRef = useRef<HTMLCanvasElement>(null)
   241→  const [candles, setCandles] = useState<Candle[]>([])
   242→  const [loading, setLoading] = useState(true)
   243→  const [err, setErr] = useState<string | null>(null)
   244→  const [barCount, setBarCount] = useState(mini ? 60 : 120)
   245→  const [dirty, setDirty] = useState(false) // false = view pinned to live edge
   246→  const offsetRef = useRef(0) // bars panned; NEGATIVE = into future space
   247→  const priceZoomRef = useRef(1)
   248→  const priceOffRef = useRef(0)
   249→  const [, forceRender] = useState(0)
   250→  const crossRef = useRef<{ x: number; y: number } | null>(null)
   251→  const syncCrossRef = useRef<{ slot: number; price: number | null } | null>(null)
   252→  const dragRef = useRef<DragState | null>(null)
   253→  const sizeRef = useRef({ w: 0, h: 0 })
   254→  const winRef = useRef({ w: 0, h: 0, plotW: 0, priceH: 0, axisH: mini ? 12 : 20, pMin: 0, pMax: 1 })
   255→  const lastRangeRef = useRef({ pMin: 0, pMax: 1 }) // survives full-future views
   256→  const lastVolRef = useRef<Record<string, number>>({})
   257→  const chartId = useRef(++CHART_SEQ)
   258→  // ---- 1M ghost overlay ----
   259→  const [ghost, setGhost] = useState<Candle[]>([])
   260→  // ---- scales baseline (null = day open ref) + pick mode ----
   261→  const baseRef = useRef<number | null>(null)
   262→  const pickRef = useRef(false)
   263→  const [baseArmed, setBaseArmed] = useState(false)
   264→  // last-drawn baseline geometry (for the grab zone) — updated every frame
   265→  const drawMetaRef = useRef({ yBase: -1, baseline: 0 })
   266→
   267→  // ---- baseline pick-mode bus (armed from the toolbar SCALES popover) ----
   268→  useEffect(() => {
   269→    const arm = () => { pickRef.current = true; setBaseArmed(true) }
   270→    const disarm = () => { pickRef.current = false; setBaseArmed(false) }
   271→    window.addEventListener(BASE_PICK_EVENT, arm)
   272→    window.addEventListener(BASE_SET_EVENT, disarm)
   273→    return () => {
   274→      window.removeEventListener(BASE_PICK_EVENT, arm)
   275→      window.removeEventListener(BASE_SET_EVENT, disarm)
   276→    }
   277→  }, [])
   278→
   279→  // ---- crosshair sync (same tf only — charts are interval-locked) ----
   280→  useEffect(() => {
   281→    const fn = (m: CrossSyncMsg) => {
   282→      if (m.src === chartId.current || m.tf !== tf) { return }
   283→      if (m.t == null) { syncCrossRef.current = null; return }
   284→      syncCrossRef.current = { slot: m.t, price: m.price } // slot stores t; resolved in draw
   285→      forceRender(n => n + 1)
   286→    }
   287→    crossBusListeners.add(fn)
   288→    return () => { crossBusListeners.delete(fn) }
   289→  }, [tf])
   290→
   291→  // ---- data loading ----
   292→  const load = useCallback(async () => {
   293→    if (!symbol) return
   294→    try {
   295→      const r = await api.candles(symbol, tf as never, 1000)
   296→      if (r.candles?.length) {
   297→        setCandles(r.candles)
   298→        lastVolRef.current = { [symbol]: r.candles[r.candles.length - 1]?.v ?? 0 }
   299→      }
   300→      setErr(null)
   301→    } catch (e) {
   302→      setErr(e instanceof Error ? e.message : 'load failed')
   303→    } finally {
   304→      setLoading(false)
   305→    }
   306→  }, [symbol, tf])
   307→
   308→  useEffect(() => {
   309→    setLoading(true)
   310→    offsetRef.current = 0
   311→    priceZoomRef.current = 1
   312→    priceOffRef.current = 0
   313→    setDirty(false)
   314→    load()
   315→    subscribeTicks([symbol])
   316→    const iv = setInterval(load, 30_000)
   317→    return () => {
   318→      unsubscribeTicks([symbol])
   319→      clearInterval(iv)
   320→    }
   321→  }, [symbol, tf, load])
   322→
   323→  // ---- 1m ghost data (only meaningful on higher timeframes) ----
   324→  useEffect(() => {
   325→    if (!showGhost || tf === '1m') { setGhost([]); return }
   326→    let dead = false
   327→    const pull = async () => {
   328→      try {
   329→        const r = await api.candles(symbol, '1m' as never, 1200)
   330→        if (!dead) setGhost(r.candles ?? [])
   331→      } catch { /* ghost is best-effort */ }
   332→    }
   333→    pull()
   334→    const iv = setInterval(pull, 30_000)
   335→    return () => { dead = true; clearInterval(iv) }
   336→  }, [symbol, tf, showGhost])
   337→
   338→  // ---- live ticks merge into the last candle ----
   339→  useEffect(() => {
   340→    return onTicks(map => {
   341→      const t: Tick | undefined = map[symbol]
   342→      if (!t) return
   343→      setCandles(prev => {
   344→        if (prev.length === 0) return prev
   345→        const arr = prev.slice()
   346→        const last = { ...arr[arr.length - 1] }
   347→        const bucket = Math.floor(t.ts / TF_MS[tf as keyof typeof TF_MS]) * TF_MS[tf as keyof typeof TF_MS]
   348→        const dayVolDelta = Math.max(0, t.vol - (lastVolRef.current[symbol] ?? t.vol))
   349→        lastVolRef.current[symbol] = t.vol
   350→        if (bucket > last.t) {
   351→          arr.push({ t: bucket, o: t.ltp, h: t.ltp, l: t.ltp, c: t.ltp, v: dayVolDelta })
   352→          if (arr.length > 1600) arr.splice(0, 200)
   353→        } else {
   354→          last.c = t.ltp
   355→          last.h = Math.max(last.h, t.ltp)
   356→          last.l = Math.min(last.l, t.ltp)
   357→          last.v += dayVolDelta
   358→          arr[arr.length - 1] = last
   359→        }
   360→        return arr
   361→      })
   362→      // ghost 1m last bar follows live ticks as well
   363→      if (showGhost && tf !== '1m') {
   364→        setGhost(prevG => {
   365→          if (prevG.length === 0) return prevG
   366→          const gArr = prevG.slice()
   367→          const gl = { ...gArr[gArr.length - 1] }
   368→          const gb = Math.floor(t.ts / 60_000) * 60_000
   369→          if (gb > gl.t) {
   370→            gArr.push({ t: gb, o: t.ltp, h: t.ltp, l: t.ltp, c: t.ltp, v: 0 })
   371→            if (gArr.length > 2400) gArr.splice(0, 200)
   372→          } else {
   373→            gl.c = t.ltp
   374→            gl.h = Math.max(gl.h, t.ltp)
   375→            gl.l = Math.min(gl.l, t.ltp)
   376→            gArr[gArr.length - 1] = gl
   377→          }
   378→          return gArr
   379→        })
   380→      }
   381→    })
   382→  }, [symbol, tf, showGhost])
   383→
   384→  // ---- render loop ----
   385→  useEffect(() => {
   386→    const canvas = canvasRef.current
   387→    const wrap = wrapRef.current
   388→    if (!canvas || !wrap) return
   389→    let raf = 0
   390→
   391→    const draw = () => {
   392→      const dpr = Math.min(2, window.devicePixelRatio || 1)
   393→      const w = wrap.clientWidth
   394→      const h = wrap.clientHeight
   395→      if (sizeRef.current.w !== w || sizeRef.current.h !== h || canvas.width !== Math.round(w * dpr)) {
   396→        canvas.width = Math.round(w * dpr)
   397→        canvas.height = Math.round(h * dpr)
   398→        canvas.style.width = `${w}px`
   399→        canvas.style.height = `${h}px`
   400→        sizeRef.current = { w, h }
   401→      }
   402→      const ctx = canvas.getContext('2d')
   403→      if (!ctx) return
   404→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   405→      ctx.clearRect(0, 0, w, h)
   406→      ctx.font = '10px ui-monospace, monospace'
   407→
   408→      const data = candles
   409→      if (data.length < 2) {
   410→        ctx.fillStyle = AXIS_TXT
   411→        ctx.textAlign = 'center'
   412→        ctx.fillText(loading ? 'loading candles…' : err || 'no data', w / 2, h / 2)
   413→        return
   414→      }
   415→
   416→      const axisW = mini ? 46 : 58
   417→      const axisH = mini ? 12 : 20
   418→      const plotW = w - axisW
   419→      const availH = h - axisH
   420→      const volH = showVolume ? Math.round(availH * (mini ? 0.14 : 0.16)) : 0
   421→      const subSlot = Math.round(availH * 0.16)
   422→      const rsiH = !mini && showRSI ? subSlot : 0
   423→      const macdH = !mini && showMACD ? subSlot : 0
   424→      const priceH = availH - (showVolume ? volH + 5 : 0) - rsiH - macdH
   425→
   426→      // visible window — offset may be NEGATIVE (future space)
   427→      const end = data.length - Math.round(offsetRef.current)
   428→      const start = Math.min(end - barCount, data.length - 2)
   429→      const slots = end - start
   430→      if (slots < 2) { offsetRef.current = 0; return }
   431→      const vis: (Candle | null)[] = new Array(slots)
   432→      for (let i = 0; i < slots; i++) vis[i] = data[start + i] ?? null
   433→      const lastC0 = data[data.length - 1]
   434→      const lastT = data[data.length - 1].t
   435→      const tfMs = TF_MS[tf as keyof typeof TF_MS] ?? 60_000
   436→      const timeOfSlot = (i: number) => {
   437→        const di = start + i
   438→        return di < data.length ? data[di].t : lastT + (di - (data.length - 1)) * tfMs
   439→      }
   440→
   441→      // day-change reference (session open) — the default scales baseline
   442→      const dayKey = (t: number) => new Date(t + 5.5 * 3600_000).toISOString().slice(0, 10)
   443→      const todayKey = dayKey(lastT)
   444→      let dayRef = lastC0.o
   445→      for (let i = data.length - 1; i >= 0; i--) {
   446→        if (dayKey(data[i].t) !== todayKey) break
   447→        dayRef = data[i].o
   448→      }
   449→      const baseline = baseRef.current ?? dayRef
   450→      const pctOf = (p: number) => (baseline ? ((p - baseline) / baseline) * 100 : 0)
   451→
   452→      // price range over real candles + overlays
   453→      let pMin = Infinity, pMax = -Infinity
   454→      for (const c of vis) { if (!c) continue; if (c.l < pMin) pMin = c.l; if (c.h > pMax) pMax = c.h }
   455→      // ghost 1m candles stretch the range too
   456→      if (showGhost && ghost.length > 0 && tfMs > 60_000) {
   457→        const tA = timeOfSlot(0), tB = timeOfSlot(slots - 1) + tfMs
   458→        const g0 = lowerBoundTime(ghost, tA), g1 = lowerBoundTime(ghost, tB)
   459→        for (let gi = g0; gi < g1; gi++) { const g = ghost[gi]; if (g.l < pMin) pMin = g.l; if (g.h > pMax) pMax = g.h }
   460→      }
   461→      const s20 = showSMA20 ? sma(data, 20) : null
   462→      const s50 = showSMA50 ? sma(data, 50) : null
   463→      const e9 = showEMA ? ema(data, 9) : null
   464→      const e21 = showEMA ? ema(data, 21) : null
   465→      const vw = showVWAP ? vwap(data) : null
   466→      const bb = showBOLL ? bollinger(data) : null

   876→          const i = Math.max(0, Math.min(data.length - 1, Math.round((idxTime - data[0].t) / tfMs)))
   877→          const v = rsiArr[i]
   878→          if (v != null) { ctx.fillStyle = v > 70 ? DOWN : v < 30 ? UP : '#f5a623'; ctx.fillText(`RSI ${v.toFixed(1)}`, x, 4) }
   879→        }
   880→      }
   881→      // focus candle: hovered > synced > last
   882→      let focus: { c: Candle; t: number } | null = { c: lastC, t: lastT }
   883→      const drawCrossAt = (slotIdx: number, own: boolean, syncPrice: number | null) => {
   884→        if (slotIdx < 0 || slotIdx >= slots) return
   885→        const x = xOf(slotIdx)
   886→        ctx.strokeStyle = CROSS
   887→        ctx.setLineDash([3, 3])
   888→        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, h - axisH); ctx.stroke()
   889→        if (own && crossRef.current) {
   890→          const y = Math.min(priceH, Math.max(0, crossRef.current.y))
   891→          ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(plotW, y); ctx.stroke()
   892→          const price = pMax - (crossRef.current.y / priceH) * (pMax - pMin)
   893→          ctx.fillStyle = '#2a3441'
   894→          ctx.fillRect(plotW + 2, y - 8, axisW - 4, 16)
   895→          ctx.fillStyle = '#e6e9ec'
   896→          ctx.textAlign = 'center'
   897→          ctx.textBaseline = 'middle'
   898→          ctx.fillText(price.toFixed(2), plotW + axisW / 2, y)
   899→        }
   900→        if (syncPrice != null && syncPrice >= pMin && syncPrice <= pMax) {
   901→          const sy = yOf(syncPrice)
   902→          ctx.strokeStyle = 'rgba(255,255,255,0.16)'
   903→          ctx.beginPath(); ctx.moveTo(0, sy); ctx.lineTo(plotW, sy); ctx.stroke()
   904→          ctx.fillStyle = '#2a3441'
   905→          ctx.fillRect(plotW + 2, sy - 8, axisW - 4, 16)
   906→          ctx.fillStyle = '#e6e9ec'
   907→          ctx.textAlign = 'center'
   908→          ctx.textBaseline = 'middle'
   909→          ctx.fillText(syncPrice.toFixed(2), plotW + axisW / 2, sy)
   910→        }
   911→        ctx.setLineDash([])
   912→        // time bubble
   913→        const t = timeOfSlot(slotIdx)
   914→        ctx.fillStyle = '#2a3441'
   915→        ctx.textAlign = 'center'
   916→        ctx.textBaseline = 'top'
   917→        ctx.fillRect(Math.min(plotW - 48, Math.max(0, x - 24)), h - axisH + 2, 48, 14)
   918→        ctx.fillStyle = '#e6e9ec'
   919→        ctx.fillText(fmtAxisTime(t, tf), Math.min(plotW - 24, Math.max(24, x)), h - axisH + 4)
   920→        const c = vis[slotIdx]
   921→        if (c) focus = { c, t }
   922→      }
   923→
   924→      // own crosshair
   925→      if (!mini && crossRef.current && crossRef.current.x < plotW && crossRef.current.y < h - axisH) {
   926→        const slotIdx = Math.min(slots - 1, Math.max(0, Math.floor(crossRef.current.x / cw)))
   927→        drawCrossAt(slotIdx, true, null)
   928→      }
   929→      // synced crosshair from another chart (same interval)
   930→      const sc = syncCrossRef.current
   931→      if (!mini && sc) {
   932→        // sc.slot carries the candle TIME (set by the sync bus)
   933→        const slot0 = timeOfSlot(0)
   934→        const slotIdx = Math.round((sc.slot - slot0) / tfMs)
   935→        drawCrossAt(slotIdx, false, sc.price)
   936→      }
   937→
   938→      // day change reference for the legend %
   939→      const fc = focus as { c: Candle; t: number } | null
   940→      if (fc) {
   941→        const pct = dayRef ? ((fc.c.c - dayRef) / dayRef) * 100 : null
   942→        drawLegend(fc.c, fc.t, pct)
   943→      }
   944→    }
   945→
   946→    const loop = () => { draw(); raf = requestAnimationFrame(loop) }
   947→    raf = requestAnimationFrame(loop)
   948→    const ro = new ResizeObserver(() => forceRender(n => n + 1))
   949→    ro.observe(wrap)
   950→    return () => { cancelAnimationFrame(raf); ro.disconnect() }
   951→  }, [candles, ghost, barCount, chartType, showVolume, showSMA20, showSMA50, showEMA, showVWAP, showBOLL, showRSI, showMACD, showGhost, ghostStyle, scales, markers, symbol, tf, loading, err, mini])
   952→
   953→  // ---- interactions ----
   954→  const onWheel = (e: React.WheelEvent) => {
   955→    e.preventDefault()
   956→    const rect = canvasRef.current!.getBoundingClientRect()
   957→    const x = e.clientX - rect.left
   958→    const win = winRef.current
   959→    const f = e.deltaY > 0 ? 1.15 : 0.87
   960→    if ((win.plotW > 0 && x >= win.plotW) || e.shiftKey) {
   961→      priceZoomRef.current = Math.max(0.15, Math.min(10, priceZoomRef.current * f))
   962→    } else {
   963→      setBarCount(b => Math.max(25, Math.min(900, Math.round(b * f))))
   964→    }
   965→    setDirty(true)
   966→  }
   967→  const onMouseDown = (e: React.MouseEvent) => {
   968→    const rect = canvasRef.current!.getBoundingClientRect()
   969→    const x = e.clientX - rect.left
   970→    const y = e.clientY - rect.top
   971→    const win = winRef.current
   972→    // baseline drag zone (scales % / pts enabled): grab within 5px of the line
   973→    const sc = scales
   974→    const baselineEnabled = sc && (sc.pct || sc.inv || sc.pts) && !mini
   975→    if (baselineEnabled && x < win.plotW && y <= win.priceH) {
   976→      const meta = drawMetaRef.current
   977→      if (meta.yBase >= 0 && Math.abs(y - meta.yBase) <= 5) {
   978→        baseRef.current = meta.baseline // freeze the (possibly default) baseline
   979→        dragRef.current = { zone: 'base', startX: e.clientX, startY: e.clientY, startOffset: 0, startPriceOff: 0, startPriceZoom: 1, pxPerPx: (win.pMax - win.pMin) / (win.priceH || 1), moved: 0 }
   980→        return
   981→      }
   982→    }
   983→    const zone: DragState['zone'] = x >= win.plotW ? 'price' : y >= win.h - win.axisH ? 'time' : 'plot'
   984→    dragRef.current = {
   985→      zone,
   986→      startX: e.clientX,
   987→      startY: e.clientY,
   988→      startOffset: offsetRef.current,
   989→      startPriceOff: priceOffRef.current,
   990→      startPriceZoom: priceZoomRef.current,
   991→      pxPerPx: win.priceH > 0 ? (win.pMax - win.pMin) / win.priceH : 1,
   992→      moved: 0,
   993→    }
   994→  }
   995→  const onMouseMove = (e: React.MouseEvent) => {
   996→    const rect = canvasRef.current!.getBoundingClientRect()
   997→    crossRef.current = { x: e.clientX - rect.left, y: e.clientY - rect.top }
   998→    const win = winRef.current
   999→    // publish to the sync bus (same-interval charts follow)
  1000→    if (win.plotW > 0 && candles.length > 1) {
  1001→      const tfMs = TF_MS[tf as keyof typeof TF_MS] ?? 60_000
  1002→      const end = candles.length - Math.round(offsetRef.current)
  1003→      const start = Math.min(end - barCount, candles.length - 2)
  1004→      const cw = win.plotW / (end - start)
  1005→      const slotIdx = Math.floor((e.clientX - rect.left) / cw)
  1006→      const di = start + slotIdx
  1007→      const lastT = candles[candles.length - 1].t
  1008→      const t = di < candles.length ? candles[di].t : lastT + (di - (candles.length - 1)) * tfMs
  1009→      const price = e.clientY - rect.top <= win.priceH
  1010→        ? win.pMax - ((e.clientY - rect.top) / win.priceH) * (win.pMax - win.pMin)
  1011→        : null
  1012→      emitCross({ src: chartId.current, tf, t, price })
  1013→    }
  1014→    const d = dragRef.current
  1015→    if (!d) return
  1016→    d.moved = Math.max(d.moved, Math.abs(e.clientX - d.startX) + Math.abs(e.clientY - d.startY))
  1017→    if (d.zone === 'base') {
  1018→      // drag the baseline — % and pts scales follow ("rearrange")
  1019→      const y = Math.min(win.priceH, Math.max(0, e.clientY - rect.top))
  1020→      baseRef.current = Number((win.pMax - (y / (win.priceH || 1)) * (win.pMax - win.pMin)).toFixed(2))
  1021→      return
  1022→    }
  1023→    if (d.zone === 'price') {
  1024→      const f = Math.exp((d.startY - e.clientY) / 180)
  1025→      priceZoomRef.current = Math.max(0.15, Math.min(10, d.startPriceZoom * f))
  1026→      return
  1027→    }
  1028→    // horizontal pan — negative offset = future space
  1029→    const cwAvg = Math.max(0.5, (sizeRef.current.w - (mini ? 46 : 58)) / barCount)
  1030→    const barsMoved = Math.round((e.clientX - d.startX) / cwAvg)
  1031→    const maxOff = candles.length - 25
  1032→    offsetRef.current = Math.max(-FUTURE_BARS, Math.min(maxOff, d.startOffset + barsMoved))
  1033→    if (d.zone === 'plot') {
  1034→      priceOffRef.current = d.startPriceOff + (e.clientY - d.startY) * d.pxPerPx
  1035→    }
  1036→    setDirty(true)
  1037→  }
  1038→  const onMouseUp = (e: React.MouseEvent) => {
  1039→    const d = dragRef.current
  1040→    dragRef.current = null
  1041→    if (!d || d.moved >= 4) return
  1042→    const rect = canvasRef.current!.getBoundingClientRect()
  1043→    const x = e.clientX - rect.left
  1044→    const y = e.clientY - rect.top
  1045→    const win = winRef.current
  1046→    if (x < win.plotW && y < win.priceH && win.pMax > win.pMin) {
  1047→      const price = Number((win.pMax - (y / win.priceH) * (win.pMax - win.pMin)).toFixed(2))
  1048→      // scales baseline pick mode — consumed by the first chart clicked
  1049→      if (pickRef.current) {
  1050→        baseRef.current = price
  1051→        pickRef.current = false
  1052→        setBaseArmed(false)
  1053→        window.dispatchEvent(new Event(BASE_SET_EVENT))
  1054→        return
  1055→      }
  1056→      onPriceClick?.(price)
  1057→    }
  1058→  }
  1059→  const onMouseLeave = () => {
  1060→    crossRef.current = null
  1061→    dragRef.current = null
  1062→    emitCross({ src: chartId.current, tf, t: null, price: null })
  1063→  }
  1064→  const resetView = () => {
  1065→    offsetRef.current = 0
  1066→    priceZoomRef.current = 1
  1067→    priceOffRef.current = 0
  1068→    baseRef.current = null // baseline returns to the day-open ref
  1069→    setBarCount(mini ? 60 : 120)
  1070→    setDirty(false)
  1071→  }
  1072→
  1073→  return (
  1074→    <div ref={wrapRef} className={`relative w-full ${height} select-none`} data-chart-root>
  1075→      <canvas
  1076→        ref={canvasRef}
  1077→        className="absolute inset-0 cursor-crosshair"
  1078→        onWheel={onWheel}
  1079→        onMouseDown={onMouseDown}
  1080→        onMouseMove={onMouseMove}
  1081→        onMouseUp={onMouseUp}
  1082→        onMouseLeave={onMouseLeave}
  1083→        onDoubleClick={resetView}
  1084→      />
  1085→      <div className="absolute right-16 top-1.5 flex gap-2 text-[9px] text-muted-foreground pointer-events-none">
  1086→        {showSMA20 && <span className="text-[#4dd0e1]">SMA20</span>}
  1087→        {showSMA50 && <span className="text-[#b48ce8]">SMA50</span>}
  1088→        {showEMA && <span className="text-[#ffd166]">EMA9/21</span>}
  1089→        {showVWAP && <span className="text-[#4dd0e1]">VWAP</span>}
  1090→        {showBOLL && <span className="text-muted-foreground">BB</span>}
  1091→        {showGhost && tf !== '1m' && <span className="text-[#e0a63f]">1M GHOST</span>}
  1092→      </div>
  1093→      {baseArmed && (
  1094→        <div className="absolute left-1/2 top-1.5 -translate-x-1/2 px-2 py-0.5 rounded bg-[#e0a63f] text-black text-[9px] font-bold pointer-events-none">
  1095→          click the chart to set the scales baseline
  1096→        </div>
  1097→      )}
  1098→      {!dirty && !mini && (
  1099→        <div className="absolute bottom-6 right-14 text-[9px] text-muted-foreground/60 pointer-events-none flex items-center gap-1">
  1100→          <span className="h-1.5 w-1.5 rounded-full bg-[#00c896] pulse-dot" /> LIVE
  1101→        </div>
  1102→      )}
  1103→      {dirty && (
  1104→        <button
  1105→          onClick={resetView}
  1106→          title="Reset chart to live price action (latest bar)"
  1107→          className="absolute bottom-6 right-14 px-1.5 py-0.5 rounded bg-primary/80 text-primary-foreground text-[9px] font-bold hover:bg-primary shadow flex items-center gap-1"
  1108→        >
  1109→          <span className="text-[8px]">▶</span> LIVE
  1110→        </button>
  1111→      )}
  1112→    </div>
  1113→  )
