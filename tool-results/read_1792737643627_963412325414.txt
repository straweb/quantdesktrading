    31→function fmtCompact(n: number | null | undefined): string {
    32→  if (n == null || isNaN(n)) return '—'
    33→  const abs = Math.abs(n)
    34→  if (abs >= 1e7) return `₹${(n / 1e7).toFixed(2)}Cr`
    35→  if (abs >= 1e5) return `₹${(n / 1e5).toFixed(2)}L`
    36→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}k`
    37→  return `₹${Math.round(n)}`
    38→}

    40→/** Slider max for the fixed-₹ capital editor — ~125% of the account's
    41→ *  available cash (or ₹10L when cash is unknown), rounded to a clean step. */
    42→function fixedSliderMax(cash: number | null | undefined): number {
    43→  const base = Math.max(100_000, Math.ceil(((cash ?? 0) * 1.25) / 50_000) * 50_000)
    44→  return Math.max(base, 1_000_000)
    45→}

    47→/** Round a rupee slider position to a clean step value. */
    48→function snapStep(v: number, step = 5_000): number {
    49→  return Math.round(v / step) * step
    50→}

   187→function CapitalChip({ acc }: { acc: AccountRow }) {
   188→  const qc = useQueryClient()
   189→  const [open, setOpen] = useState(false)
   190→  const [capital, setCapital] = useState(String(acc.initialCapitalFixed ?? acc.initialCapital))
   191→  const [pct, setPct] = useState(String(acc.capitalPct ?? 100))
   192→  const [base, setBase] = useState<'FIXED' | 'AVAILABLE_CASH_PCT'>(acc.capitalBase ?? 'FIXED')
   193→  const [mode, setMode] = useState<CapitalMode>(acc.mode ?? 'INITIAL_PLUS_PNL')
   194→  const [busy, setBusy] = useState(false)
   195→
   196→  // (re)seed the editor fields each time the popover opens — avoids the
   197→  // setState-in-effect cascade lint rule
   198→  const openEditor = (next: boolean) => {
   199→    setOpen(next)
   200→    if (next) {
   201→      setCapital(String(acc.initialCapitalFixed ?? acc.initialCapital))
   202→      setPct(String(acc.capitalPct ?? 100))
   203→      setBase(acc.capitalBase ?? 'FIXED')
   204→      setMode(acc.mode ?? 'INITIAL_PLUS_PNL')
   205→    }
   206→  }
   207→
   208→  const toggleEnabled = async (next: boolean) => {
   209→    try {
   210→      await api.patchAccount(acc.id, { isOrderEnabled: next })
   211→      await qc.invalidateQueries({ queryKey: ['accounts'] })
   212→      toast.success(`${acc.label} order routing ${next ? 'enabled' : 'disabled'}`)
   213→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Update failed') }
   214→  }
   215→
   216→  const cash = acc.availableCash ?? null
   217→  const pctPreview = base === 'AVAILABLE_CASH_PCT' && cash != null ? Math.max(0, cash * ((Number(pct) || 0) / 100)) : null
   218→
   219→  const save = async () => {
   220→    setBusy(true)
   221→    try {
   222→      const body: Record<string, unknown> = { mode, capitalBase: base }
   223→      if (base === 'AVAILABLE_CASH_PCT') {
   224→        body.capitalPct = Math.min(100, Math.max(0, Number(pct) || 0))
   225→      } else {
   226→        body.initialCapital = Number(capital) || 0
   227→      }
   228→      await api.patchAccount(acc.id, body)
   229→      await qc.invalidateQueries({ queryKey: ['accounts'] })
   230→      await qc.invalidateQueries({ queryKey: ['funds'] })
   231→      const desc = base === 'AVAILABLE_CASH_PCT'
   232→        ? `${body.capitalPct}% of available cash${cash != null ? ` (₹${fmtCompact(pctPreview ?? 0)})` : ''}`
   233→        : fmtCompact(Number(capital) || 0)
   234→      toast.success(`${acc.label} capital updated — ${desc} · ${MODE_SHORT[mode]}`)
   235→      setOpen(false)
   236→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Update failed') }
   237→    setBusy(false)
   238→  }
   239→
   240→  return (
   241→    <Popover open={open} onOpenChange={openEditor}>
   242→      <PopoverTrigger asChild>
   243→        <button
   244→          className={`flex items-center gap-1 h-6 pl-1 pr-1.5 rounded-full border transition-colors ${
   245→            acc.isOrderEnabled
   246→              ? 'border-[#00c896]/40 bg-up text-foreground hover:bg-up/70'
   247→              : 'border-border bg-secondary/50 text-muted-foreground hover:text-foreground'
   248→          }`}
   249→          title={`${acc.label} (${acc.brokerType}) — usable ${fmtCompact(acc.usableCapital)} · ${acc.capitalBase === 'AVAILABLE_CASH_PCT' ? `${acc.capitalPct}% of available cash` : 'fixed ₹ capital'} · click to manage capital sizing`}
   250→        >
   251→          <span className={`h-1.5 w-1.5 rounded-full shrink-0 ${acc.isOrderEnabled ? 'bg-[#00c896]' : 'bg-muted-foreground/50'}`} />
   252→          <span className={`h-4 w-4 rounded text-[8px] font-bold grid place-items-center shrink-0 ${brokerIcon(acc.brokerType).cls}`}>
   253→            {brokerIcon(acc.brokerType).glyph}
   254→          </span>
   255→          <span className="tabular font-bold text-[9px]">{fmtCompact(acc.usableCapital)}</span>
   256→          {acc.capitalBase === 'AVAILABLE_CASH_PCT' && (
   257→            <span className="text-[8px] font-bold text-[#e0a63f] tabular">{acc.capitalPct}%</span>
   258→          )}
   259→        </button>
   260→      </PopoverTrigger>
   261→      <PopoverContent align="end" className="w-64 p-3 space-y-2.5 text-[11px]">
   262→        <div className="flex items-start justify-between gap-2">
   263→          <div className="min-w-0">
   264→            <div className="font-bold text-xs truncate">{acc.label}</div>
   265→            <div className="text-[9px] text-muted-foreground uppercase tracking-wide">{acc.brokerType}</div>
   266→          </div>
   267→          <div className="text-right shrink-0">
   268→            <div className="text-[9px] text-muted-foreground">usable</div>
   269→            <div className="tabular font-bold text-[#00c896]">{fmtCompact(acc.usableCapital)}</div>
   270→          </div>
   271→        </div>
   272→        <div className="flex items-center justify-between">
   273→          <span className="text-muted-foreground">Order routing</span>

   500→// ------------------------------------------------------------
   501→// Feed status chip — click to see / switch the market-data
   502→// source. Radio semantics: exactly ONE feed source is active
   503→// (Sim built-in or one broker account's feed adapter).
   504→// Round-8: each account row carries ICON BUTTONS for
   505→// connect / disconnect (account active state) and live feed
   506→// enable / disable (set as the active source or fall back to
   507→// sim), plus the account's live PnL and capital.
   508→// Round-10: per-row LIVE-FEED health (ticking / last tick age /
   509→// last error from the feed service), a capital quick editor
   510→// (% of cash / fixed ₹, editable anytime — even while connected)
   511→// and a one-click "pull real funds" action.
   512→// ------------------------------------------------------------
   513→function FeedChip({ up }: { up: boolean }) {
   514→  const qc = useQueryClient()
   515→  const [open, setOpen] = useState(false)
   516→  const [busy, setBusy] = useState(false)
   517→  const [capEdit, setCapEdit] = useState<string | null>(null)
   518→  const [capBase, setCapBase] = useState<'FIXED' | 'AVAILABLE_CASH_PCT'>('AVAILABLE_CASH_PCT')
   519→  const [capVal, setCapVal] = useState('25')
   520→  // admin plane over the websocket (2s push) — replaces the feed-source +
   521→  // feed-status REST polls; REST stays as the fallback when the socket is down
   522→  const adm = useSyncChannel<SyncAdmin>('sync:admin')
   523→  const { data } = useQuery({ queryKey: ['feed-source'], queryFn: () => api.feedSource(), enabled: !adm, refetchInterval: 8000 })
   524→  // live broker feed engine health (ticking? last error?)
   525→  const { data: feedSt } = useQuery({ queryKey: ['feed-status'], queryFn: () => api.feedStatus(), enabled: !adm, refetchInterval: 5000, retry: false })
   526→  const feedStatus = adm?.feed ?? feedSt
   527→  // realtime PnL / capital / connection state straight from sync:core
   528→  const sync = useSyncChannel<SyncCore>('sync:core')
   529→  const { data: restAccs } = useQuery({
   530→    queryKey: ['accounts'],
   531→    queryFn: () => api.accounts(),
   532→    enabled: !sync,
   533→    refetchInterval: 5000,
   534→  })
   535→  const accounts: AccountRow[] = sync?.accounts
   536→    ? (sync.accounts as unknown as AccountRow[])
   537→    : (restAccs?.accounts ?? [])
   538→  const source = adm?.feedSource.source ?? data?.source ?? 'sim'
   539→  const simDisabled = adm?.feedSource.simDisabled ?? data?.simDisabled ?? false
   540→  const sources = adm?.feedSource.sources ?? data?.sources ?? []
   541→  const bumpDataEpoch = useTerminal(s => s.bumpDataEpoch)
   542→  const current = sources.find(s => s.id === source)
   543→  const accById = new Map(accounts.map(a => [a.id, a]))
   544→  const tickAge = feedStatus?.lastTickTs ? Math.max(0, Math.round((Date.now() - feedStatus.lastTickTs) / 1000)) : null
   545→
   546→  const refresh = async () => {
   547→    await Promise.all([
   548→      qc.invalidateQueries({ queryKey: ['feed-source'] }),
   549→      qc.invalidateQueries({ queryKey: ['accounts'] }),
   550→      qc.invalidateQueries({ queryKey: ['feed-status'] }),
   551→    ])
   552→  }
   553→
   554→  const pick = async (id: string) => {
   555→    if (id === source || busy) return
   556→    // Live-Only mode: sandbox accounts (no broker session) can never be the
   557→    // data source — block at the UI level (the API refuses them too)
   558→    if (simDisabled && id !== 'sim') {
   559→      const row = sources.find(s => s.id === id)
   560→      const sessioned = accById.get(id)?.hasSession ?? row?.hasSession ?? false
   561→      if (!sessioned) {
   562→        toast.error(`${row?.label ?? 'This account'} has no broker session (sandbox) — Live-Only mode requires a live account. Create the session in Admin → Session.`)
   563→        return
   564→      }
   565→    }
   566→    setBusy(true)
   567→    try {
   568→      await api.setFeedSource(id)
   569→      await refresh()
   570→      // purge every client-side market-data holder (candles, ghost, ticks,
   571→      // depth books) — sim data must never linger under a real-broker source
   572→      bumpDataEpoch()
   573→      const label = sources.find(s => s.id === id)?.label ?? id
   574→      toast.success(`Market data source → ${label}${id === 'sim' ? '' : ' (sim data cleared)'}`)
   575→    } catch (e) {
   576→      toast.error(e instanceof Error ? e.message : 'Switch failed')
   577→    } finally { setBusy(false) }
   578→  }
   579→
   580→  const toggleAccount = async (id: string, next: boolean) => {
   581→    if (busy) return
   582→    setBusy(true)
   583→    try {
   584→      await api.patchAccount(id, { isActive: next })
   585→      await refresh()
   586→      const a = accById.get(id)
   587→      toast.success(`${a?.label ?? 'Account'} ${next ? 'connected' : 'disconnected'}${!next && source === id ? ' — feed fell back to Sim' : ''}`)
   588→    } catch (e) {
   589→      toast.error(e instanceof Error ? e.message : 'Update failed')
   590→    } finally { setBusy(false) }
   591→  }
   592→
   593→  const toggleFeed = async (id: string) => {
   594→    // if this account already feeds the terminal -> disable (back to sim);
   595→    // otherwise enable it as the active source
   596→    await pick(source === id ? 'sim' : id)
   597→  }
   598→
   599→  const pullFunds = async (id: string) => {
   600→    setBusy(true)
   601→    try {
   602→      const r = await api.syncFunds(id)
   603→      if (r.ok) toast.success(`Funds pulled — ${r.message ?? fmtCompact(r.cash ?? 0)}`)
   604→      else toast.error(r.message ?? 'Funds pull failed — check Admin → Broker Diagnostics')
   605→      await refresh()
   606→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Funds pull failed') }
   607→    setBusy(false)
   608→  }
   609→
   610→  const saveCap = async (id: string) => {
   611→    setBusy(true)
   612→    try {
   613→      const body: Record<string, unknown> = { capitalBase: capBase }
   614→      if (capBase === 'AVAILABLE_CASH_PCT') body.capitalPct = Math.min(100, Math.max(1, Number(capVal) || 25))
   615→      else body.initialCapital = Number(capVal) || 0
   616→      await api.patchAccount(id, body)
   617→      await refresh()
   618→      toast.success(`Capital updated — ${capBase === 'AVAILABLE_CASH_PCT' ? `${body.capitalPct}% of available cash` : fmtCompact(Number(capVal) || 0)}`)
   619→      setCapEdit(null)
   620→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Update failed') }
   621→    setBusy(false)
   622→  }
   623→
   624→  const openCapEdit = (acc: AccountRow) => {
   625→    setCapEdit(acc.id)
   626→    setCapBase(acc.capitalBase ?? 'AVAILABLE_CASH_PCT')
   627→    setCapVal(String(acc.capitalBase === 'AVAILABLE_CASH_PCT' ? (acc.capitalPct ?? 100) : (acc.initialCapitalFixed ?? acc.initialCapital)))
   628→  }
   629→
   630→  // live-feed health for the ACTIVE broker source
   631→  const liveHealthy = !!feedStatus?.sourceIsLive && !!feedStatus?.fresh
   632→  const liveErr = feedStatus?.sourceIsLive ? feedStatus?.lastError : null
   633→
   634→  return (
   635→    <Popover open={open} onOpenChange={setOpen}>
   636→      <PopoverTrigger asChild>
   637→        <button
   638→          className={`flex items-center gap-1 px-1.5 py-1 rounded border transition-colors ${
   639→            up ? 'border-[#00c896]/30 bg-up/40 text-[#00c896] hover:bg-up/60' : 'border-[#ff4d5e]/30 bg-down/40 text-[#ff4d5e] hover:bg-down/60'
   640→          }`}
   641→          title={`Market data source — click to view / switch${source !== 'sim' ? ` · live feed ${liveHealthy ? 'ticking' : `STALLED${liveErr ? `: ${liveErr}` : ''}`}` : ''}`}
   642→        >
   643→          <CircleDot className={`h-3 w-3 ${up ? 'pulse-dot' : ''}`} />
   644→          <span className="font-bold">FEED {up ? 'LIVE' : 'DOWN'}</span>
   645→          {source !== 'sim' && (
   646→            <span
   647→              className={`h-1.5 w-1.5 rounded-full ${liveHealthy ? 'bg-[#00c896] pulse-dot' : 'bg-[#e0a63f]'}`}
   648→              title={liveHealthy ? `live broker data ticking (${feedStatus?.symbols} symbols · last ${tickAge}s ago)` : `live data stalled${liveErr ? ` — ${liveErr}` : ' — waiting for ticks'}`}
   649→            />
   650→          )}
   651→          <span className="max-w-[110px] truncate text-foreground/80">{current?.label ?? 'Sim Feed'}</span>
   652→          {simDisabled && <span className="px-1 rounded bg-[#e0a63f]/20 text-[#e0a63f] text-[8px] font-bold" title="Live-Only mode (admin) — sim feed disabled">LIVE-ONLY</span>}
   653→        </button>
   654→      </PopoverTrigger>
   655→      <PopoverContent align="end" className="w-[360px] p-2 text-[11px] space-y-1">
   656→        <div className="px-1 pb-1 text-[9px] text-muted-foreground leading-snug">
   657→          Market data source — one feed active at a time. Live rows show real feed health; click <b>cap</b> to change % / fixed capital anytime, even while connected.
   658→        </div>
   659→        {feedStatus?.sourceIsLive && (
   660→          <div className={`mx-1 rounded px-2 py-1.5 text-[9px] leading-snug border ${
   661→            liveHealthy ? 'border-[#00c896]/30 bg-up/30 text-[#00c896]' : 'border-[#e0a63f]/40 bg-[#e0a63f]/10 text-[#e0a63f]'
   662→          }`}>
   663→            {liveHealthy
   664→              ? <>Live broker feed ticking — {feedStatus.symbols} symbols · last tick {tickAge}s ago · {feedStatus.polls} polls</>
   665→              : <>Live feed NOT flowing{feedStatus.lastError ? ` — ${feedStatus.lastError}` : ''}{!feedStatus.sessionPresent ? ' · no session — create it in Admin → Session' : ''} · details in Admin → Broker Diagnostics</>}
   666→          </div>
   667→        )}
   668→        {sources.map(s => {
   669→          const acc = s.id === 'sim' ? null : accById.get(s.id)
   670→          const isSrc = s.id === source
   671→          return (
   672→            <div
   673→              key={s.id}
   674→              role="button"
   675→              tabIndex={0}
   676→              onClick={() => pick(s.id)}
   677→              onKeyDown={e => { if (e.key === 'Enter') pick(s.id) }}
   678→              className={`w-full flex items-center gap-2 px-2 py-1.5 rounded text-left transition-colors cursor-pointer ${
   679→                isSrc ? 'bg-primary/20 border border-primary/50' : 'hover:bg-secondary border border-transparent'
   680→              } ${acc && !acc.isActive ? 'opacity-60' : ''}`}
   681→            >
   682→              <span className={`h-2.5 w-2.5 rounded-full border shrink-0 flex items-center justify-center ${
   683→                isSrc ? 'border-primary' : 'border-muted-foreground/40'
   684→              }`}>
   685→                {isSrc && <span className="h-1.5 w-1.5 rounded-full bg-primary" />}
   686→              </span>
   687→              <span className={`h-4 w-4 rounded text-[8px] font-bold grid place-items-center shrink-0 ${s.id === 'sim' ? 'bg-slate-500/25 text-slate-300' : brokerIcon(s.brokerType ?? '').cls}`}>
   688→                {s.id === 'sim' ? '⚡' : brokerIcon(s.brokerType ?? '').glyph}

   694→                </span>
   695→                {acc ? (

   720→                      title="Edit capital — % of available cash or fixed ₹ (works while connected)"

   749→                        ? 'border-[#00c896]/40 text-[#00c896] hover:bg-[#00c896]/20'
   750→                        : 'border-[#ff4d5e]/40 text-[#ff4d5e] hover:bg-[#ff4d5e]/20'

   759→                    title={
   760→                      isSrc ? 'Disable this account feed (back to Sim)'
   761→                      : simDisabled && !(acc.hasSession ?? sources.find(x => x.id === acc.id)?.hasSession)

   790→                <button
   791→                  onClick={() => { setCapBase('AVAILABLE_CASH_PCT'); setCapVal(String(acc.capitalPct ?? 100)) }}
   792→                  className={`px-2 py-1 rounded text-[10px] font-semibold border ${capBase === 'AVAILABLE_CASH_PCT' ? 'bg-primary/20 border-primary/50' : 'bg-secondary/50 border-border text-muted-foreground'}`}
   793→                >% of cash</button>
   794→                <button
   795→                  onClick={() => { setCapBase('FIXED'); setCapVal(String((acc.initialCapitalFixed ?? acc.initialCapital) || 500000)) }}
   796→                  className={`px-2 py-1 rounded text-[10px] font-semibold border ${capBase === 'FIXED' ? 'bg-primary/20 border-primary/50' : 'bg-secondary/50 border-border text-muted-foreground'}`}

   800→                <div className="space-y-1">

   805→                      max={100}
   806→                      step={1}
   807→                      onValueChange={v => setCapVal(String(v[0]))}
   808→                      className="flex-1 [&_[data-slot=slider-thumb]]:h-3 [&_[data-slot=slider-thumb]]:w-3"

   811→                    <Input
   812→                      value={capVal}

   815→                      placeholder="25"
