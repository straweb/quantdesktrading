    21→function fmtCompact(n: number | null | undefined): string {
    22→  if (n == null || isNaN(n)) return '—'
    23→  const abs = Math.abs(n)
    24→  if (abs >= 1e7) return `₹${(n / 1e7).toFixed(2)}Cr`
    25→  if (abs >= 1e5) return `₹${(n / 1e5).toFixed(2)}L`
    26→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}k`
    27→  return `₹${Math.round(n)}`
    28→}

    38→  const totalUsable = accounts.reduce((x, a) => x + a.usableCapital, 0)
    39→  const openOrders = (core?.orders ?? []).filter(o => o.status === 'OPEN' || o.status === 'PENDING').length

    41→  const fillRate = totalOrders > 0 ? (totalOrders - openOrders) / totalOrders * 100 : null
    42→
    43→  // realized P&L curve (cumulative over the session's trades, oldest first)

    45→    const trades = [...(core?.trades ?? [])].sort((a, b) => new Date(a.ts).getTime() - new Date(b.ts).getTime())

    47→    const pts: { t: string; v: number }[] = []
    48→    for (const t of trades) {
    49→      if (t.realizedPnl == null) continue
    50→      cum += t.realizedPnl
    51→      pts.push({ t: t.ts, v: cum })
    52→    }
    53→    return pts
    54→  }, [core])
    55→

    58→      <div className="p-3 space-y-3 max-w-[1500px] mx-auto">

   153→// ---------------- MASTER dashboard — system health strip ----------------
   154→// Live admin overview from the sync:admin websocket: feed engine state,
   155→// per-broker session presence, symbol universe, copy-trade safety switches.
   156→function SystemHealth({ adm }: { adm: SyncAdmin | null }) {
   157→  const feed = adm?.feed
   158→  const meta = adm?.accountsMeta ?? []
   159→  const tickAge = feed?.lastTickTs != null ? Math.max(0, Math.round((Date.now() - feed.lastTickTs) / 1000)) : null
   160→  const liveErr = feed?.sourceIsLive ? feed.lastError : null
   161→  return (
   162→    <Card title="Master — system health">
   163→      <div className="grid grid-cols-2 md:grid-cols-4 gap-2">
   164→        <HealthCell
   165→          label="Feed engine"
   166→          main={feed == null ? '—' : feed.sourceIsLive ? (feed.fresh ? 'TICKING' : 'STALLED') : 'SIM FEED'}
   167→          tone={feed == null ? 'neutral' : feed.sourceIsLive && !feed.fresh ? 'warn' : 'ok'}
   168→          sub={feed == null ? 'websocket connecting…' : feed.sourceIsLive
   169→            ? `${feed.symbols} symbols · last tick ${tickAge != null ? `${tickAge}s` : '—'} · ${feed.polls} polls${liveErr ? ` · ${liveErr}` : ''}`
   170→            : `built-in sim · ${feed.symbols ?? 0} symbols`}
   171→        />
   172→        <HealthCell
   173→          label="Broker sessions"
   174→          main={meta.length === 0 ? '—' : `${meta.filter(m => m.hasCreds).length}/${meta.length} live`}
   175→          tone={meta.length === 0 ? 'neutral' : meta.filter(m => m.brokerType !== 'SIMFEED' && m.isActive && !m.hasCreds).length > 0 ? 'warn' : 'ok'}
   176→          sub={meta.map(m => `${m.label}${m.hasCreds ? '' : ' (no session)'}`).join(' · ') || 'no live broker accounts'}
   177→        />
   178→        <HealthCell
   179→          label="Symbol universe"
   180→          main={adm ? `${adm.syncInfo.counts.symbols}` : '—'}
   181→          tone="neutral"
   182→          sub={adm ? `${adm.syncInfo.counts.mappings} broker mappings · synced ${adm.syncInfo.lastSync ? fmtTime(adm.syncInfo.lastSync) : 'never'}` : ''}
   183→        />
   184→        <HealthCell
   185→          label="Copy trade + recorder"
   186→          main={adm ? `${adm.copyTrade?.enabled ? 'COPY ON' : 'COPY OFF'}${adm.copyTrade?.killSwitch ? ' · KILL' : ''} · REC ${adm.tickRecorder?.enabled ? 'ON' : 'OFF'}` : '—'}

   358→/** Tiny SVG line chart — cumulative realized P&L. */
   359→function PnlCurve({ pts }: { pts: { t: string; v: number }[] }) {
   360→  if (pts.length < 2) {
   361→    return <div className="h-24 flex items-center justify-center text-[10px] text-muted-foreground">no closed trades yet this session</div>
   362→  }
   363→  const W = 480, H = 96, P = 4
   364→  const vs = pts.map(p => p.v)
   365→  const min = Math.min(0, ...vs), max = Math.max(0, ...vs)
   366→  const span = max - min || 1
   367→  const x = (i: number) => P + (i / (pts.length - 1)) * (W - 2 * P)
   368→  const y = (v: number) => H - P - ((v - min) / span) * (H - 2 * P)
   369→  const d = pts.map((p, i) => `${i === 0 ? 'M' : 'L'}${x(i).toFixed(1)},${y(p.v).toFixed(1)}`).join(' ')
   370→  const last = vs[vs.length - 1]
   371→  const col = last >= 0 ? '#00c896' : '#ff4d5e'
   372→  return (
   373→    <div>
   374→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-24">
   375→        <line x1={P} x2={W - P} y1={y(0)} y2={y(0)} stroke="rgba(255,255,255,0.15)" strokeDasharray="3,3" />
   376→        <path d={d} fill="none" stroke={col} strokeWidth="1.6" />
   377→      </svg>
   378→      <div className={`text-[9px] tabular ${last >= 0 ? 'term-up' : 'term-down'}`}>
   379→        cumulative {fmtInr(last)} over {pts.length} closes · last {fmtTime(pts[pts.length - 1].t)}
   380→      </div>
   381→    </div>
   382→  )
   383→}

   513→function GttManager() {
   514→  const qc = useQueryClient()
   515→  // GTT rules ride the sync:admin websocket (2s push) — REST fallback
   516→  const adm = useSyncChannel<SyncAdmin>('sync:admin')
   517→  const { data: rest } = useQuery({ queryKey: ['gtt'], queryFn: () => api.gttRules(), enabled: !adm, refetchInterval: 3000 })
   518→  const rules = (adm?.gtt as unknown as GttRuleRow[] | undefined) ?? rest?.rules ?? []
   519→  const [symbol, setSymbol] = useState('NIFTY')
   520→  const [side, setSide] = useState<'BUY' | 'SELL'>('BUY')
   521→  const [qty, setQty] = useState('1')
   522→  const [trigger, setTrigger] = useState<'ABOVE' | 'BELOW'>('ABOVE')
   523→  const [triggerPrice, setTriggerPrice] = useState('')
   524→  const [orderType, setOrderType] = useState<'LIMIT' | 'MARKET'>('LIMIT')
   525→  const [busy, setBusy] = useState(false)
   526→
   527→  const create = async () => {
   528→    if (!symbol || !Number(triggerPrice) || !Number(qty)) { toast.error('Fill symbol, qty and trigger price'); return }
   529→    setBusy(true)
   530→    try {
   531→      await api.createGtt({ symbol, side, qty: Number(qty), trigger, triggerPrice: Number(triggerPrice), orderType, productType: 'MIS', repeatDaily: true })
   532→      await qc.invalidateQueries({ queryKey: ['gtt'] })
   533→      toast.success(`GTT armed — ${side} ${symbol} ${trigger.toLowerCase()} ${triggerPrice}`)
   534→      setTriggerPrice('')
   535→    } catch (e) { toast.error(e instanceof Error ? e.message : 'GTT create failed') }
   536→    setBusy(false)
   537→  }
   538→  const toggle = async (r: GttRuleRow, armed: boolean) => {
   539→    await api.patchGtt(r.id, { armed }).catch(e => toast.error(e instanceof Error ? e.message : 'failed'))
   540→    void qc.invalidateQueries({ queryKey: ['gtt'] })
   541→  }
   542→  const remove = async (id: string) => {
   543→    await api.deleteGtt(id).catch(e => toast.error(e instanceof Error ? e.message : 'failed'))
   544→    void qc.invalidateQueries({ queryKey: ['gtt'] })
   545→  }
   546→
   547→  return (
   548→    <Card title="GTT — good-till-triggered orders (all brokers)">
   549→      <div className="flex flex-wrap items-center gap-1.5 text-[10px] mb-2">
   550→        <Input value={symbol} onChange={e => setSymbol(e.target.value.toUpperCase())} placeholder="symbol" className="h-6 w-32 text-[10px] tabular bg-secondary/60" />
   551→        <button onClick={() => setSide(side === 'BUY' ? 'SELL' : 'BUY')} className={`px-2 py-0.5 rounded text-[9px] font-bold ${side === 'BUY' ? 'bg-[#00c896]/20 text-[#00c896]' : 'bg-[#ff4d5e]/20 text-[#ff4d5e]'}`}>{side}</button>
   552→        <Input value={qty} onChange={e => setQty(e.target.value.replace(/[^0-9]/g, ''))} placeholder="qty" className="h-6 w-14 text-[10px] tabular bg-secondary/60" />
   553→        <button
   554→          onClick={() => setTrigger(trigger === 'ABOVE' ? 'BELOW' : 'ABOVE')}
   555→          className="px-2 py-0.5 rounded text-[9px] font-bold bg-secondary/70 text-muted-foreground hover:text-foreground"
   556→          title="Trigger when LTP crosses above / below the trigger price"
   557→        >
   558→          {trigger === 'ABOVE' ? '≥' : '≤'}
   559→        </button>
   560→        <Input value={triggerPrice} onChange={e => setTriggerPrice(e.target.value.replace(/[^0-9.]/g, ''))} placeholder="trigger ₹" className="h-6 w-24 text-[10px] tabular bg-secondary/60" />
   561→        <button
   562→          onClick={() => setOrderType(orderType === 'LIMIT' ? 'MARKET' : 'LIMIT')}
   563→          className="px-2 py-0.5 rounded text-[9px] font-bold bg-secondary/70 text-muted-foreground hover:text-foreground"
   564→        >{orderType}</button>
   565→        <Button size="sm" className="h-6 px-2 text-[10px]" onClick={create} disabled={busy}><Plus className="h-3 w-3" /> Arm GTT</Button>
   566→      </div>
   567→      {rules.length === 0 && <div className="text-[10px] text-muted-foreground py-1">No GTT rules — arm one above (fires server-side, works for sim + live accounts).</div>}
   568→      <div className="space-y-1">
   569→        {rules.map(r => (
   570→          <div key={r.id} className="flex items-center gap-2 px-1.5 py-1 rounded bg-secondary/30 text-[10px]">
   571→            <span className={`font-bold ${r.side === 'BUY' ? 'text-[#00c896]' : 'text-[#ff4d5e]'}`}>{r.side}</span>
   572→            <span className="font-medium tabular">{r.symbol}</span>
   573→            <span className="text-muted-foreground tabular">×{r.qty}</span>
   574→            <span className="text-muted-foreground">when LTP {r.trigger === 'ABOVE' ? '≥' : '≤'} <b className="text-foreground tabular">{r.triggerPrice}</b> ({r.orderType})</span>
   575→            {r.lastTriggeredAt && (
   576→              <span className={`text-[9px] ${r.lastStatus === 'SUCCESS' ? 'term-up' : 'term-down'}`}>
   577→                {r.lastStatus} {fmtTime(r.lastTriggeredAt)}
   578→              </span>
   579→            )}
   580→            <div className="flex-1" />
   581→            <Switch checked={r.armed} onCheckedChange={v => toggle(r, v)} />
   582→            <button onClick={() => remove(r.id)} className="p-0.5 text-muted-foreground hover:text-[#ff4d5e]"><Trash2 className="h-3 w-3" /></button>
   583→          </div>
   584→        ))}
   585→      </div>
   586→    </Card>
   587→  )
   588→}
