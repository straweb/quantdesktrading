    31→    31→function fmtCompact(n: number | null | undefined): string {
    32→    32→  if (n == null || isNaN(n)) return '—'
    33→    33→  const abs = Math.abs(n)
    34→    34→  if (abs >= 1e7) return `₹${(n / 1e7).toFixed(2)}Cr`
    35→    35→  if (abs >= 1e5) return `₹${(n / 1e5).toFixed(2)}L`
    36→    36→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}k`
    37→    37→  return `₹${Math.round(n)}`
    38→    38→}

    40→    40→/** Slider max for the fixed-₹ capital editor — ~125% of the account's
    41→    41→ *  available cash (or ₹10L when cash is unknown), rounded to a clean step. */
    42→    42→function fixedSliderMax(cash: number | null | undefined): number {
    43→    43→  const base = Math.max(100_000, Math.ceil(((cash ?? 0) * 1.25) / 50_000) * 50_000)
    44→    44→  return Math.max(base, 1_000_000)
    45→    45→}

   187→   187→function CapitalChip({ acc }: { acc: AccountRow }) {
   188→   188→  const qc = useQueryClient()
   189→   189→  const [open, setOpen] = useState(false)
   190→   190→  const [capital, setCapital] = useState(String(acc.initialCapitalFixed ?? acc.initialCapital))
   191→   191→  const [pct, setPct] = useState(String(acc.capitalPct ?? 100))
   192→   192→  const [base, setBase] = useState<'FIXED' | 'AVAILABLE_CASH_PCT'>(acc.capitalBase ?? 'FIXED')
   193→   193→  const [mode, setMode] = useState<CapitalMode>(acc.mode ?? 'INITIAL_PLUS_PNL')
   194→   194→  const [busy, setBusy] = useState(false)
   195→   195→
   196→   196→  // (re)seed the editor fields each time the popover opens — avoids the
   197→   197→  // setState-in-effect cascade lint rule
   198→   198→  const openEditor = (next: boolean) => {
   199→   199→    setOpen(next)
   200→   200→    if (next) {
   201→   201→      setCapital(String(acc.initialCapitalFixed ?? acc.initialCapital))
   202→   202→      setPct(String(acc.capitalPct ?? 100))
   203→   203→      setBase(acc.capitalBase ?? 'FIXED')
   204→   204→      setMode(acc.mode ?? 'INITIAL_PLUS_PNL')
   205→   205→    }
   206→   206→  }
   207→   207→
   208→   208→  const toggleEnabled = async (next: boolean) => {
   209→   209→    try {
   210→   210→      await api.patchAccount(acc.id, { isOrderEnabled: next })
   211→   211→      await qc.invalidateQueries({ queryKey: ['accounts'] })
   212→   212→      toast.success(`${acc.label} order routing ${next ? 'enabled' : 'disabled'}`)
   213→   213→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Update failed') }
   214→   214→  }
   215→   215→
   216→   216→  const cash = acc.availableCash ?? null
   217→   217→  const pctPreview = base === 'AVAILABLE_CASH_PCT' && cash != null ? Math.max(0, cash * ((Number(pct) || 0) / 100)) : null
   218→   218→
   219→   219→  const save = async () => {
   220→   220→    setBusy(true)
   221→   221→    try {
   222→   222→      const body: Record<string, unknown> = { mode, capitalBase: base }
   223→   223→      if (base === 'AVAILABLE_CASH_PCT') {
   224→   224→        body.capitalPct = Math.min(100, Math.max(0, Number(pct) || 0))
   225→   225→      } else {
   226→   226→        body.initialCapital = Number(capital) || 0
   227→   227→      }
   228→   228→      await api.patchAccount(acc.id, body)
   229→   229→      await qc.invalidateQueries({ queryKey: ['accounts'] })
   230→   230→      await qc.invalidateQueries({ queryKey: ['funds'] })
   231→   231→      const desc = base === 'AVAILABLE_CASH_PCT'
   232→   232→        ? `${body.capitalPct}% of available cash${cash != null ? ` (₹${fmtCompact(pctPreview ?? 0)})` : ''}`
   233→   233→        : fmtCompact(Number(capital) || 0)
   234→   234→      toast.success(`${acc.label} capital updated — ${desc} · ${MODE_SHORT[mode]}`)
   235→   235→      setOpen(false)
   236→   236→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Update failed') }
   237→   237→    setBusy(false)
   238→   238→  }
   239→   239→
   240→   240→  return (
   241→   241→    <Popover open={open} onOpenChange={openEditor}>
   242→   242→      <PopoverTrigger asChild>
   243→   243→        <button
   244→   244→          className={`flex items-center gap-1 h-6 pl-1 pr-1.5 rounded-full border transition-colors ${
   245→   245→            acc.isOrderEnabled
   246→   246→              ? 'border-[#00c896]/40 bg-up text-foreground hover:bg-up/70'
   247→   247→              : 'border-border bg-secondary/50 text-muted-foreground hover:text-foreground'
   248→   248→          }`}
   249→   249→          title={`${acc.label} (${acc.brokerType}) — usable ${fmtCompact(acc.usableCapital)} · ${acc.capitalBase === 'AVAILABLE_CASH_PCT' ? `${acc.capitalPct}% of available cash` : 'fixed ₹ capital'} · click to manage capital sizing`}
   250→   250→        >
   251→   251→          <span className={`h-1.5 w-1.5 rounded-full shrink-0 ${acc.isOrderEnabled ? 'bg-[#00c896]' : 'bg-muted-foreground/50'}`} />
   252→   252→          <span className={`h-4 w-4 rounded text-[8px] font-bold grid place-items-center shrink-0 ${brokerIcon(acc.brokerType).cls}`}>
   253→   253→            {brokerIcon(acc.brokerType).glyph}
   254→   254→          </span>
   255→   255→          <span className="tabular font-bold text-[9px]">{fmtCompact(acc.usableCapital)}</span>
   256→   256→          {acc.capitalBase === 'AVAILABLE_CASH_PCT' && (
   257→   257→            <span className="text-[8px] font-bold text-[#e0a63f] tabular">{acc.capitalPct}%</span>
   258→   258→          )}
   259→   259→        </button>

   263→   263→          <div className="min-w-0">
   264→   264→            <div className="font-bold text-xs truncate">{acc.label}</div>
   265→   265→            <div className="text-[9px] text-muted-foreground uppercase tracking-wide">{acc.brokerType}</div>
