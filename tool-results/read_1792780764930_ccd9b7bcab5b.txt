    21→function fmtCompact(n: number | null | undefined): string {
    22→  if (n == null || isNaN(n)) return '—'
    23→  const abs = Math.abs(n)
    24→  if (abs >= 1e7) return `₹${(n / 1e7).toFixed(2)}Cr`
    25→  if (abs >= 1e5) return `₹${(n / 1e5).toFixed(2)}L`
    26→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}k`
    27→  return `₹${Math.round(n)}`
    28→}

    38→  const totalUsable = accounts.reduce((x, a) => x + a.usableCapital, 0)
    39→  const openOrders = (core?.orders ?? []).filter(o => o.status === 'OPEN' || o.status === 'PENDING').length
    40→  const totalOrders = (core?.orders ?? []).length
    41→  const fillRate = totalOrders > 0 ? (totalOrders - openOrders) / totalOrders * 100 : null
    42→
    43→  // realized P&L curve (cumulative over the session's trades, oldest first)

    45→    const trades = [...(core?.trades ?? [])].sort((a, b) => new Date(a.ts).getTime() - new Date(b.ts).getTime())
    46→    let cum = 0
    47→    const pts: { t: string; v: number }[] = []
    48→    for (const t of trades) {
    49→      if (t.realizedPnl == null) continue
    50→      cum += t.realizedPnl
    51→      pts.push({ t: t.ts, v: cum })
    52→    }
    53→    return pts
    54→  }, [core])
    55→
    56→  return (

    58→      <div className="p-3 space-y-3 max-w-[1500px] mx-auto">

   154→// ---------------- MASTER dashboard — system health strip ----------------
   155→// Live admin overview from the sync:admin websocket: feed engine state,
   156→// per-broker session presence, symbol universe, copy-trade safety switches.
   157→function SystemHealth({ adm }: { adm: SyncAdmin | null }) {
   158→  const feed = adm?.feed
   159→  const meta = adm?.accountsMeta ?? []
   160→  const tickAge = feed?.lastTickTs != null ? Math.max(0, Math.round((Date.now() - feed.lastTickTs) / 1000)) : null
   161→  const liveErr = feed?.sourceIsLive ? feed.lastError : null
   162→  return (
   163→    <Card title="Master — system health">
   164→      <div className="grid grid-cols-2 md:grid-cols-4 gap-2">
   165→        <HealthCell
   166→          label="Feed engine"
   167→          main={feed == null ? '—' : feed.sourceIsLive ? (feed.fresh ? 'TICKING' : 'STALLED') : 'SIM FEED'}
   168→          tone={feed == null ? 'neutral' : feed.sourceIsLive && !feed.fresh ? 'warn' : 'ok'}
   169→          sub={feed == null ? 'websocket connecting…' : feed.sourceIsLive
   170→            ? `${feed.symbols} symbols · last tick ${tickAge != null ? `${tickAge}s` : '—'} · ${feed.polls} polls${liveErr ? ` · ${liveErr}` : ''}`
   171→            : `built-in sim · ${feed.symbols ?? 0} symbols`}
   172→        />
   173→        <HealthCell
   174→          label="Broker sessions"
   175→          main={meta.length === 0 ? '—' : `${meta.filter(m => m.hasCreds).length}/${meta.length} live`}
   176→          tone={meta.length === 0 ? 'neutral' : meta.filter(m => m.brokerType !== 'SIMFEED' && m.isActive && !m.hasCreds).length > 0 ? 'warn' : 'ok'}
   177→          sub={meta.map(m => `${m.label}${m.hasCreds ? '' : ' (no session)'}`).join(' · ') || 'no live broker accounts'}
   178→        />
   179→        <HealthCell
   180→          label="Symbol universe"
   181→          main={adm ? `${adm.syncInfo.counts.symbols}` : '—'}
   182→          tone="neutral"
   183→          sub={adm ? `${adm.syncInfo.counts.mappings} broker mappings · synced ${adm.syncInfo.lastSync ? fmtTime(adm.syncInfo.lastSync) : 'never'}` : ''}
   184→        />
   185→        <HealthCell
   186→          label="Copy trade + recorder"
   187→          main={adm ? `${adm.copyTrade?.enabled ? 'COPY ON' : 'COPY OFF'}${adm.copyTrade?.killSwitch ? ' · KILL' : ''} · REC ${adm.tickRecorder?.enabled ? 'ON' : 'OFF'}` : '—'}
   188→          tone={adm?.copyTrade?.killSwitch ? 'warn' : 'neutral'}
   189→          sub={adm?.copyTrade ? `share ${adm.copyTrade.capitalSharePct}% · slippage ${adm.copyTrade.slippagePct}% · ${adm.tickRecorder?.symbols.join('/') ?? ''}` : ''}
   190→        />
   191→      </div>
   192→    </Card>
   193→  )

   218→      <div className="flex items-center justify-between">

   361→/** Tiny SVG line chart — cumulative realized P&L. */
   362→function PnlCurve({ pts }: { pts: { t: string; v: number }[] }) {
   363→  if (pts.length < 2) {
   364→    return <div className="h-24 flex items-center justify-center text-[10px] text-muted-foreground">no closed trades yet this session</div>
   365→  }
   366→  const W = 480, H = 96, P = 4
   367→  const vs = pts.map(p => p.v)
   368→  const min = Math.min(0, ...vs), max = Math.max(0, ...vs)
   369→  const span = max - min || 1
   370→  const x = (i: number) => P + (i / (pts.length - 1)) * (W - 2 * P)
   371→  const y = (v: number) => H - P - ((v - min) / span) * (H - 2 * P)
   372→  const d = pts.map((p, i) => `${i === 0 ? 'M' : 'L'}${x(i).toFixed(1)},${y(p.v).toFixed(1)}`).join(' ')
   373→  const last = vs[vs.length - 1]
   374→  const col = last >= 0 ? '#00c896' : '#ff4d5e'
   375→  return (
   376→    <div>
   377→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-24">
   378→        <line x1={P} x2={W - P} y1={y(0)} y2={y(0)} stroke="rgba(255,255,255,0.15)" strokeDasharray="3,3" />
   379→        <path d={d} fill="none" stroke={col} strokeWidth="1.6" />
   380→      </svg>
   381→      <div className={`text-[9px] tabular ${last >= 0 ? 'term-up' : 'term-down'}`}>
   382→        cumulative {fmtInr(last)} over {pts.length} closes · last {fmtTime(pts[pts.length - 1].t)}
   383→      </div>
   384→    </div>
   385→  )
   386→}
