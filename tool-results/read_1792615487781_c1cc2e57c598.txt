    71→// ---------- seeded PRNG for deterministic history ----------
    72→function mulberry32(seed: number) {
    73→  return function () {
    74→    let t = (seed += 0x6d2b79f5)
    75→    t = Math.imul(t ^ (t >>> 15), t | 1)
    76→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
    77→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    78→  }
    79→}

    81→function hashStr(s: string): number {
    82→  let h = 5381
    83→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) >>> 0
    84→  return h
    85→}

    87→function volaFor(instrumentType: string): number {
    88→  // per-250ms-tick volatility (annualizes to realistic daily moves)
    89→  if (instrumentType === 'INDEX') return 0.00006
    90→  if (instrumentType === 'FUT') return 0.0001
    91→  if (instrumentType === 'CE' || instrumentType === 'PE') return 0.0007
    92→  return 0.00012
    93→}

    95→function volaHist(instrumentType: string): number {
    96→  // per 5s history sub-step
    97→  if (instrumentType === 'INDEX') return 0.00018
    98→  if (instrumentType === 'FUT') return 0.0003
    99→  if (instrumentType === 'CE' || instrumentType === 'PE') return 0.0022
   100→  return 0.00035
   101→}

   156→function openTickSession(): void {
   157→  if (!gTimers.__tickRecTimer) {
   158→    gTimers.__tickRecTimer = setInterval(recordTickOnce, 250)
   159→  }
   160→  if (!tickCfg.enabled) { closeTickSession(); return }
   161→  if (gTick.__tickSession) return
   162→  try {
   163→    const now = Date.now()
   164→    const day = istDateStr(now)
   165→    const dir = path.join(TICK_DIR, day)
   166→    mkdirSync(dir, { recursive: true })
   167→    const d = new Date(now + 5.5 * 3600_000)
   168→    const file = path.join(dir, `session-${d.toISOString().slice(11, 19).replaceAll(':', '')}.db`)
   169→    const DatabaseAny = Database as unknown as new (p: string) => InstanceType<typeof Database>
   170→    const dbh = new DatabaseAny(file)
   171→    dbh.exec(`
   172→      CREATE TABLE IF NOT EXISTS ticks (
   173→        ts INTEGER NOT NULL, symbol TEXT NOT NULL, ltp REAL, bid REAL, ask REAL,
   174→        vol INTEGER, high REAL, low REAL, open REAL, prevClose REAL
   175→      );
   176→      CREATE INDEX IF NOT EXISTS ix_ticks_sym_ts ON ticks(symbol, ts);
   177→      CREATE TABLE IF NOT EXISTS meta (k TEXT PRIMARY KEY, v TEXT);
   178→    `)
   179→    dbh.prepare(`INSERT OR REPLACE INTO meta (k, v) VALUES ('startedAt', ?), ('symbols', ?)`).run(
   180→      new Date(now).toISOString(), tickCfg.symbols.join(','),
   181→    )
   182→    gTick.__tickSession = { db: dbh, file, startedAt: new Date(now).toISOString(), istDate: day, rows: 0 }
   183→    console.log(`[market-feed] tick recorder ON — session ${file} (${tickCfg.symbols.join(', ')})`)
   184→  } catch (e) {
   185→    console.error('[market-feed] tick session open failed:', e)
   186→  }
   187→}

   196→function recordTickOnce(): void {
   197→  const s = gTick.__tickSession
   198→  if (!tickCfg.enabled) { if (s) closeTickSession(); return }
   199→  if (!s) { openTickSession(); return }
   200→  // new IST day -> new session file (a fresh db per trading session)
   201→  if (s.istDate !== istDateStr()) { closeTickSession(); openTickSession(); return }
   202→  if (!s.db) return
   203→  const now = Date.now()
   204→  const stmt = s.db.prepare(`INSERT INTO ticks (ts, symbol, ltp, bid, ask, vol, high, low, open, prevClose) VALUES (?,?,?,?,?,?,?,?,?,?)`)
   205→  const many: unknown[][] = []
   206→  for (const sym of tickCfg.symbols) {
   207→    const st = S.states.get(sym)
   208→    if (!st) continue
   209→    many.push([now, sym, st.ltp, st.bid, st.ask, st.vol, st.high, st.low, st.open, st.prevClose])
   210→  }
   211→  if (many.length === 0) return
   212→  try {
   213→    s.db.transaction(() => { for (const r of many) stmt.run(...(r as [unknown])) })()
   214→    s.rows += many.length
   215→  } catch (e) {
   216→    console.error('[market-feed] tick write failed:', e)
   217→  }
   218→}

   228→  const out: { file: string; bytes: number; mtime: string }[] = []

   233→      for (const f of readdirSync(dayDir)) {

   236→        const stt = statSync(p)
   237→        out.push({ file: path.relative(path.resolve(process.cwd(), '../..'), p), bytes: stt.size, mtime: stt.mtime.toISOString() })

   239→    }
   240→  } catch { /* ignore */ }
   241→  return out.sort((a, b) => (a.file < b.file ? 1 : -1)).slice(0, 40)

   287→function bucketStart(ts: number, tf: string): number {
   288→  const ms = TF_MS[tf] ?? 60000
   289→  return Math.floor(ts / ms) * ms
   290→}

   292→function buildHistory(entry: UniverseEntry) {
   293→  const rnd = mulberry32(hashStr(entry.symbol) % 2 ** 31)
   294→  // independent stream for volume so adding index volume never perturbs the
   295→  // deterministic price path (same rnd() call sequence as before)
   296→  const vrnd = mulberry32((hashStr(entry.symbol) ^ 0x9e3779b9) >>> 0)
   297→  const vola = volaHist(entry.instrumentType)
   298→  const now = Date.now()
   299→  const oneMin = TF_MS['1m']
   300→  const startTs = bucketStart(now, '1m') - DAYS_BACK * 24 * 3600_000
   301→  const bars: Candle[] = []
   302→  let price = entry.basePrice * (0.995 + rnd() * 0.01)
   303→  const MEAN_REV = 0.0006
   304→
   305→  for (let t = startTs; t <= bucketStart(now, '1m'); t += oneMin) {
   306→    const o = price
   307→    let h = o, l = o, c = o
   308→    // 12 sub-steps per minute
   309→    for (let i = 0; i < 12; i++) {
   310→      const shock = (rnd() - 0.5) * 2 * vola
   311→      c = c * (1 + shock + MEAN_REV * (entry.basePrice - c) / entry.basePrice)
   312→      h = Math.max(h, c)
   313→      l = Math.min(l, c)
   314→    }
   315→    const v = Math.round((entry.instrumentType === 'INDEX' ? 2200 + vrnd() * 19000 : 400 + vrnd() * 2600) * (entry.lotSize > 1 ? 1 : 8))
   316→    bars.push({ t, o: r2(o), h: r2(h), l: r2(l), c: r2(c), v })
   317→    price = c
   318→  }
   319→  S.candles.set(`${entry.symbol}|1m`, bars)
   320→  return { last: bars[bars.length - 1]?.c ?? entry.basePrice, prevDayClose: entry.basePrice * 0.997 }
   321→}

   323→function r2(n: number) { return Math.round(n * 100) / 100 }

   325→function loadUniverse(list: UniverseEntry[]) {
   326→  for (const e of list) {
   327→    if (S.states.has(e.symbol)) continue
   328→    const { last, prevDayClose } = buildHistory(e)
   329→    const bars = S.candles.get(`${e.symbol}|1m`)!
   330→    const todayBars = bars.slice(-375)
   331→    S.states.set(e.symbol, {
   332→      symbol: e.symbol,
   333→      instrumentType: e.instrumentType,
   334→      lotSize: e.lotSize,
   335→      base: e.basePrice,
   336→      ltp: last,
   337→      prevClose: r2(prevDayClose),
   338→      open: todayBars[0]?.o ?? last,
   339→      high: Math.max(...todayBars.map(b => b.h)),
   340→      low: Math.min(...todayBars.map(b => b.l)),
   341→      vol: todayBars.reduce((s, b) => s + b.v, 0),
   342→      vola: volaFor(e.instrumentType),
   343→      bid: r2(last * 0.9995),
   344→      ask: r2(last * 1.0005),
   345→      lastTickDir: 1,
   346→    })
   347→  }
   348→}

   350→// aggregate 1m -> tf (cached arrays appended on rollover)
   351→function getSeries(symbol: string, tf: string): Candle[] {
   352→  const key = `${symbol}|${tf}`
   353→  if (tf === '1m') return S.candles.get(key) ?? []
   354→  const base = S.candles.get(`${symbol}|1m`) ?? []
   355→  let series = S.candles.get(key)
   356→  if (!series) {
   357→    series = aggregate(base, tf)
   358→    S.candles.set(key, series)
   359→  }
   360→  return series
   361→}

   363→function aggregate(base: Candle[], tf: string): Candle[] {
   364→  const ms = TF_MS[tf]
   365→  const out: Candle[] = []
   366→  let cur: Candle | null = null
   367→  for (const b of base) {
   368→    const bs = bucketStart(b.t, tf)
   369→    if (!cur || cur.t !== bs) {
   370→      if (cur) out.push(cur)
   371→      cur = { t: bs, o: b.o, h: b.h, l: b.l, c: b.c, v: b.v }
   372→    } else {
   373→      cur.h = Math.max(cur.h, b.h)
   374→      cur.l = Math.min(cur.l, b.l)
   375→      cur.c = b.c
   376→      cur.v += b.v
   377→    }
   378→  }
   379→  if (cur) out.push(cur)
   380→  return out
   381→}

   383→// ---------- tick engine ----------
   384→function tickOnce() {
   385→  const now = Date.now()
   386→  for (const st of S.states.values()) {
   387→    const shock = (Math.random() - 0.5) * 2 * st.vola
   388→    const meanRev = (st.base - st.ltp) / st.base * 0.0006
   389→    st.ltp = r2(st.ltp * (1 + shock + meanRev))
   390→    st.lastTickDir = shock >= 0 ? 1 : -1
   391→    st.high = Math.max(st.high, st.ltp)
   392→    st.low = Math.min(st.low, st.ltp)
   393→    st.bid = r2(st.ltp - Math.max(0.05, st.ltp * 0.0004))
   394→    st.ask = r2(st.ltp + Math.max(0.05, st.ltp * 0.0004))
   395→    // per-tick traded volume — EQ/FUT/OPT scale with lot size, indices get a
   396→    // synthetic flow calibrated to match the historical per-minute volumes
   397→    // (history: ~2200-21000/min for INDEX, ~3200-24000/min for lot>1 EQ)
   398→    const dv = st.instrumentType !== 'INDEX'
   399→      ? Math.round(Math.random() * st.lotSize * 2)
   400→      : Math.round(10 + Math.random() * 60)
   401→    st.vol += dv
   402→
   403→    // update 1m candle
   404→    const key = `${st.symbol}|1m`
   405→    const bars = S.candles.get(key)
   406→    if (bars) {
   407→      const bs = bucketStart(now, '1m')
   408→      let cur = bars[bars.length - 1]
   409→      if (!cur || cur.t !== bs) {
   410→        cur = { t: bs, o: st.ltp, h: st.ltp, l: st.ltp, c: st.ltp, v: dv }
   411→        bars.push(cur)
   412→        // keep memory bounded
   413→        if (bars.length > DAYS_BACK * 1440 + 500) bars.splice(0, 200)
   414→        // invalidate higher tf caches lazily (they re-aggregate tail)
   415→        for (const tf of Object.keys(TF_MS)) {
   416→          if (tf === '1m') continue
   417→          const sk = `${st.symbol}|${tf}`
   418→          const series = S.candles.get(sk)
   419→          if (series) {
   420→            const b2 = bucketStart(now, tf)
   421→            const last2 = series[series.length - 1]
   422→            if (last2 && last2.t === b2) {
   423→              last2.h = Math.max(last2.h, st.ltp)
   424→              last2.l = Math.min(last2.l, st.ltp)
   425→              last2.c = st.ltp
   426→              last2.v += dv
   427→            } else {
   428→              series.push({ t: b2, o: st.ltp, h: st.ltp, l: st.ltp, c: st.ltp, v: dv })
   429→            }
   430→          }
   431→        }
   432→      } else {
   433→        cur.h = Math.max(cur.h, st.ltp)
   434→        cur.l = Math.min(cur.l, st.ltp)
   435→        cur.c = st.ltp
   436→        cur.v += dv
   437→      }
   438→    }
   439→  }
   440→}

   481→        } catch (e) { json(400, { ok: false, error: String(e) }) }
