    36→/** Re-scale one side's enabled rows so they always sum to 100%.
    37→ *  PCT mode only — FIXED (₹) allocations are absolute, never normalized. */
    38→function normalizeSide(rows: StrikeRowCfg[], type: 'CE' | 'PE'): StrikeRowCfg[] {
    39→  const side = rows.filter(r => r.type === type && r.enabled)
    40→  if (side.length === 0) return rows
    41→  const list = side.map(r => ({ ...r }))
    42→  const n = list.length
    43→  // zero/unset allocs enter with an equal-share weight, then everything
    44→  // is proportionally scaled so the side sums to exactly 100
    45→  for (const r of list) if (!(r.allocPct > 0)) r.allocPct = 100 / n
    46→  const total = list.reduce((s, r) => s + r.allocPct, 0)
    47→  for (const r of list) r.allocPct = Math.round((r.allocPct / total) * 1000) / 10
    48→  // dump rounding drift on the largest allocation
    49→  const drift = Math.round((100 - list.reduce((s, r) => s + r.allocPct, 0)) * 10) / 10
    50→  if (drift !== 0) {
    51→    const big = list.reduce((a, b) => (b.allocPct > a.allocPct ? b : a))
    52→    big.allocPct = Math.round(Math.max(0, big.allocPct + drift) * 10) / 10
    53→  }
    54→  const fixed = new Map(list.map(r => [r.id, r] as const))
    55→  return rows.map(r => fixed.get(r.id) ?? r)
    56→}

   286→  const deployRows = async (type: 'CE' | 'PE', rows: StrikeRowCfg[]) => {

   861→// ------------------------------------------------------------
   862→// Per-strike allocation cell — % (with inline slider) or fixed ₹.
   863→// Local edit state, commits on blur/Enter/release (re-normalizing
   864→// on every keystroke would fight the typist).
   865→// ------------------------------------------------------------
   866→function AllocCell({ value, mode, enabled, onCommit }: {
   867→  value: number
   868→  mode: AllocMode
   869→  enabled: boolean
   870→  onCommit: (v: number) => void
   871→}) {
   872→  const [txt, setTxt] = useState(mode === 'FIXED' ? String(Math.round(value)) : String(value))
   873→  const [focus, setFocus] = useState(false)
   874→  // eslint-disable-next-line react-hooks/set-state-in-effect -- sync slider -> text when not focused
   875→  useEffect(() => { if (!focus) setTxt(mode === 'FIXED' ? String(Math.round(value)) : String(Math.round(value * 10) / 10)) }, [value, focus, mode])
   876→
   877→  if (mode === 'FIXED') {
   878→    return (
   879→      <input
   880→        value={txt}
   881→        disabled={!enabled}
   882→        onChange={e => setTxt(e.target.value.replace(/[^0-9]/g, ''))}
   883→        onFocus={() => setFocus(true)}
   884→        onBlur={() => { setFocus(false); onCommit(Number(txt)) }}
   885→        onKeyDown={e => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   886→        title="Fixed ₹ allocation per account for this strike"
   887→        className="h-4.5 w-11 shrink-0 rounded bg-secondary/60 text-center text-[9px] tabular outline-none focus:ring-1 ring-primary/60 disabled:opacity-40"
   888→      />
   889→    )
   890→  }
   891→  return (
   892→    <span className="flex items-center gap-1 flex-1 min-w-0 max-w-[150px] shrink" title="Alloc % of usable capital — drag the slider or type; side auto-normalizes to 100%">
   893→      <input
   894→        type="range"
   895→        min={0}
   896→        max={100}
   897→        step={1}
   898→        value={Math.max(0, Math.min(100, Number(txt) || 0))}
   899→        disabled={!enabled}
   900→        onChange={e => setTxt(e.target.value)}
   901→        onMouseUp={e => onCommit(Number((e.target as HTMLInputElement).value))}
   902→        onTouchEnd={e => onCommit(Number((e.target as HTMLInputElement).value))}
   903→        className="h-1 flex-1 min-w-8 accent-[#3b82f6] cursor-pointer disabled:opacity-40"
   904→      />
   905→      <input
   906→        value={txt}
   907→        disabled={!enabled}
   908→        onChange={e => setTxt(e.target.value.replace(/[^0-9.]/g, ''))}
   909→        onFocus={() => setFocus(true)}
   910→        onBlur={() => { setFocus(false); onCommit(Number(txt)) }}
   911→        onKeyDown={e => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   912→        className="h-4.5 w-9 shrink-0 rounded bg-secondary/60 text-center text-[9px] tabular outline-none focus:ring-1 ring-primary/60 disabled:opacity-40"
   913→      />
   914→    </span>
   915→  )
   916→}

   922→// ------------------------------------------------------------
   923→function StrikeRowEditor({ row, cell, lots, pos, lotSize, onApply, onExit, onEntry, onTarget, onStop, onRemove }: {
   924→  row: StrikeRowCfg
   925→  cell: OptionQuoteCell | null
   926→  lots: number
   927→  pos: { netQty: number; avgPrice: number; unrealizedPnl: number } | null
   928→  lotSize: number
   929→  onApply: (deltaLots: number) => Promise<void>
   930→  onExit: () => Promise<void>
   931→  onEntry: (side: 'BUY' | 'SELL', price: number) => Promise<void>
   932→  onTarget: (price: number) => Promise<void>
   933→  onStop: (price: number) => Promise<void>
   934→  onRemove: () => void
   935→}) {
   936→  const [delta, setDelta] = useState('')
   937→  const [entry, setEntry] = useState('')
   938→  const [target, setTarget] = useState('')
   939→  const [stop, setStop] = useState('')
   940→  const [busy, setBusy] = useState(false)
   941→  useEffect(() => setDelta(''), [lots])
   942→
   943→  const flat = !pos || pos.netQty === 0
   944→  const num = (s: string) => (s === '' ? NaN : Number(s))
   945→  const ok = (s: string) => Number.isFinite(num(s)) && num(s) > 0
   946→
   947→  const run = async (fn: () => Promise<void>) => {
   948→    setBusy(true)
   949→    try { await fn() } finally { setBusy(false) }
   950→  }
   951→
   952→  return (
   953→    <div className="px-2 pb-2 pt-1 space-y-1.5 border-t border-border/40 bg-secondary/20">
   954→      <div className="flex items-center justify-between text-[10px]">
   955→        <span className="text-muted-foreground">Position — {cell?.symbol ?? row.type}</span>
   956→        <span className="tabular">
   957→          {pos ? (
   958→            <>
   959→              {pos.netQty > 0 ? 'LONG' : 'SHORT'} {lots} lot(s) @ {fmtMoney(pos.avgPrice)} ·{' '}
   960→              <b className={pos.unrealizedPnl >= 0 ? 'term-up' : 'term-down'}>{fmtMoney(pos.unrealizedPnl)}</b>
   961→            </>
   962→          ) : 'flat'}
   963→        </span>
   964→      </div>
   965→
   966→      {/* price orders: entry / target / stoploss */}
   967→      <div className="space-y-1 rounded bg-card/60 p-1.5">
   968→        <div className="text-[8px] font-semibold uppercase tracking-wide text-muted-foreground">orders at price</div>
   969→        <div className="flex items-center gap-1 text-[10px]">
   970→          <span className="w-10 shrink-0 text-muted-foreground">Entry</span>
   971→          <Input
   972→            value={entry}
   973→            onChange={e => setEntry(e.target.value.replace(/[^0-9.]/g, ''))}
   974→            placeholder="limit ₹"
   975→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   976→          />
   977→          <button
   978→            disabled={busy || !ok(entry)}
   979→            onClick={() => run(async () => { await onEntry('BUY', num(entry)); setEntry('') })}
   980→            title={`BUY LIMIT ${row.strike} ${row.type} at this price (${row.allocPct}% per account)`}
   981→            className="px-1.5 h-6 rounded bg-up text-[#00c896] hover:brightness-125 text-[9px] font-bold disabled:opacity-40"
   982→          >B@</button>
   983→          <button
   984→            disabled={busy || !ok(entry)}
   985→            onClick={() => run(async () => { await onEntry('SELL', num(entry)); setEntry('') })}
   986→            title={`SELL LIMIT ${row.strike} ${row.type} at this price (${row.allocPct}% per account)`}
   987→            className="px-1.5 h-6 rounded bg-down text-[#ff4d5e] hover:brightness-125 text-[9px] font-bold disabled:opacity-40"
   988→          >S@</button>
   989→        </div>
   990→        <div className="flex items-center gap-1 text-[10px]">
   991→          <span className="w-10 shrink-0 text-muted-foreground">Target</span>
   992→          <Input
   993→            value={target}
   994→            onChange={e => setTarget(e.target.value.replace(/[^0-9.]/g, ''))}
   995→            placeholder={pos && pos.netQty > 0 ? 'book long ₹' : 'book short ₹'}
   996→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   997→          />
   998→          <button
   999→            disabled={busy || flat || !ok(target)}
  1000→            onClick={() => run(async () => { await onTarget(num(target)); setTarget('') })}
  1001→            title={`Reduce-only LIMIT at this price — closes the open position when LTP reaches it`}
  1002→            className="px-1.5 h-6 rounded bg-secondary text-foreground hover:bg-accent text-[9px] font-bold disabled:opacity-40"
  1003→          >Place</button>
  1004→        </div>
  1005→        <div className="flex items-center gap-1 text-[10px]">
  1006→          <span className="w-10 shrink-0 text-muted-foreground">StopLoss</span>
  1007→          <Input
  1008→            value={stop}
  1009→            onChange={e => setStop(e.target.value.replace(/[^0-9.]/g, ''))}
  1010→            placeholder="trigger ₹"
  1011→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
  1012→          />
  1013→          <button
  1014→            disabled={busy || flat || !ok(stop)}
  1015→            onClick={() => run(async () => { await onStop(num(stop)); setStop('') })}
  1016→            title={`Reduce-only SL-M — market-exits the open position when LTP triggers at this price`}
  1017→            className="px-1.5 h-6 rounded bg-secondary text-foreground hover:bg-accent text-[9px] font-bold disabled:opacity-40"
  1018→          >Arm</button>
  1019→        </div>
  1020→        <div className="text-[8px] text-muted-foreground leading-snug">
  1021→          Entry rests as LIMIT until hit · Target/SL attach to the current position ({lots} lot(s) · lot {lotSize}) and auto-cancel if it closes
  1022→        </div>
  1023→      </div>
  1024→
  1025→      <div className="flex items-center gap-1 text-[10px]">
  1026→        <span className="text-muted-foreground">Δ Lots /leg</span>
  1027→        <Input
  1028→          value={delta}
  1029→          onChange={e => setDelta(e.target.value.replace(/[^0-9-]/g, ''))}
  1030→          placeholder="+2 / -1"
  1031→          className="h-6 w-16 text-[10px] bg-secondary/60"
  1032→        />
  1033→        <Button
  1034→          size="sm" variant="secondary" className="h-6 px-2 text-[10px]" disabled={busy || !delta || delta === '-' || delta === '+0'}
  1035→          onClick={async () => { setBusy(true); await onApply(Number(delta)); setBusy(false) }}
  1036→        >Apply</Button>
  1037→        <span className="text-[9px] text-muted-foreground">+ buys, − exits (per enabled account) · now {lots} lot(s) agg</span>
  1038→        <div className="flex-1" />
  1039→        <Button
  1040→          size="sm" variant="secondary" className="h-6 px-2 text-[10px] text-[#ff4d5e]" disabled={busy || !pos}
  1041→          onClick={async () => { setBusy(true); await onExit(); setBusy(false) }}
  1042→        >Exit full</Button>
  1043→        <Button size="sm" variant="ghost" className="h-6 px-1.5 text-[10px] text-muted-foreground hover:text-[#ff4d5e]" onClick={onRemove}>
  1044→          <Trash2 className="h-3 w-3" />
  1045→        </Button>
  1046→      </div>
  1047→      <div className="flex items-center gap-1 text-[10px]">
  1048→        <span className="text-[9px] text-muted-foreground">alloc {row.allocPct}% · lot {lotSize} · orders fan out to all enabled accounts</span>
  1049→      </div>
  1050→    </div>
  1051→  )
