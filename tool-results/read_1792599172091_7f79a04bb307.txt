    30→/** Re-scale one side's enabled rows so they always sum to 100%. */
    31→function normalizeSide(rows: StrikeRowCfg[], type: 'CE' | 'PE'): StrikeRowCfg[] {
    32→  const side = rows.filter(r => r.type === type && r.enabled)
    33→  if (side.length === 0) return rows
    34→  const list = side.map(r => ({ ...r }))
    35→  const n = list.length
    36→  // zero/unset allocs enter with an equal-share weight, then everything
    37→  // is proportionally scaled so the side sums to exactly 100
    38→  for (const r of list) if (!(r.allocPct > 0)) r.allocPct = 100 / n
    39→  const total = list.reduce((s, r) => s + r.allocPct, 0)
    40→  for (const r of list) r.allocPct = Math.round((r.allocPct / total) * 1000) / 10
    41→  // dump rounding drift on the largest allocation
    42→  const drift = Math.round((100 - list.reduce((s, r) => s + r.allocPct, 0)) * 10) / 10
    43→  if (drift !== 0) {
    44→    const big = list.reduce((a, b) => (b.allocPct > a.allocPct ? b : a))
    45→    big.allocPct = Math.round(Math.max(0, big.allocPct + drift) * 10) / 10
    46→  }
    47→  const fixed = new Map(list.map(r => [r.id, r] as const))
    48→  return rows.map(r => fixed.get(r.id) ?? r)
    49→}

    51→export function StrikesPanel({ ws }: { ws: WorkspaceCfg }) {

    53→  const st: StrikesCfg = useMemo(() => ws.strikes ?? defaultStrikes(), [ws])
    54→
    55→  const { data: undData } = useQuery({
    56→    queryKey: ['optionUnderlyings'],
    57→    queryFn: () => api.optionUnderlyings(),
    58→    staleTime: 5 * 60_000,
    59→  })
    60→  const underlyings = undData?.underlyings ?? []
    61→
    62→  // auto-bind the workspace's F&O symbol on first open
    63→  useEffect(() => {
    64→    if (!st.underlying && underlyings.some(u => u.underlying === ws.symbol)) {
    65→      patchWs(ws.id, { strikes: { ...st, underlying: ws.symbol } })

    69→  const underlying = st.underlying
    70→  const strikesParam = useMemo(() => [...new Set(st.rows.map(r => r.strike))].sort((a, b) => a - b), [st.rows])

    89→    return m
    90→  }, [posData])
    91→
    92→  const [specific, setSpecific] = useState('')
    93→  const [sign, setSign] = useState<1 | -1>(1)
    94→  const [busy, setBusy] = useState(false)
    95→  const [expanded, setExpanded] = useState<string | null>(null)
    96→
    97→  const setStrikes = (next: Partial<StrikesCfg>) => patchWs(ws.id, { strikes: { ...st, ...next } })
    98→
    99→  const addStrike = async (strike: number, type: 'CE' | 'PE') => {
   100→    if (!underlying) { toast.error('Pick an F&O underlying first'); return }
   101→    const id = `${strike}:${type}`
   102→    // read fresh store state (consecutive/parallel adds must not clobber each other)
   103→    const before = useTerminal.getState().workspaces.find(w => w.id === ws.id)?.strikes ?? st
   104→    if (before.rows.some(r => r.id === id)) { toast.info(`${strike} ${type} already in list`); return }
   105→    try {
   106→      const r = await api.ensureStrike(underlying, strike, st.expiry)
   107→      if (r.created > 0) toast.success(`Contracts ready — ${r.symbols.join(', ')}`)
   108→    } catch (e) {
   109→      toast.error(e instanceof Error ? e.message : 'Failed to create contracts')
   110→      return
   111→    }
   112→    // re-read AFTER the await so an interleaved add is not overwritten
   113→    const cur = useTerminal.getState().workspaces.find(w => w.id === ws.id)?.strikes ?? before
   114→    if (cur.rows.some(r => r.id === id)) { return }
   115→    const rows = normalizeSide([...cur.rows, { id, strike, type, enabled: true, allocPct: 0 }], type)
   116→    patchWs(ws.id, { strikes: { ...cur, rows } })
   117→    setTimeout(() => refetchChain(), 150)
   118→  }
   119→
   120→  /** ATM -> CE@ceAtm + PE@peAtm · offset n -> CE/PE n steps from their side ATM.
   121→   *  Sequential (not Promise.all) — parallel adds would race on the store. */
   122→  const addOffset = async (n: number) => {
   123→    if (!meta) { toast.error('Chain not loaded yet'); return }
   124→    if (n === 0) {
   125→      await addStrike(meta.ceAtm, 'CE')
   126→      await addStrike(meta.peAtm, 'PE')
   127→      return
   128→    }
   129→    const ce = meta.ceAtm + sign * n * meta.step
   130→    const pe = meta.peAtm + sign * n * meta.step
   131→    await addStrike(ce, 'CE')
   132→    await addStrike(pe, 'PE')
   133→  }
   134→
   135→  const addSpecific = async () => {
   136→    const raw = Number(specific)
   137→    if (!underlying || !Number.isFinite(raw) || raw <= 0) { toast.error('Enter a valid strike'); return }
   138→    const step = meta?.step ?? 50
   139→    const snapped = Math.round(raw / step) * step
   140→    if (snapped !== raw) toast.info(`Snapped ${raw} → ${snapped} (step ${step})`)
   141→    setSpecific('')
   142→    await addStrike(snapped, 'CE')
   143→    await addStrike(snapped, 'PE')
   144→  }

   146→  const toggleRow = (row: StrikeRowCfg) => {
   147→    const rows = st.rows.map(r => (r.id === row.id ? { ...r, enabled: !r.enabled } : r))
   148→    setStrikes({ rows: normalizeSide(rows, row.type) })

   150→

   152→    const clamped = Math.max(0, Math.min(100, Number.isFinite(v) ? v : 0))
   153→    const rows = st.rows.map(r => (r.id === row.id ? { ...r, allocPct: clamped } : r))
   154→    setStrikes({ rows: normalizeSide(rows, row.type) })
   155→  }
   156→
   157→  const removeRow = (row: StrikeRowCfg) => {
   158→    const rows = st.rows.filter(r => r.id !== row.id)
   159→    setStrikes({ rows: normalizeSide(rows, row.type) })
   160→    setExpanded(null)

   163→  const pinToPane = (row: StrikeRowCfg) => {

   167→
   168→  // --- order helpers ---
   169→  const rowsFor = (type: 'CE' | 'PE') => st.rows.filter(r => r.type === type && r.enabled)
   170→
   171→  const cellFor = (row: StrikeRowCfg): OptionQuoteCell | null => {
   172→    const c = rowByStrike.get(row.strike)
   173→    if (!c) return null
   174→    return row.type === 'CE' ? c.ce : c.pe

   176→
   177→  const deploy = async (type: 'CE' | 'PE') => {
   178→    const rows = rowsFor(type)
   179→    if (rows.length === 0) { toast.error(`No enabled ${type} strikes`); return }
   180→    setBusy(true)
   181→    let filled = 0, failed = 0
   182→    for (const row of rows) {
   183→      const cell = cellFor(row)
   184→      if (!cell?.exists || !(row.allocPct > 0)) { failed++; continue }
   185→      try {
   186→        const res = await api.placeOrder({
   187→          symbol: cell.symbol, exchange: 'NFO', side: 'BUY', qty: 0,
   188→          orderType: 'MARKET', productType: 'MIS', source: 'MANUAL', capitalPct: row.allocPct,
   189→        })
   190→        filled += res.legs.filter(l => l.status === 'FILLED').length
   191→        if (res.status === 'FAILED') failed++
   192→      } catch { failed++ }
   193→    }
   194→    setBusy(false)
   195→    filled > 0
   196→      ? toast.success(`Deploy ${type} — ${filled} legs filled${failed ? `, ${failed} skipped` : ''}`)
   197→      : toast.error(`Deploy ${type} failed — ${failed} strikes skipped`)
   198→  }
   199→
   200→  const cancelType = async (type: 'CE' | 'PE') => {
   201→    const syms = rowsFor(type).map(r => cellFor(r)?.symbol).filter((s): s is string => !!s)
   202→    if (syms.length === 0) { toast.error(`No enabled ${type} strikes`); return }
   203→    try {
   204→      const r = await api.cancelAll(syms)
   205→      r.cancelled > 0 ? toast.success(`Cancelled ${r.cancelled} pending ${type} orders`) : toast.info(`No pending ${type} orders`)
   206→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Cancel failed') }
   207→  }
   208→
   209→  const peSwap = async () => {
   210→    const ceRows = rowsFor('CE')
   211→    const ceSyms = ceRows.map(r => cellFor(r)?.symbol).filter((s): s is string => !!s)
   212→    setBusy(true)
   213→    try {
   214→      if (ceSyms.length > 0) {
   215→        const so = await api.squareOffAll(false, ceSyms)
   216→        toast.success(`CE squared off — ${so.filledLegs} legs`)
   217→      }
   218→      // flip on: deploy PE on enabled PE rows (auto-add PE ATM if none listed)
   219→      let peRows = rowsFor('PE')
   220→      if (peRows.length === 0) {
   221→        if (meta) {
   222→          await addStrike(meta.peAtm, 'PE')
   223→          await new Promise(r => setTimeout(r, 500))
   224→          const ws2 = useTerminal.getState().workspaces.find(w => w.id === ws.id)
   225→          peRows = (ws2?.strikes?.rows ?? []).filter(r => r.type === 'PE' && r.enabled)
   226→        }
   227→      }
   228→      let filled = 0, failed = 0
   229→      const allStrikes = [...new Set(st.rows.map(r => r.strike))]
   230→      const cd = await api.optionChain(underlying!, st.expiry, allStrikes)
   231→      const fresh = new Map<number, ChainStrikeRow>(cd.rows.map(r => [r.strike, r]))
   232→      for (const row of peRows) {
   233→        const c = fresh.get(row.strike)
   234→        const cell = row.type === 'PE' ? c?.pe : c?.ce
   235→        if (!cell?.exists) { failed++; continue }
   236→        try {
   237→          const res = await api.placeOrder({
   238→            symbol: cell.symbol, exchange: 'NFO', side: 'BUY', qty: 0,
   239→            orderType: 'MARKET', productType: 'MIS', source: 'MANUAL', capitalPct: row.allocPct,
   240→          })
   241→          filled += res.legs.filter(l => l.status === 'FILLED').length
   242→          if (res.status === 'FAILED') failed++
   243→        } catch { failed++ }
   244→      }
   245→      toast.success(`PE Swap complete — ${filled} legs filled${failed ? `, ${failed} skipped` : ''}`)
   246→    } catch (e) {
   247→      toast.error(e instanceof Error ? e.message : 'PE Swap failed')
   248→    } finally { setBusy(false) }
   249→  }
   250→
   251→  const rowBuy = async (row: StrikeRowCfg) => {
   252→    const cell = cellFor(row)
   253→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   254→    try {
   255→      const res = await api.placeOrder({
   256→        symbol: cell.symbol, exchange: 'NFO', side: 'BUY', qty: 0,
   257→        orderType: 'MARKET', productType: 'MIS', source: 'MANUAL', capitalPct: row.allocPct,
   258→      })
   259→      res.status === 'FAILED' ? toast.error(`BUY failed — ${res.message}`) : toast.success(`BUY ${row.strike} ${row.type} — ${res.legs.filter(l => l.status === 'FILLED').length}/${res.legs.length} legs`)
   260→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Order failed') }
   261→  }
   262→
   263→  const rowSell = async (row: StrikeRowCfg) => {
   264→    const cell = cellFor(row)
   265→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   266→    const pos = posBySymbol.get(cell.symbol)
   267→    if (pos && pos.netQty > 0) {
   268→      // long position -> Sell means exit
   269→      const so = await api.squareOffAll(false, [cell.symbol])
   270→      toast.success(`Exited long — ${so.filledLegs} legs`)
   271→      return
   272→    }
   273→    try {
   274→      const res = await api.placeOrder({
   275→        symbol: cell.symbol, exchange: 'NFO', side: 'SELL', qty: 0,
   276→        orderType: 'MARKET', productType: 'MIS', source: 'MANUAL', capitalPct: row.allocPct,
   277→      })
   278→      res.status === 'FAILED' ? toast.error(`SELL failed — ${res.message}`) : toast.success(`SELL ${row.strike} ${row.type} — ${res.legs.filter(l => l.status === 'FILLED').length}/${res.legs.length} legs`)
   279→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Order failed') }
   280→  }
   281→
   282→  const rowExit = async (row: StrikeRowCfg) => {
   283→    const cell = cellFor(row)
   284→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   285→    const pos = posBySymbol.get(cell.symbol)
   286→    if (!pos || pos.netQty === 0) { toast.info('No open position'); return }
   287→    try {
   288→      const so = await api.squareOffAll(false, [cell.symbol])
   289→      so.orders > 0 ? toast.success(`Exit ${row.strike} ${row.type} — ${so.filledLegs} legs`) : toast.info('Nothing to exit')
   290→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Exit failed') }
   291→  }
   292→
   293→  const ceRows = useMemo(() => st.rows.filter(r => r.type === 'CE').sort((a, b) => a.strike - b.strike), [st.rows])
   294→  const peRows = useMemo(() => st.rows.filter(r => r.type === 'PE').sort((a, b) => a.strike - b.strike), [st.rows])
   295→  const sideSum = (type: 'CE' | 'PE') =>
   296→    Math.round(st.rows.filter(r => r.type === type && r.enabled).reduce((s, r) => s + r.allocPct, 0) * 10) / 10
   297→
   298→  const offsetFor = (row: StrikeRowCfg): number | null => {
   299→    const chainRow = rowByStrike.get(row.strike)
   300→    if (!chainRow) return null
   301→    return row.type === 'CE' ? chainRow.offsetCe : chainRow.offsetPe
   302→  }
   303→
   304→  const renderRow = (row: StrikeRowCfg) => {
   305→    const cell = cellFor(row)
   306→    const offset = offsetFor(row)
   307→    const pos = cell?.exists ? posBySymbol.get(cell.symbol) : undefined
   308→    const lots = pos ? Math.abs(pos.netQty) / (meta?.lotSize || 1) : 0
   309→    const pinned = (row.type === 'CE' ? ws.panes?.find(p => p.key === 'ce') : ws.panes?.find(p => p.key === 'pe'))?.strike === row.strike
   310→    return (
   311→      <div key={row.id} className={expanded === row.id ? 'bg-accent/40' : ''}>
   312→        <div className="flex items-center gap-1 px-1.5 py-1">
   313→          <button
   314→            onClick={() => toggleRow(row)}
   315→            title={row.enabled ? 'Enabled for deploy — click to disable' : 'Disabled — click to enable'}
   316→            className={`h-3.5 w-3.5 rounded-sm border flex items-center justify-center shrink-0 ${
   317→              row.enabled ? 'bg-primary border-primary' : 'border-muted-foreground/40'
   318→            }`}
   319→          >
   320→            {row.enabled && <Power className="h-2.5 w-2.5 text-primary-foreground" />}
   321→          </button>
   322→          <AllocInput value={row.allocPct} enabled={row.enabled} onCommit={v => setAlloc(row, v)} />
   323→          <button
   324→            onClick={() => pinToPane(row)}
   325→            title={`Load ${row.strike} ${row.type} into the ${row.type} chart pane`}
   326→            className={`w-[52px] shrink-0 text-left leading-tight hover:bg-accent/70 rounded px-0.5 -mx-0.5 ${pinned ? 'ring-1 ring-primary/60 rounded' : ''}`}
   327→          >
   328→            <span className="block tabular font-semibold text-[11px]">{row.strike}</span>
   329→            <span className={`block text-[8px] font-bold ${row.type === 'CE' ? 'text-[#00c896]' : 'text-[#ff4d5e]'}`}>
   330→              {row.type}{offset == null ? '' : offset === 0 ? ' ·ATM' : offset > 0 ? ` +${offset}` : ` ${offset}`}
   331→            </span>
   332→          </button>
   333→          <div className="flex-1 min-w-0 text-right pr-1">
   334→            <div className={`tabular ${cell && cell.ltp >= cell.prevClose ? 'term-up' : 'term-down'}`}>
   335→              {cell?.exists ? fmtMoney(cell.ltp) : '—'}
   336→            </div>
   337→            <div className="tabular text-[9px] text-muted-foreground">
   338→              {cell?.exists ? `${fmtMoney(cell.high, 1)}/${fmtMoney(cell.low, 1)}` : 'H/L —'}
   339→            </div>
   340→          </div>
   341→          {lots > 0 && (
   342→            <span className={`text-[8px] px-1 rounded ${pos && pos.netQty > 0 ? 'bg-up text-[#00c896]' : 'bg-down text-[#ff4d5e]'}`} title={`${lots} lots @ ${fmtMoney(pos!.avgPrice)}`}>
   343→              {pos && pos.netQty > 0 ? 'L' : 'S'}{lots}
   344→            </span>
   345→          )}
   346→          <div className="flex gap-0.5 shrink-0">
   347→            <button onClick={() => rowBuy(row)} className="px-1 py-0.5 rounded bg-up text-[#00c896] hover:brightness-125 text-[9px] font-bold">B</button>
   348→            <button onClick={() => rowSell(row)} className="px-1 py-0.5 rounded bg-down text-[#ff4d5e] hover:brightness-125 text-[9px] font-bold">S</button>
   349→            <button onClick={() => rowExit(row)} className="px-1 py-0.5 rounded bg-secondary text-muted-foreground hover:text-foreground text-[9px] font-bold">X</button>
   350→            <button
   351→              onClick={() => setExpanded(expanded === row.id ? null : row.id)}
   352→              className={`p-0.5 rounded ${expanded === row.id ? 'text-foreground' : 'text-muted-foreground hover:text-foreground'}`}
   353→              title="Edit open position"
   354→            >
   355→              <Pencil className="h-2.5 w-2.5" />
   356→            </button>
   357→          </div>
   358→        </div>
   359→
   360→        {expanded === row.id && (
   361→          <StrikeRowEditor
   362→            row={row}
   363→            cell={cell}
   364→            lots={lots}
   365→            pos={pos ?? null}
   366→            lotSize={meta?.lotSize ?? 1}
   367→            onApply={async (deltaLots) => {
   368→              const cell2 = cellFor(row)
   369→              if (!cell2?.exists) return
   370→              if (deltaLots > 0) {
   371→                const res = await api.placeOrder({
   372→                  symbol: cell2.symbol, exchange: 'NFO', side: 'BUY', qty: deltaLots,
   373→                  orderType: 'MARKET', productType: 'MIS', source: 'MANUAL',
   374→                })
   375→                res.status === 'FAILED' ? toast.error(`Add failed — ${res.message}`) : toast.success(`+${deltaLots} lot(s)/leg — ${res.legs.filter(l => l.status === 'FILLED').length} legs`)
   376→              } else if (deltaLots < 0) {
   377→                await api.placeOrder({
   378→                  symbol: cell2.symbol, exchange: 'NFO', side: 'SELL', qty: Math.abs(deltaLots),
   379→                  orderType: 'MARKET', productType: 'MIS', source: 'MANUAL', reduceOnly: true,
   380→                })
   381→                toast.success(`${deltaLots} lot(s)/leg exited`)
   382→              }
   383→            }}
   384→            onExit={async () => {
   385→              const cell2 = cellFor(row)
   386→              if (!cell2) return
   387→              const so = await api.squareOffAll(false, [cell2.symbol])
   388→              toast.success(`Exited — ${so.filledLegs} legs`)
   389→            }}
   390→            onRemove={() => removeRow(row)}
   391→          />
   392→        )}
   393→      </div>
   394→    )
   395→  }
   396→
   397→  const sideHeader = (type: 'CE' | 'PE') => {
   398→    const sum = sideSum(type)
   399→    const n = st.rows.filter(r => r.type === type && r.enabled).length
   400→    const tint = type === 'CE' ? 'text-[#00c896]' : 'text-[#ff4d5e]'
   401→    return (
   402→      <div className="flex items-center gap-1.5 px-1.5 py-0.5 bg-secondary/40 border-b border-border/60 sticky top-0 z-10">
   403→        <span className={`text-[9px] font-bold ${tint}`}>{type}</span>
   404→        <span className="text-[8px] text-muted-foreground">
   405→          {type === 'CE' && meta ? `ATM ${meta.ceAtm}` : meta ? `ATM ${meta.peAtm}` : ''}

   525→            Deploy PE
   526→          </Button>
   527→          <Button size="sm" variant="secondary" className="h-7 text-[10px]" disabled={busy || !underlying} onClick={() => cancelType('CE')}>

   529→          </Button>

   532→          </Button>

   540→// Per-strike alloc % — local edit state, commits on blur/Enter
   541→// (re-normalizing on every keystroke would fight the typist)
   542→// ------------------------------------------------------------
   543→function AllocInput({ value, enabled, onCommit }: { value: number; enabled: boolean; onCommit: (v: number) => void }) {
   544→  const [txt, setTxt] = useState(String(value))
   545→  const [focus, setFocus] = useState(false)
   546→  useEffect(() => { if (!focus) setTxt(String(Math.round(value * 10) / 10)) }, [value, focus])
   547→  return (
   548→    <input
   549→      value={txt}
   550→      disabled={!enabled}
   551→      onChange={e => setTxt(e.target.value.replace(/[^0-9.]/g, ''))}
   552→      onFocus={() => setFocus(true)}
   553→      onBlur={() => { setFocus(false); onCommit(Number(txt)) }}
   554→      onKeyDown={e => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   555→      title="Alloc % of usable capital — side auto-normalizes to 100%"
   556→      className="h-4.5 w-9 shrink-0 rounded bg-secondary/60 text-center text-[9px] tabular outline-none focus:ring-1 ring-primary/60 disabled:opacity-40"
   557→    />
   558→  )
   559→}
