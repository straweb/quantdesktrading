    71→    71→// ---------- seeded PRNG for deterministic history ----------
    72→    72→function mulberry32(seed: number) {
    73→    73→  return function () {
    74→    74→    let t = (seed += 0x6d2b79f5)
    75→    75→    t = Math.imul(t ^ (t >>> 15), t | 1)
    76→    76→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
    77→    77→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    78→    78→  }
    79→    79→}

    81→    81→function hashStr(s: string): number {
    82→    82→  let h = 5381
    83→    83→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) >>> 0
    84→    84→  return h
    85→    85→}

   287→   287→function bucketStart(ts: number, tf: string): number {
   288→   288→  const ms = TF_MS[tf] ?? 60000

   292→   292→function buildHistory(entry: UniverseEntry) {
   293→   293→  const rnd = mulberry32(hashStr(entry.symbol) % 2 ** 31)
   294→   294→  // independent stream for volume so adding index volume never perturbs the
   295→   295→  // deterministic price path (same rnd() call sequence as before)
   296→   296→  const vrnd = mulberry32((hashStr(entry.symbol) ^ 0x9e3779b9) >>> 0)
   297→   297→  const vola = volaHist(entry.instrumentType)
   298→   298→  const now = Date.now()
   299→   299→  const oneMin = TF_MS['1m']
   300→   300→  const startTs = bucketStart(now, '1m') - DAYS_BACK * 24 * 3600_000
   301→   301→  const bars: Candle[] = []
   302→   302→  let price = entry.basePrice * (0.995 + rnd() * 0.01)
   303→   303→  const MEAN_REV = 0.0006
   304→   304→
   305→   305→  for (let t = startTs; t <= bucketStart(now, '1m'); t += oneMin) {
   306→   306→    const o = price
   307→   307→    let h = o, l = o, c = o
   308→   308→    // 12 sub-steps per minute
   309→   309→    for (let i = 0; i < 12; i++) {
   310→   310→      const shock = (rnd() - 0.5) * 2 * vola
   311→   311→      c = c * (1 + shock + MEAN_REV * (entry.basePrice - c) / entry.basePrice)
   312→   312→      h = Math.max(h, c)
   313→   313→      l = Math.min(l, c)
   314→   314→    }
   315→   315→    const v = Math.round((entry.instrumentType === 'INDEX' ? 2200 + vrnd() * 19000 : 400 + vrnd() * 2600) * (entry.lotSize > 1 ? 1 : 8))
   316→   316→    bars.push({ t, o: r2(o), h: r2(h), l: r2(l), c: r2(c), v })
   317→   317→    price = c
   318→   318→  }
   319→   319→  S.candles.set(`${entry.symbol}|1m`, bars)
   320→   320→  return { last: bars[bars.length - 1]?.c ?? entry.basePrice, prevDayClose: entry.basePrice * 0.997 }
   321→   321→}

   363→   363→function aggregate(base: Candle[], tf: string): Candle[] {
   364→   364→  const ms = TF_MS[tf]
   365→   365→  const out: Candle[] = []
   366→   366→  let cur: Candle | null = null
   367→   367→  for (const b of base) {
   368→   368→    const bs = bucketStart(b.t, tf)
   369→   369→    if (!cur || cur.t !== bs) {
   370→   370→      if (cur) out.push(cur)
   371→   371→      cur = { t: bs, o: b.o, h: b.h, l: b.l, c: b.c, v: b.v }
   372→   372→    } else {
   373→   373→      cur.h = Math.max(cur.h, b.h)
   374→   374→      cur.l = Math.min(cur.l, b.l)
   375→   375→      cur.c = b.c
   376→   376→      cur.v += b.v
   377→   377→    }
   378→   378→  }
   379→   379→  if (cur) out.push(cur)
   380→   380→  return out
   381→   381→}

   383→   383→// ---------- tick engine ----------
   384→   384→function tickOnce() {
   385→   385→  const now = Date.now()
   386→   386→  for (const st of S.states.values()) {
   387→   387→    const shock = (Math.random() - 0.5) * 2 * st.vola
   388→   388→    const meanRev = (st.base - st.ltp) / st.base * 0.0006
   389→   389→    st.ltp = r2(st.ltp * (1 + shock + meanRev))
   390→   390→    st.lastTickDir = shock >= 0 ? 1 : -1
   391→   391→    st.high = Math.max(st.high, st.ltp)
   392→   392→    st.low = Math.min(st.low, st.ltp)
   393→   393→    st.bid = r2(st.ltp - Math.max(0.05, st.ltp * 0.0004))
   394→   394→    st.ask = r2(st.ltp + Math.max(0.05, st.ltp * 0.0004))
   395→   395→    // per-tick traded volume — EQ/FUT/OPT scale with lot size, indices get a
   396→   396→    // synthetic flow calibrated to match the historical per-minute volumes
   397→   397→    // (history: ~2200-21000/min for INDEX, ~3200-24000/min for lot>1 EQ)
   398→   398→    const dv = st.instrumentType !== 'INDEX'
   399→   399→      ? Math.round(Math.random() * st.lotSize * 2)
   400→   400→      : Math.round(10 + Math.random() * 60)
   401→   401→    st.vol += dv
   402→   402→
   403→   403→    // update 1m candle
   404→   404→    const key = `${st.symbol}|1m`
   405→   405→    const bars = S.candles.get(key)
   406→   406→    if (bars) {
   407→   407→      const bs = bucketStart(now, '1m')
   408→   408→      let cur = bars[bars.length - 1]
   409→   409→      if (!cur || cur.t !== bs) {
   410→   410→        cur = { t: bs, o: st.ltp, h: st.ltp, l: st.ltp, c: st.ltp, v: dv }
   411→   411→        bars.push(cur)
   412→   412→        // keep memory bounded
   413→   413→        if (bars.length > DAYS_BACK * 1440 + 500) bars.splice(0, 200)
   414→   414→        // invalidate higher tf caches lazily (they re-aggregate tail)
   415→   415→        for (const tf of Object.keys(TF_MS)) {
   416→   416→          if (tf === '1m') continue
   417→   417→          const sk = `${st.symbol}|${tf}`
   418→   418→          const series = S.candles.get(sk)
   419→   419→          if (series) {
   420→   420→            const b2 = bucketStart(now, tf)
   421→   421→            const last2 = series[series.length - 1]
   422→   422→            if (last2 && last2.t === b2) {
   423→   423→              last2.h = Math.max(last2.h, st.ltp)
   424→   424→              last2.l = Math.min(last2.l, st.ltp)
   425→   425→              last2.c = st.ltp
   426→   426→              last2.v += dv
   427→   427→            } else {
   428→   428→              series.push({ t: b2, o: st.ltp, h: st.ltp, l: st.ltp, c: st.ltp, v: dv })
   429→   429→            }
   430→   430→          }
   431→   431→        }
   432→   432→      } else {
   433→   433→        cur.h = Math.max(cur.h, st.ltp)
   434→   434→        cur.l = Math.min(cur.l, st.ltp)
   435→   435→        cur.c = st.ltp
   436→   436→        cur.v += dv
   437→   437→      }
   438→   438→    }
   439→   439→  }
   440→   440→}
