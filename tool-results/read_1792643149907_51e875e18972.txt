    34→/** Re-scale one side's enabled rows so they always sum to 100%.
    35→ *  PCT mode only — FIXED (₹) allocations are absolute, never normalized. */
    36→function normalizeSide(rows: StrikeRowCfg[], type: 'CE' | 'PE'): StrikeRowCfg[] {
    37→  const side = rows.filter(r => r.type === type && r.enabled)
    38→  if (side.length === 0) return rows
    39→  const list = side.map(r => ({ ...r }))
    40→  const n = list.length
    41→  // zero/unset allocs enter with an equal-share weight, then everything
    42→  // is proportionally scaled so the side sums to exactly 100
    43→  for (const r of list) if (!(r.allocPct > 0)) r.allocPct = 100 / n
    44→  const total = list.reduce((s, r) => s + r.allocPct, 0)
    45→  for (const r of list) r.allocPct = Math.round((r.allocPct / total) * 1000) / 10
    46→  // dump rounding drift on the largest allocation
    47→  const drift = Math.round((100 - list.reduce((s, r) => s + r.allocPct, 0)) * 10) / 10
    48→  if (drift !== 0) {
    49→    const big = list.reduce((a, b) => (b.allocPct > a.allocPct ? b : a))
    50→    big.allocPct = Math.round(Math.max(0, big.allocPct + drift) * 10) / 10
    51→  }
    52→  const fixed = new Map(list.map(r => [r.id, r] as const))
    53→  return rows.map(r => fixed.get(r.id) ?? r)
    54→}
