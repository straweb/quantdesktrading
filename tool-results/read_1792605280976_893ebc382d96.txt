    30→/** Re-scale one side's enabled rows so they always sum to 100%. */
    31→function normalizeSide(rows: StrikeRowCfg[], type: 'CE' | 'PE'): StrikeRowCfg[] {
    32→  const side = rows.filter(r => r.type === type && r.enabled)
    33→  if (side.length === 0) return rows
    34→  const list = side.map(r => ({ ...r }))
    35→  const n = list.length
    36→  // zero/unset allocs enter with an equal-share weight, then everything
    37→  // is proportionally scaled so the side sums to exactly 100
    38→  for (const r of list) if (!(r.allocPct > 0)) r.allocPct = 100 / n
    39→  const total = list.reduce((s, r) => s + r.allocPct, 0)
    40→  for (const r of list) r.allocPct = Math.round((r.allocPct / total) * 1000) / 10
    41→  // dump rounding drift on the largest allocation
    42→  const drift = Math.round((100 - list.reduce((s, r) => s + r.allocPct, 0)) * 10) / 10
    43→  if (drift !== 0) {
    44→    const big = list.reduce((a, b) => (b.allocPct > a.allocPct ? b : a))
    45→    big.allocPct = Math.round(Math.max(0, big.allocPct + drift) * 10) / 10
    46→  }
    47→  const fixed = new Map(list.map(r => [r.id, r] as const))
    48→  return rows.map(r => fixed.get(r.id) ?? r)
    49→}

    51→export function StrikesPanel({ ws }: { ws: WorkspaceCfg }) {
    52→  const patchWs = useTerminal(s => s.patchWs)
    53→  const st: StrikesCfg = useMemo(() => ws.strikes ?? defaultStrikes(), [ws])
    54→
    55→  const { data: undData } = useQuery({
    56→    queryKey: ['optionUnderlyings'],
    57→    queryFn: () => api.optionUnderlyings(),
    58→    staleTime: 5 * 60_000,
    59→  })
    60→  const underlyings = undData?.underlyings ?? []
    61→
    62→  // auto-bind the workspace's F&O symbol on first open
    63→  useEffect(() => {
    64→    if (!st.underlying && underlyings.some(u => u.underlying === ws.symbol)) {
    65→      patchWs(ws.id, { strikes: { ...st, underlying: ws.symbol } })
    66→    }
    67→  }, [st.underlying, underlyings, ws.symbol, ws.id, patchWs]) // eslint-disable-line react-hooks/exhaustive-deps
    68→
    69→  const underlying = st.underlying
    70→  const strikesParam = useMemo(() => [...new Set(st.rows.map(r => r.strike))].sort((a, b) => a - b), [st.rows])
    71→
    72→  const { data: chainData, refetch: refetchChain } = useQuery({
    73→    queryKey: ['optionChain', underlying, st.expiry, strikesParam.join(',')],
    74→    queryFn: () => api.optionChain(underlying!, st.expiry, strikesParam),
    75→    enabled: !!underlying,
    76→    refetchInterval: 1000,
    77→  })
    78→  const meta = chainData?.meta ?? null
    79→  const rowByStrike = useMemo(() => {
    80→    const m = new Map<number, ChainStrikeRow>()
    81→    for (const r of chainData?.rows ?? []) m.set(r.strike, r)
    82→    return m
    83→  }, [chainData])
    84→
    85→  const { data: posData } = useQuery({ queryKey: ['positions'], queryFn: () => api.positions(), refetchInterval: 2000 })
    86→  const posBySymbol = useMemo(() => {
    87→    const m = new Map<string, { netQty: number; avgPrice: number; unrealizedPnl: number }>()
    88→    for (const p of posData?.bySymbol ?? []) m.set(p.symbol, p)
    89→    return m
    90→  }, [posData])

    92→  const [specific, setSpecific] = useState('')
    93→  const [sign, setSign] = useState<1 | -1>(1)
    94→  const [busy, setBusy] = useState(false)
    95→  const [expanded, setExpanded] = useState<string | null>(null)
    96→
    97→  const setStrikes = (next: Partial<StrikesCfg>) => patchWs(ws.id, { strikes: { ...st, ...next } })
    98→
    99→  /** Replace rows + re-normalize the affected side. Callers build rows from the
   100→   *  LATEST persisted state (rapid consecutive toggles must not clobber each
   101→   *  other's change); this re-reads the cfg for the remaining fields. */
   102→  const commitRows = (nextRows: StrikeRowCfg[], type: 'CE' | 'PE') => {
   103→    const cur = useTerminal.getState().workspaces.find(w => w.id === ws.id)?.strikes ?? st
   104→    patchWs(ws.id, { strikes: { ...cur, rows: normalizeSide(nextRows, type) } })
   105→  }
   106→
   107→  const addStrike = async (strike: number, type: 'CE' | 'PE') => {
   108→    if (!underlying) { toast.error('Pick an F&O underlying first'); return }
   109→    const id = `${strike}:${type}`
   110→    // read fresh store state (consecutive/parallel adds must not clobber each other)
   111→    const before = useTerminal.getState().workspaces.find(w => w.id === ws.id)?.strikes ?? st
   112→    if (before.rows.some(r => r.id === id)) { toast.info(`${strike} ${type} already in list`); return }
   113→    try {
   114→      const r = await api.ensureStrike(underlying, strike, st.expiry)
   115→      if (r.created > 0) toast.success(`Contracts ready — ${r.symbols.join(', ')}`)
   116→    } catch (e) {
   117→      toast.error(e instanceof Error ? e.message : 'Failed to create contracts')
   118→      return
   119→    }
   120→    // re-read AFTER the await so an interleaved add is not overwritten
   121→    const cur = useTerminal.getState().workspaces.find(w => w.id === ws.id)?.strikes ?? before
   122→    if (cur.rows.some(r => r.id === id)) { return }
   123→    const rows = normalizeSide([...cur.rows, { id, strike, type, enabled: true, allocPct: 0 }], type)
   124→    patchWs(ws.id, { strikes: { ...cur, rows } })
   125→    setTimeout(() => refetchChain(), 150)
   126→  }

   128→  /** ATM -> CE@ceAtm + PE@peAtm · offset n -> CE/PE n steps from their side ATM.
   129→   *  Sequential (not Promise.all) — parallel adds would race on the store. */
   130→  const addOffset = async (n: number) => {
   131→    if (!meta) { toast.error('Chain not loaded yet'); return }
   132→    if (n === 0) {
   133→      await addStrike(meta.ceAtm, 'CE')
   134→      await addStrike(meta.peAtm, 'PE')
   135→      return
   136→    }
   137→    const ce = meta.ceAtm + sign * n * meta.step
   138→    const pe = meta.peAtm + sign * n * meta.step
   139→    await addStrike(ce, 'CE')
   140→    await addStrike(pe, 'PE')
   141→  }

   143→  const addSpecific = async () => {
   144→    const raw = Number(specific)
   145→    if (!underlying || !Number.isFinite(raw) || raw <= 0) { toast.error('Enter a valid strike'); return }
   146→    const step = meta?.step ?? 50
   147→    const snapped = Math.round(raw / step) * step
   148→    if (snapped !== raw) toast.info(`Snapped ${raw} → ${snapped} (step ${step})`)
   149→    setSpecific('')
   150→    await addStrike(snapped, 'CE')
   151→    await addStrike(snapped, 'PE')
   152→  }

   179→  // --- order helpers ---

   182→  const cellFor = (row: StrikeRowCfg): OptionQuoteCell | null => {

   188→  const deploy = async (type: 'CE' | 'PE') => {
   189→    const rows = rowsFor(type) // ONLY enabled strikes participate
   190→    if (rows.length === 0) { toast.error(`No enabled ${type} strikes`); return }
   191→    setBusy(true)
   192→    let strikesOk = 0, strikesFail = 0, fills = 0, skipped = 0
   193→    for (const row of rows) {
   194→      const cell = cellFor(row)
   195→      if (!cell?.exists || !(row.allocPct > 0)) { strikesFail++; continue }
   196→      try {
   197→        const res = await api.placeOrder({
   198→          symbol: cell.symbol, exchange: 'NFO', side: 'BUY', qty: 0,
   199→          orderType: 'MARKET', productType: 'MIS', source: 'MANUAL', capitalPct: row.allocPct,
   200→        })
   201→        const f = res.legs.filter(l => l.status === 'FILLED').length
   202→        fills += f
   203→        skipped += res.legs.length - f
   204→        f > 0 ? strikesOk++ : strikesFail++
   205→      } catch { strikesFail++ }
   206→    }
   207→    setBusy(false)
   208→    strikesOk > 0
   209→      ? toast.success(`Deploy ${type}: ${strikesOk}/${rows.length} enabled strikes · ${fills} account fills${skipped ? ` · ${skipped} legs skipped` : ''}`)
   210→      : toast.error(`Deploy ${type} failed — no fills (${strikesFail} strikes skipped)`)
   211→  }

   213→  const cancelType = async (type: 'CE' | 'PE') => {
   214→    const syms = rowsFor(type).map(r => cellFor(r)?.symbol).filter((s): s is string => !!s)
   215→    if (syms.length === 0) { toast.error(`No enabled ${type} strikes`); return }
   216→    try {
   217→      const r = await api.cancelAll(syms)
   218→      r.cancelled > 0 ? toast.success(`Cancelled ${r.cancelled} pending ${type} orders`) : toast.info(`No pending ${type} orders`)
   219→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Cancel failed') }
   220→  }

   222→  const peSwap = async () => {
   223→    const ceRows = rowsFor('CE')
   224→    const ceSyms = ceRows.map(r => cellFor(r)?.symbol).filter((s): s is string => !!s)
   225→    setBusy(true)
   226→    try {
   227→      if (ceSyms.length > 0) {
   228→        const so = await api.squareOffAll(false, ceSyms)
   229→        toast.success(`CE squared off — ${so.filledLegs} legs`)
   230→      }
   231→      // flip on: deploy PE on enabled PE rows (auto-add PE ATM if none listed)
   232→      let peRows = rowsFor('PE')
   233→      if (peRows.length === 0) {
   234→        if (meta) {
   235→          await addStrike(meta.peAtm, 'PE')
   236→          await new Promise(r => setTimeout(r, 500))
   237→          const ws2 = useTerminal.getState().workspaces.find(w => w.id === ws.id)
   238→          peRows = (ws2?.strikes?.rows ?? []).filter(r => r.type === 'PE' && r.enabled)
   239→        }
   240→      }
   241→      let strikesOk = 0, strikesFail = 0, fills = 0, skipped = 0
   242→      const allStrikes = [...new Set(st.rows.map(r => r.strike))]
   243→      const cd = await api.optionChain(underlying!, st.expiry, allStrikes)
   244→      const fresh = new Map<number, ChainStrikeRow>(cd.rows.map(r => [r.strike, r]))
   245→      for (const row of peRows) {
   246→        const c = fresh.get(row.strike)
   247→        const cell = row.type === 'PE' ? c?.pe : c?.ce
   248→        if (!cell?.exists) { strikesFail++; continue }
   249→        try {
   250→          const res = await api.placeOrder({
   251→            symbol: cell.symbol, exchange: 'NFO', side: 'BUY', qty: 0,
   252→            orderType: 'MARKET', productType: 'MIS', source: 'MANUAL', capitalPct: row.allocPct,
   253→          })
   254→          const f = res.legs.filter(l => l.status === 'FILLED').length
   255→          fills += f
   256→          skipped += res.legs.length - f
   257→          f > 0 ? strikesOk++ : strikesFail++
   258→        } catch { strikesFail++ }
   259→      }
   260→      strikesOk > 0
   261→        ? toast.success(`PE Swap: CE flat · ${strikesOk}/${peRows.length} enabled PE strikes · ${fills} fills${skipped ? ` · ${skipped} skipped` : ''}`)
   262→        : toast.error(`PE Swap failed — no PE fills`)
   263→    } catch (e) {
   264→      toast.error(e instanceof Error ? e.message : 'PE Swap failed')
   265→    } finally { setBusy(false) }
   266→  }

   268→  /** B = plain market BUY of THIS strike; S = plain market SELL of THIS strike.
   269→   *  Sizing = the row's alloc % of each enabled account's usable capital. */
   270→  const rowOrder = async (row: StrikeRowCfg, side: 'BUY' | 'SELL') => {
   271→    const cell = cellFor(row)
   272→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   273→    try {
   274→      const res = await api.placeOrder({
   275→        symbol: cell.symbol, exchange: 'NFO', side, qty: 0,
   276→        orderType: 'MARKET', productType: 'MIS', source: 'MANUAL', capitalPct: row.allocPct,
   277→      })
   278→      res.status === 'FAILED'
   279→        ? toast.error(`${side} failed — ${res.message}`)
   280→        : toast.success(`${side} ${row.strike} ${row.type} @ MKT — ${res.legs.filter(l => l.status === 'FILLED').length}/${res.legs.length} accounts`)
   281→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Order failed') }
   282→  }
   283→
   284→  const rowExit = async (row: StrikeRowCfg) => {
   285→    const cell = cellFor(row)
   286→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   287→    const pos = posBySymbol.get(cell.symbol)
   288→    if (!pos || pos.netQty === 0) { toast.info('No open position'); return }
   289→    try {
   290→      const so = await api.squareOffAll(false, [cell.symbol])
   291→      so.orders > 0 ? toast.success(`Exit ${row.strike} ${row.type} — ${so.filledLegs} legs`) : toast.info('Nothing to exit')
   292→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Exit failed') }
   293→  }

   297→  /** LIMIT entry at a specific price — auto-sized by the row's alloc % per account. */
   298→  const placeEntry = async (row: StrikeRowCfg, side: 'BUY' | 'SELL', price: number) => {
   299→    const cell = cellFor(row)
   300→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   301→    if (!Number.isFinite(price) || price <= 0) { toast.error('Enter a valid price'); return }
   302→    try {
   303→      const res = await api.placeOrder({
   304→        symbol: cell.symbol, exchange: 'NFO', side, qty: 0,
   305→        orderType: 'LIMIT', productType: 'MIS', source: 'MANUAL',
   306→        capitalPct: row.allocPct, limitPrice: price,
   307→      })
   308→      const f = res.legs.filter(l => l.status === 'FILLED').length
   309→      const r = res.legs.filter(l => l.status === 'OPEN').length
   310→      if (f + r === 0) toast.error(`Entry failed — ${res.message}`)
   311→      else toast.success(`${side} ${row.strike} ${row.type} LIMIT @ ${fmtMoney(price)} — ${f} filled${r ? ` · ${r} resting` : ''}`)
   312→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Order failed') }
   313→  }
   314→
   315→  /** Target — reduce-only LIMIT at the price (books the open position there). */
   316→  const placeTarget = async (row: StrikeRowCfg, price: number) => {
   317→    const cell = cellFor(row)
   318→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   319→    if (!Number.isFinite(price) || price <= 0) { toast.error('Enter a valid price'); return }
   320→    const pos = posBySymbol.get(cell.symbol)
   321→    if (!pos || pos.netQty === 0) { toast.info('No open position to set a target on'); return }
   322→    const side = pos.netQty > 0 ? 'SELL' : 'BUY'
   323→    const lots = Math.abs(pos.netQty) / (meta?.lotSize || 1)
   324→    try {
   325→      const res = await api.placeOrder({
   326→        symbol: cell.symbol, exchange: 'NFO', side, qty: lots,
   327→        orderType: 'LIMIT', productType: 'MIS', source: 'MANUAL',
   328→        limitPrice: price, reduceOnly: true, strategyId: 'reduce',
   329→      })
   330→      const f = res.legs.filter(l => l.status === 'FILLED').length
   331→      const r = res.legs.filter(l => l.status === 'OPEN').length
   332→      if (f + r === 0) toast.error(`Target failed — ${res.message}`)
   333→      else toast.success(`Target: ${side} ${lots} lot(s) LIMIT @ ${fmtMoney(price)} — resting until hit`)
   334→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Order failed') }
   335→  }
   336→
   337→  /** StopLoss — reduce-only SL-M triggered at the price. */
   338→  const placeStop = async (row: StrikeRowCfg, price: number) => {
   339→    const cell = cellFor(row)
   340→    if (!cell?.exists) { toast.error('Quote unavailable'); return }
   341→    if (!Number.isFinite(price) || price <= 0) { toast.error('Enter a valid price'); return }
   342→    const pos = posBySymbol.get(cell.symbol)
   343→    if (!pos || pos.netQty === 0) { toast.info('No open position to protect'); return }
   344→    const side = pos.netQty > 0 ? 'SELL' : 'BUY'
   345→    const lots = Math.abs(pos.netQty) / (meta?.lotSize || 1)
   346→    try {
   347→      const res = await api.placeOrder({
   348→        symbol: cell.symbol, exchange: 'NFO', side, qty: lots,
   349→        orderType: 'SL-M', productType: 'MIS', source: 'MANUAL',
   350→        triggerPrice: price, reduceOnly: true, strategyId: 'reduce',
   351→      })
   352→      const f = res.legs.filter(l => l.status === 'FILLED').length
   353→      const r = res.legs.filter(l => l.status === 'OPEN').length
   354→      if (f + r === 0) toast.error(`SL failed — ${res.message}`)
   355→      else toast.success(`SL: ${side} ${lots} lot(s) SL-M trigger @ ${fmtMoney(price)} — armed`)
   356→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Order failed') }
   357→  }

   359→  const ceRows = useMemo(() => st.rows.filter(r => r.type === 'CE').sort((a, b) => a.strike - b.strike), [st.rows])
   360→  const peRows = useMemo(() => st.rows.filter(r => r.type === 'PE').sort((a, b) => a.strike - b.strike), [st.rows])
   361→  const sideSum = (type: 'CE' | 'PE') =>
   362→    Math.round(st.rows.filter(r => r.type === type && r.enabled).reduce((s, r) => s + r.allocPct, 0) * 10) / 10
   363→
   364→  const offsetFor = (row: StrikeRowCfg): number | null => {
   365→    const chainRow = rowByStrike.get(row.strike)
   366→    if (!chainRow) return null
   367→    return row.type === 'CE' ? chainRow.offsetCe : chainRow.offsetPe
   368→  }

   370→  const renderRow = (row: StrikeRowCfg) => {
   371→    const cell = cellFor(row)
   372→    const offset = offsetFor(row)
   373→    const pos = cell?.exists ? posBySymbol.get(cell.symbol) : undefined
   374→    const lots = pos ? Math.abs(pos.netQty) / (meta?.lotSize || 1) : 0
   375→    const pinned = (row.type === 'CE' ? ws.panes?.find(p => p.key === 'ce') : ws.panes?.find(p => p.key === 'pe'))?.strike === row.strike
   376→    return (

   396→              {row.type}{offset == null ? '' : offset === 0 ? ' ·ATM' : offset > 0 ? ` +${offset}` : ` ${offset}`}
   397→            </span>

   612→function AllocInput({ value, enabled, onCommit }: { value: number; enabled: boolean; onCommit: (v: number) => void }) {
   613→  const [txt, setTxt] = useState(String(value))
   614→  const [focus, setFocus] = useState(false)
   615→  useEffect(() => { if (!focus) setTxt(String(Math.round(value * 10) / 10)) }, [value, focus])
   616→  return (
   617→    <input
   618→      value={txt}
   619→      disabled={!enabled}
   620→      onChange={e => setTxt(e.target.value.replace(/[^0-9.]/g, ''))}
   621→      onFocus={() => setFocus(true)}
   622→      onBlur={() => { setFocus(false); onCommit(Number(txt)) }}
   623→      onKeyDown={e => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   624→      title="Alloc % of usable capital — side auto-normalizes to 100%"
   625→      className="h-4.5 w-9 shrink-0 rounded bg-secondary/60 text-center text-[9px] tabular outline-none focus:ring-1 ring-primary/60 disabled:opacity-40"
   626→    />
   627→  )
   628→}

   634→// ------------------------------------------------------------
   635→function StrikeRowEditor({ row, cell, lots, pos, lotSize, onApply, onExit, onEntry, onTarget, onStop, onRemove }: {
   636→  row: StrikeRowCfg
   637→  cell: OptionQuoteCell | null
   638→  lots: number
   639→  pos: { netQty: number; avgPrice: number; unrealizedPnl: number } | null
   640→  lotSize: number
   641→  onApply: (deltaLots: number) => Promise<void>
   642→  onExit: () => Promise<void>
   643→  onEntry: (side: 'BUY' | 'SELL', price: number) => Promise<void>
   644→  onTarget: (price: number) => Promise<void>
   645→  onStop: (price: number) => Promise<void>
   646→  onRemove: () => void
   647→}) {
   648→  const [delta, setDelta] = useState('')
   649→  const [entry, setEntry] = useState('')
   650→  const [target, setTarget] = useState('')
   651→  const [stop, setStop] = useState('')
   652→  const [busy, setBusy] = useState(false)
   653→  useEffect(() => setDelta(''), [lots])
   654→
   655→  const flat = !pos || pos.netQty === 0
   656→  const num = (s: string) => (s === '' ? NaN : Number(s))
   657→  const ok = (s: string) => Number.isFinite(num(s)) && num(s) > 0
   658→
   659→  const run = async (fn: () => Promise<void>) => {
   660→    setBusy(true)
   661→    try { await fn() } finally { setBusy(false) }
   662→  }
   663→
   664→  return (
   665→    <div className="px-2 pb-2 pt-1 space-y-1.5 border-t border-border/40 bg-secondary/20">
   666→      <div className="flex items-center justify-between text-[10px]">
   667→        <span className="text-muted-foreground">Position — {cell?.symbol ?? row.type}</span>
   668→        <span className="tabular">
   669→          {pos ? (
   670→            <>
   671→              {pos.netQty > 0 ? 'LONG' : 'SHORT'} {lots} lot(s) @ {fmtMoney(pos.avgPrice)} ·{' '}
   672→              <b className={pos.unrealizedPnl >= 0 ? 'term-up' : 'term-down'}>{fmtMoney(pos.unrealizedPnl)}</b>
   673→            </>
   674→          ) : 'flat'}
   675→        </span>
   676→      </div>
   677→
   678→      {/* price orders: entry / target / stoploss */}
   679→      <div className="space-y-1 rounded bg-card/60 p-1.5">
   680→        <div className="text-[8px] font-semibold uppercase tracking-wide text-muted-foreground">orders at price</div>
   681→        <div className="flex items-center gap-1 text-[10px]">
   682→          <span className="w-10 shrink-0 text-muted-foreground">Entry</span>
   683→          <Input
   684→            value={entry}
   685→            onChange={e => setEntry(e.target.value.replace(/[^0-9.]/g, ''))}
   686→            placeholder="limit ₹"
   687→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   688→          />
   689→          <button
   690→            disabled={busy || !ok(entry)}
   691→            onClick={() => run(async () => { await onEntry('BUY', num(entry)); setEntry('') })}
   692→            title={`BUY LIMIT ${row.strike} ${row.type} at this price (${row.allocPct}% per account)`}
   693→            className="px-1.5 h-6 rounded bg-up text-[#00c896] hover:brightness-125 text-[9px] font-bold disabled:opacity-40"
   694→          >B@</button>
   695→          <button
   696→            disabled={busy || !ok(entry)}
   697→            onClick={() => run(async () => { await onEntry('SELL', num(entry)); setEntry('') })}
   698→            title={`SELL LIMIT ${row.strike} ${row.type} at this price (${row.allocPct}% per account)`}
   699→            className="px-1.5 h-6 rounded bg-down text-[#ff4d5e] hover:brightness-125 text-[9px] font-bold disabled:opacity-40"
   700→          >S@</button>
   701→        </div>
   702→        <div className="flex items-center gap-1 text-[10px]">
   703→          <span className="w-10 shrink-0 text-muted-foreground">Target</span>
   704→          <Input
   705→            value={target}
   706→            onChange={e => setTarget(e.target.value.replace(/[^0-9.]/g, ''))}
   707→            placeholder={pos && pos.netQty > 0 ? 'book long ₹' : 'book short ₹'}
   708→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   709→          />
   710→          <button
   711→            disabled={busy || flat || !ok(target)}
   712→            onClick={() => run(async () => { await onTarget(num(target)); setTarget('') })}
   713→            title={`Reduce-only LIMIT at this price — closes the open position when LTP reaches it`}
   714→            className="px-1.5 h-6 rounded bg-secondary text-foreground hover:bg-accent text-[9px] font-bold disabled:opacity-40"
   715→          >Place</button>
   716→        </div>
   717→        <div className="flex items-center gap-1 text-[10px]">
   718→          <span className="w-10 shrink-0 text-muted-foreground">StopLoss</span>
   719→          <Input
   720→            value={stop}
   721→            onChange={e => setStop(e.target.value.replace(/[^0-9.]/g, ''))}
   722→            placeholder="trigger ₹"
   723→            className="h-6 flex-1 min-w-0 text-[10px] tabular bg-secondary/60"
   724→          />
   725→          <button
   726→            disabled={busy || flat || !ok(stop)}
   727→            onClick={() => run(async () => { await onStop(num(stop)); setStop('') })}
   728→            title={`Reduce-only SL-M — market-exits the open position when LTP triggers at this price`}
   729→            className="px-1.5 h-6 rounded bg-secondary text-foreground hover:bg-accent text-[9px] font-bold disabled:opacity-40"
   730→          >Arm</button>
   731→        </div>
   732→        <div className="text-[8px] text-muted-foreground leading-snug">
   733→          Entry rests as LIMIT until hit · Target/SL attach to the current position ({lots} lot(s) · lot {lotSize}) and auto-cancel if it closes
   734→        </div>
   735→      </div>
   736→
   737→      <div className="flex items-center gap-1 text-[10px]">
   738→        <span className="text-muted-foreground">Δ Lots /leg</span>
   739→        <Input
   740→          value={delta}
   741→          onChange={e => setDelta(e.target.value.replace(/[^0-9-]/g, ''))}
   742→          placeholder="+2 / -1"
   743→          className="h-6 w-16 text-[10px] bg-secondary/60"
   744→        />
   745→        <Button
   746→          size="sm" variant="secondary" className="h-6 px-2 text-[10px]" disabled={busy || !delta || delta === '-' || delta === '+0'}
   747→          onClick={async () => { setBusy(true); await onApply(Number(delta)); setBusy(false) }}
   748→        >Apply</Button>
   749→        <span className="text-[9px] text-muted-foreground">+ buys, − exits (per enabled account) · now {lots} lot(s) agg</span>
   750→        <div className="flex-1" />
   751→        <Button
   752→          size="sm" variant="secondary" className="h-6 px-2 text-[10px] text-[#ff4d5e]" disabled={busy || !pos}
   753→          onClick={async () => { setBusy(true); await onExit(); setBusy(false) }}
   754→        >Exit full</Button>
   755→        <Button size="sm" variant="ghost" className="h-6 px-1.5 text-[10px] text-muted-foreground hover:text-[#ff4d5e]" onClick={onRemove}>
   756→          <Trash2 className="h-3 w-3" />
   757→        </Button>
   758→      </div>
   759→      <div className="flex items-center gap-1 text-[10px]">
   760→        <span className="text-[9px] text-muted-foreground">alloc {row.allocPct}% · lot {lotSize} · orders fan out to all enabled accounts</span>
   761→      </div>
   762→    </div>
   763→  )
   764→}
