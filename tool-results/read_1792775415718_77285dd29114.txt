    36→/** Re-scale one side's enabled rows so they always sum to 100%.
    37→ *  PCT mode only — FIXED (₹) allocations are absolute, never normalized. */
    38→export function normalizeSide(rows: StrikeRowCfg[], type: 'CE' | 'PE'): StrikeRowCfg[] {
    39→  const side = rows.filter(r => r.type === type && r.enabled)
    40→  if (side.length === 0) return rows
    41→  const list = side.map(r => ({ ...r }))
    42→  const n = list.length
    43→  // zero/unset allocs enter with an equal-share weight, then everything
    44→  // is proportionally scaled so the side sums to exactly 100
    45→  for (const r of list) if (!(r.allocPct > 0)) r.allocPct = 100 / n
    46→  const total = list.reduce((s, r) => s + r.allocPct, 0)
    47→  for (const r of list) r.allocPct = Math.round((r.allocPct / total) * 1000) / 10
    48→  // dump rounding drift on the largest allocation
    49→  const drift = Math.round((100 - list.reduce((s, r) => s + r.allocPct, 0)) * 10) / 10
    50→  if (drift !== 0) {
    51→    const big = list.reduce((a, b) => (b.allocPct > a.allocPct ? b : a))
    52→    big.allocPct = Math.round(Math.max(0, big.allocPct + drift) * 10) / 10
    53→  }
    54→  const fixed = new Map(list.map(r => [r.id, r] as const))
    55→  return rows.map(r => fixed.get(r.id) ?? r)
    56→}

   751→// ------------------------------------------------------------
   752→// Per-strike allocation cell — % (with inline slider) or fixed ₹.
   753→// Local edit state, commits on blur/Enter/release (re-normalizing
   754→// on every keystroke would fight the typist).
   755→// ------------------------------------------------------------
   756→function AllocCell({ value, mode, enabled, onCommit }: {
   757→  value: number
   758→  mode: AllocMode
   759→  enabled: boolean
   760→  onCommit: (v: number) => void
   761→}) {
   762→  const [txt, setTxt] = useState(mode === 'FIXED' ? String(Math.round(value)) : String(value))
   763→  const [focus, setFocus] = useState(false)
   764→  // eslint-disable-next-line react-hooks/set-state-in-effect -- sync slider -> text when not focused
   765→  useEffect(() => { if (!focus) setTxt(mode === 'FIXED' ? String(Math.round(value)) : String(Math.round(value * 10) / 10)) }, [value, focus, mode])
   766→
   767→  if (mode === 'FIXED') {
   768→    return (
   769→      <input
   770→        value={txt}
   771→        disabled={!enabled}
   772→        onChange={e => setTxt(e.target.value.replace(/[^0-9]/g, ''))}
   773→        onFocus={() => setFocus(true)}
   774→        onBlur={() => { setFocus(false); onCommit(Number(txt)) }}
   775→        onKeyDown={e => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   776→        title="Fixed ₹ allocation per account for this strike"

   944→// ============================================================
   945→// StrikeActions — the QUICK-DEPLOY bar (PnL strip + alloc mode +
   946→// Deploy/Cancel/Swap/Exit-All). Rendered by the right dock BELOW
   947→// the content for EVERY tab (Added Strikes / Next Strikes /
   948→// Auto Bot / Order Pad) so the deploy buttons live in the SAME
   949→// place and stay consistent no matter which tab is selected.
   950→// ============================================================
   951→export function StrikeActions({ ws }: { ws: WorkspaceCfg }) {
   952→  const patchWs = useTerminal(s => s.patchWs)
   953→  const st: StrikesCfg = useMemo(() => ws.strikes ?? defaultStrikes(), [ws])
   954→  const underlying = st.underlying
   955→  const allocMode: AllocMode = ws.allocMode ?? 'PCT'
   956→  const [busy, setBusy] = useState(false)
   957→
   958→  // accounts: representative usable capital (alloc context)
   959→  const coreAcc = useSyncChannel<SyncCore>('sync:core')
   960→  const core = coreAcc
   961→  const accounts = (coreAcc?.accounts as unknown as AccountRow[] | undefined) ?? []
   962→  const repUsable = useMemo(() => {
   963→    const on = accounts.filter(a => a.isOrderEnabled)
   964→    return (on[0] ?? accounts[0])?.usableCapital ?? 0
   965→  }, [accounts])
   966→
   967→  // live chain (same channel the strikes panel uses) — quotes for the rows
   968→  const chainSync = useSyncData<SyncChainResp>(
   969→    'sync:chain',
   970→    d => !!d && (d as { meta?: { underlying?: string } }).meta?.underlying === underlying,
   971→    underlying ? { chains: [{ underlying, expiry: st.expiry ?? null }] } : undefined,
   972→  )
   973→  const rowByStrike = useMemo(() => {
   974→    const m = new Map<number, ChainStrikeRow>()
   975→    for (const r of chainSync?.rows ?? []) m.set(r.strike, r)
   976→    return m
   977→  }, [chainSync])
   978→  const meta = chainSync?.meta ?? null
   979→
   980→  const rowsFor = (type: 'CE' | 'PE') => st.rows.filter(r => r.type === type && r.enabled)
   981→  const cellFor = (row: StrikeRowCfg): OptionQuoteCell | null => {
   982→    const c = rowByStrike.get(row.strike)
   983→    if (!c) return null
   984→    return row.type === 'CE' ? c.ce : c.pe
   985→  }
   986→  const sideSum = (type: 'CE' | 'PE') => {
   987→    const on = st.rows.filter(r => r.type === type && r.enabled)
   988→    if (allocMode === 'FIXED') return on.reduce((s, r) => s + (r.allocAmt ?? 0), 0)
   989→    return Math.round(on.reduce((s, r) => s + r.allocPct, 0) * 10) / 10
   990→  }
   991→  const sizingFor = (row: StrikeRowCfg): Record<string, unknown> =>
   992→    allocMode === 'FIXED'
   993→      ? { tag: `abs:${Math.max(0, row.allocAmt ?? 0)}` }
   994→      : { capitalPct: row.allocPct }
   995→
   996→  // ---- PnL rollup (unrealized + realized, per account) ----
   997→  const posBySymbol = useMemo(() => {
   998→    const m = new Map<string, { netQty: number; unrealizedPnl: number; realizedPnl: number }>()
   999→    for (const p of core?.positions ?? []) {
  1000→      const cur = m.get(p.symbol)
  1001→      if (!cur) { m.set(p.symbol, { netQty: p.netQty, unrealizedPnl: p.unrealizedPnl, realizedPnl: p.realizedPnl }); continue }
  1002→      cur.netQty += p.netQty
  1003→      cur.unrealizedPnl += p.unrealizedPnl
  1004→      cur.realizedPnl += p.realizedPnl
  1005→    }
  1006→    return m
  1007→  }, [core])
  1008→  const pnl = useMemo(() => {
  1009→    let totalU = 0, totalR = 0
  1010→    const byAccount = new Map<string, { label: string; u: number; r: number }>()
  1011→    for (const p of core?.positions ?? []) {
  1012→      totalU += p.unrealizedPnl
  1013→      totalR += p.realizedPnl
  1014→      const a = byAccount.get(p.accountId) ?? { label: p.accountLabel, u: 0, r: 0 }
  1015→      a.u += p.unrealizedPnl
  1016→      a.r += p.realizedPnl
  1017→      byAccount.set(p.accountId, a)
  1018→    }
  1019→    return { totalU, totalR, byAccount }
  1020→  }, [core])
  1021→
  1022→  /** Deploy every enabled strike of one side (fresh quotes, never stale). */
  1023→  const deployRows = async (type: 'CE' | 'PE', rows: StrikeRowCfg[]) => {
  1024→    let strikesOk = 0, strikesFail = 0, fills = 0, skipped = 0
  1025→    const allStrikes = [...new Set(rows.map(r => r.strike))]
  1026→    const cd = allStrikes.length ? await api.optionChain(underlying!, st.expiry, allStrikes) : null
  1027→    const fresh = new Map<number, ChainStrikeRow>((cd?.rows ?? []).map(r => [r.strike, r]))
  1028→    for (const row of rows) {
  1029→      const c = fresh.get(row.strike)
  1030→      const cell = c ? (row.type === 'CE' ? c.ce : c.pe) : cellFor(row)
  1031→      const alloc = allocMode === 'FIXED' ? (row.allocAmt ?? 0) : row.allocPct
  1032→      if (!cell?.exists || !(alloc > 0)) { strikesFail++; continue }
  1033→      try {
  1034→        const res = await api.placeOrder({
  1035→          symbol: cell.symbol, exchange: 'NFO', side: 'BUY', qty: 0,
  1036→          orderType: 'MARKET', productType: 'MIS', source: 'MANUAL',
  1037→          ...sizingFor(row),
  1038→        })
  1039→        const f = res.legs.filter(l => l.status === 'FILLED').length
  1040→        fills += f
  1041→        skipped += res.legs.length - f
  1042→        f > 0 ? strikesOk++ : strikesFail++
  1043→      } catch { strikesFail++ }
  1044→    }
  1045→    return { strikesOk, strikesFail, fills, skipped, total: rows.length }
  1046→  }
  1047→
  1048→  const deploy = async (type: 'CE' | 'PE') => {
  1049→    const rows = rowsFor(type)
  1050→    if (rows.length === 0) { toast.error(`No enabled ${type} strikes`); return }
  1051→    setBusy(true)
  1052→    const r = await deployRows(type, rows)
  1053→    setBusy(false)
  1054→    r.strikesOk > 0
  1055→      ? toast.success(`Deploy ${type}: ${r.strikesOk}/${rows.length} enabled strikes · ${r.fills} account fills${r.skipped ? ` · ${r.skipped} legs skipped` : ''}`)
  1056→      : toast.error(`Deploy ${type} failed — no fills (${r.strikesFail} strikes skipped)`)
  1057→  }
  1058→
  1059→  const cancelType = async (type: 'CE' | 'PE') => {
  1060→    const syms = rowsFor(type).map(r => cellFor(r)?.symbol).filter((s): s is string => !!s)
  1061→    if (syms.length === 0) { toast.error(`No enabled ${type} strikes`); return }
  1062→    try {
  1063→      const r = await api.cancelAll(syms)
  1064→      r.cancelled > 0 ? toast.success(`Cancelled ${r.cancelled} pending ${type} orders`) : toast.info(`No pending ${type} orders`)
  1065→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Cancel failed') }
  1066→  }
  1067→
  1068→  /** SWAP: close side `from`, then deploy the opposite side (auto-adds the
  1069→   *  target side's ATM if nothing is listed). */
  1070→  const swapSide = async (from: 'CE' | 'PE') => {
  1071→    const to = from === 'CE' ? 'PE' : 'CE'
  1072→    const fromSyms = rowsFor(from).map(r => cellFor(r)?.symbol).filter((s): s is string => !!s)
  1073→    setBusy(true)
  1074→    try {
  1075→      let closedLegs = 0
  1076→      if (fromSyms.length > 0) {
  1077→        const so = await api.squareOffAll(false, fromSyms)
  1078→        closedLegs = so.filledLegs
  1079→      }
  1080→      let toRows = rowsFor(to)
  1081→      if (toRows.length === 0 && meta) {
  1082→        await ensureStrikeAdd(ws, to === 'CE' ? meta.ceAtm : meta.peAtm, to)
  1083→        await new Promise(r => setTimeout(r, 500))
  1084→        const ws2 = useTerminal.getState().workspaces.find(w => w.id === ws.id)
  1085→        toRows = (ws2?.strikes?.rows ?? []).filter(r => r.type === to && r.enabled)
  1086→      }
  1087→      const r = await deployRows(to, toRows)
  1088→      r.strikesOk > 0
  1089→        ? toast.success(`Swap ${from}→${to}: ${from} flat (${closedLegs} legs) · ${r.strikesOk}/${toRows.length} ${to} strikes · ${r.fills} fills${r.skipped ? ` · ${r.skipped} skipped` : ''}`)
  1090→        : toast.error(`Swap ${from}→${to} — ${from} flat but no ${to} fills`)
  1091→    } catch (e) {
  1092→      toast.error(e instanceof Error ? e.message : `Swap ${from}→${to} failed`)
  1093→    } finally { setBusy(false) }
  1094→  }
  1095→
  1096→  const exitAll = async () => {
  1097→    setBusy(true)
  1098→    try {
  1099→      const so = await api.squareOffAll(false)
  1100→      so.orders > 0
  1101→        ? toast.success(`Exit All — ${so.orders} orders · ${so.filledLegs} legs closed`)
  1102→        : toast.info('No open positions')
  1103→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Exit All failed') }
  1104→    finally { setBusy(false) }
  1105→  }
  1106→
  1107→  return (
  1108→    <div className="border-t border-border p-1.5 space-y-1 bg-card/60 shrink-0">
  1109→      {/* session PnL strip — total / net / per account */}
  1110→      <div className="rounded bg-secondary/30 px-1.5 py-1 space-y-0.5" title="Total PnL = open (unrealized) + booked (realized) · Net = unrealized + realized">
  1111→        <div className="flex items-center justify-between text-[9px] tabular">
  1112→          <span className="text-muted-foreground">Open <b className={pnl.totalU >= 0 ? 'term-up' : 'term-down'}>{fmtMoney(pnl.totalU, 0)}</b></span>
  1113→          <span className="text-muted-foreground">Booked <b className={pnl.totalR >= 0 ? 'term-up' : 'term-down'}>{fmtMoney(pnl.totalR, 0)}</b></span>
  1114→          <span className="text-muted-foreground">Net <b className={pnl.totalU + pnl.totalR >= 0 ? 'term-up' : 'term-down'}>{fmtMoney(pnl.totalU + pnl.totalR, 0)}</b></span>
  1115→          <span className="text-muted-foreground" title="Representative usable capital for % sizing">cap {fmtMoney(repUsable, 0)}</span>
  1116→        </div>
  1117→        {pnl.byAccount.size > 0 && (
  1118→          <div className="flex flex-wrap gap-x-2 gap-y-0.5 text-[8px] tabular">
  1119→            {[...pnl.byAccount.values()].map(a => {
  1120→              const net = a.u + a.r
  1121→              return (
  1122→                <span key={a.label} className="text-muted-foreground">
  1123→                  {a.label} <b className={net >= 0 ? 'term-up' : 'term-down'}>{net >= 0 ? '+' : ''}{fmtMoney(net, 0)}</b>
  1124→                </span>
  1125→              )
  1126→            })}
  1127→          </div>
  1128→        )}
  1129→      </div>
  1130→      <div className="flex items-center justify-between text-[8px] text-muted-foreground px-0.5">
  1131→        <span>enabled strikes deploy at their alloc</span>
  1132→        <span>
  1133→          {allocMode === 'FIXED' ? (
  1134→            <><b className="text-[#00c896]">CE ₹{sideSum('CE').toLocaleString('en-IN')}</b> · <b className="text-[#ff4d5e]">PE ₹{sideSum('PE').toLocaleString('en-IN')}</b></>
  1135→          ) : (
  1136→            <><b className="text-[#00c896]">CE Σ{sideSum('CE')}%</b> · <b className="text-[#ff4d5e]">PE Σ{sideSum('PE')}%</b></>
  1137→          )}
  1138→        </span>
  1139→        <div className="flex rounded overflow-hidden border border-border">
  1140→          {(['PCT', 'FIXED'] as AllocMode[]).map(m => (
  1141→            <button
  1142→              key={m}
  1143→              onClick={() => patchWs(ws.id, { allocMode: m })}
  1144→              title={m === 'PCT' ? '% of each account\'s usable capital (auto-normalized)' : 'Fixed ₹ per strike per account'}
  1145→              className={`px-1.5 py-0.5 text-[8px] font-bold ${allocMode === m ? 'bg-primary text-primary-foreground' : 'bg-secondary text-muted-foreground hover:text-foreground'}`}
  1146→            >{m === 'PCT' ? '%' : '₹'}</button>
  1147→          ))}
  1148→        </div>
  1149→      </div>
  1150→      <div className="grid grid-cols-2 gap-1">
  1151→        <Button size="sm" className="h-7 text-[10px] bg-up text-[#00c896] hover:bg-[#00c896] hover:text-black" disabled={busy || !underlying} onClick={() => deploy('CE')}>
  1152→          Deploy CE
  1153→        </Button>
  1154→        <Button size="sm" className="h-7 text-[10px] bg-down text-[#ff4d5e] hover:bg-[#ff4d5e] hover:text-black" disabled={busy || !underlying} onClick={() => deploy('PE')}>
  1155→          Deploy PE
  1156→        </Button>
  1157→        <Button size="sm" variant="secondary" className="h-7 text-[10px]" disabled={busy || !underlying} onClick={() => cancelType('CE')}>
  1158→          Cancel CE
  1159→        </Button>

  1177→// ============================================================
  1178→// Shared add-strike helper — ensure contracts exist, then append
  1179→// the row (side-normalized).  Used by the strikes desk AND the
  1180→// Next-expiry browser tab.
  1181→// ============================================================
  1182→export async function ensureStrikeAdd(ws: WorkspaceCfg, strike: number, type: 'CE' | 'PE', expiryOverride?: string | null) {
  1183→  const st0 = useTerminal.getState().workspaces.find(w => w.id === ws.id)?.strikes ?? ws.strikes ?? defaultStrikes()
  1184→  const underlying = st0.underlying
  1185→  if (!underlying) { toast.error('Pick an F&O underlying first'); return false }
  1186→  const id = `${strike}:${type}`
  1187→  if (st0.rows.some(r => r.id === id)) { toast.info(`${strike} ${type} already in list`); return false }
  1188→  try {
  1189→    const r = await api.ensureStrike(underlying, strike, expiryOverride ?? st0.expiry)
  1190→    if (r.created > 0) toast.success(`Contracts ready — ${r.symbols.join(', ')}`)
  1191→  } catch (e) {
  1192→    toast.error(e instanceof Error ? e.message : 'Failed to create contracts')
  1193→    return false
  1194→  }
  1195→  const cur = useTerminal.getState().workspaces.find(w => w.id === ws.id)?.strikes ?? st0
  1196→  if (cur.rows.some(r => r.id === id)) return false
  1197→  const rows = normalizeSide([...cur.rows, { id, strike, type, enabled: true, allocPct: 0 }], type)
  1198→  useTerminal.getState().patchWs(ws.id, { strikes: { ...cur, rows } })
  1199→  return true
  1200→}

  1202→// ============================================================
  1203→// NextStrikes — NEXT-expiry strike browser tab.  Shows the ATM±8
  1204→// chain of the NEXT expiry with live CE/PE LTPs (and delta when an
  1205→// FNO session provides greeks); "+CE / +PE" adds that strike to the
  1206→// added-strikes desk and switches the desk to this expiry so quotes
  1207→// resolve.
  1208→// ============================================================
  1209→export function NextStrikes({ ws }: { ws: WorkspaceCfg }) {
  1210→  const patchWs = useTerminal(s => s.patchWs)
  1211→  const st: StrikesCfg = useMemo(() => ws.strikes ?? defaultStrikes(), [ws])
  1212→  const underlying = st.underlying
  1213→  const [adding, setAdding] = useState<string | null>(null)
  1214→
  1215→  const { data: restChain } = useQuery({
  1216→    queryKey: ['optionChainNext', underlying, st.expiry],
  1217→    queryFn: () => api.optionChain(underlying!, null, []),
  1218→    enabled: !!underlying,
  1219→    refetchInterval: 4000,
  1220→    retry: false,
  1221→  })
  1222→  const expiries = restChain?.meta.expiries ?? []
  1223→  const curIdx = st.expiry ? expiries.indexOf(st.expiry) : 0
  1224→  const nextExpiry = expiries.length > 1
  1225→    ? expiries[Math.min(expiries.length - 1, (curIdx < 0 ? 0 : curIdx) + 1)]
  1226→    : null
  1227→  const { data: nextChain } = useQuery({
  1228→    queryKey: ['optionChainNextRows', underlying, nextExpiry],
  1229→    queryFn: () => api.optionChain(underlying!, nextExpiry, []),
  1230→    enabled: !!underlying && !!nextExpiry,
  1231→    refetchInterval: 4000,
  1232→    retry: false,
  1233→  })
  1234→
  1235→  const switchToNext = () => {
  1236→    if (!nextExpiry) return
  1237→    const cur = useTerminal.getState().workspaces.find(w => w.id === ws.id)?.strikes ?? st
  1238→    patchWs(ws.id, { strikes: { ...cur, expiry: nextExpiry } })
  1239→    toast.info(`Strikes desk switched to expiry ${nextExpiry}`)
  1240→  }
  1241→
  1242→  const add = async (strike: number, type: 'CE' | 'PE') => {
  1243→    if (!nextExpiry) return
  1244→    setAdding(`${strike}:${type}`)
  1245→    const cur = useTerminal.getState().workspaces.find(w => w.id === ws.id)?.strikes ?? st
  1246→    if (cur.expiry !== nextExpiry) {
  1247→      patchWs(ws.id, { strikes: { ...cur, expiry: nextExpiry } })
  1248→    }
  1249→    await ensureStrikeAdd(ws, strike, type, nextExpiry)
  1250→    setAdding(null)
  1251→  }
  1252→
  1253→  if (!underlying) {
  1254→    return (
  1255→      <div className="p-3 text-center text-[10px] text-muted-foreground">
  1256→        Pick an F&O underlying in the Added Strikes tab first.
  1257→      </div>
  1258→    )
  1259→  }
  1260→  const rows = (nextChain?.rows ?? []).filter(r => Math.abs(r.offset) <= 8)
  1261→  const meta = nextChain?.meta
  1262→  return (
  1263→    <div className="h-full flex flex-col text-[11px]">
  1264→      <div className="px-1.5 py-1 border-b border-border shrink-0 space-y-1">
  1265→        <div className="flex items-center justify-between">
  1266→          <span className="text-[10px] font-bold">{underlying} · next expiry</span>
  1267→          {nextExpiry && (
  1268→            <button onClick={switchToNext} title="Load this expiry into the strikes desk selector"
  1269→              className="px-1.5 py-0.5 rounded bg-primary/25 text-[9px] font-bold hover:bg-primary/40">
  1270→              use {nextExpiry}
  1271→            </button>
  1272→          )}
  1273→        </div>
  1274→        <div className="flex items-center justify-between text-[9px] text-muted-foreground tabular">
  1275→          <span>{nextExpiry ? `exp ${nextExpiry}` : 'no next expiry in the master'}</span>
  1276→          <span>spot <b className={meta && meta.spot >= meta.prevClose ? 'term-up' : 'term-down'}>{meta ? fmtMoney(meta.spot) : '—'}</b></span>
  1277→          <span>C{meta?.ceAtm ?? '—'} · P{meta?.peAtm ?? '—'}</span>
  1278→        </div>
  1279→        <div className="text-[9px] text-muted-foreground">
  1280→          +CE / +PE adds the strike to the added desk and switches it to this expiry.
  1281→        </div>
  1282→      </div>
  1283→      <div className="flex-1 min-h-0 overflow-y-auto">
  1284→        {!nextExpiry && (
  1285→          <div className="p-3 text-center text-[10px] text-muted-foreground">This underlying has a single expiry — nothing to browse.</div>
  1286→        )}
  1287→        {nextExpiry && rows.length === 0 && (
  1288→          <div className="p-3 text-center text-[10px] text-muted-foreground">loading next-expiry chain…</div>
  1289→        )}
  1290→        {rows.map(r => {

  1292→          const addedPe = st.rows.some(x => x.id === `${r.strike}:PE`)
  1293→          return (

  1318→                onClick={() => add(r.strike, 'PE')}

  1320→                className={`px-1 py-0.5 rounded text-[8px] font-bold ${addedPe ? 'bg-primary/25 text-foreground' : 'bg-down text-[#ff4d5e] hover:brightness-125'}`}
