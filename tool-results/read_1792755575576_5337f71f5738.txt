    35→    35→
    36→    36→/** Re-scale one side's enabled rows so they always sum to 100%.
    37→    37→ *  PCT mode only — FIXED (₹) allocations are absolute, never normalized. */
    38→    38→function normalizeSide(rows: StrikeRowCfg[], type: 'CE' | 'PE'): StrikeRowCfg[] {
    39→    39→  const side = rows.filter(r => r.type === type && r.enabled)
    40→    40→  if (side.length === 0) return rows
    41→    41→  const list = side.map(r => ({ ...r }))
    42→    42→  const n = list.length
    43→    43→  // zero/unset allocs enter with an equal-share weight, then everything
    44→    44→  // is proportionally scaled so the side sums to exactly 100
    45→    45→  for (const r of list) if (!(r.allocPct > 0)) r.allocPct = 100 / n
    46→    46→  const total = list.reduce((s, r) => s + r.allocPct, 0)
    47→    47→  for (const r of list) r.allocPct = Math.round((r.allocPct / total) * 1000) / 10
    48→    48→  // dump rounding drift on the largest allocation
    49→    49→  const drift = Math.round((100 - list.reduce((s, r) => s + r.allocPct, 0)) * 10) / 10
    50→    50→  if (drift !== 0) {
    51→    51→    const big = list.reduce((a, b) => (b.allocPct > a.allocPct ? b : a))
    52→    52→    big.allocPct = Math.round(Math.max(0, big.allocPct + drift) * 10) / 10
    53→    53→  }
    54→    54→  const fixed = new Map(list.map(r => [r.id, r] as const))
    55→    55→  return rows.map(r => fixed.get(r.id) ?? r)
    56→    56→}
